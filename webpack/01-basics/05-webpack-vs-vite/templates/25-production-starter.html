// Webpack vs Vite — Template 25
// Focus: production starter | Scenario: admin portal
const path = require("path");
const HtmlWebpackPlugin = require("html-webpack-plugin");
const MiniCssExtractPlugin = require("mini-css-extract-plugin");
const webpack = require("webpack");

const isProd = process.env.NODE_ENV === "production";

module.exports = {
  mode: isProd ? "production" : "development",
  entry: "./src/index.js",
  output: {
    path: path.resolve(__dirname, "dist"),
    filename: isProd ? "js/[name].[contenthash:8].js" : "js/[name].js",
    clean: true,
    publicPath: "/",
  },
  devtool: isProd ? "source-map" : "eval-cheap-module-source-map",
  resolve: { extensions: [".js", ".json"], alias: { "@": path.resolve(__dirname, "src") } },
  module: {
    rules: [
      { test: /\.m?js$/, exclude: /node_modules/, use: "babel-loader" },
      {
        test: /\.s?css$/,
        use: [isProd ? MiniCssExtractPlugin.loader : "style-loader", "css-loader", "sass-loader"],
      },
      {
        test: /\.(png|jpe?g|gif|svg|woff2?)$/i,
        type: "asset",
        parser: { dataUrlCondition: { maxSize: 8 * 1024 } },
      },
    ],
  },
  plugins: [
    new HtmlWebpackPlugin({ template: "./public/index.html" }),
    new webpack.DefinePlugin({
      "process.env.APP_FOCUS": JSON.stringify("production starter"),
      "process.env.APP_SCENARIO": JSON.stringify("admin portal"),
    }),
    ...(isProd ? [new MiniCssExtractPlugin({ filename: "css/[name].[contenthash:8].css" })] : []),
  ],
  optimization: { splitChunks: { chunks: "all" }, runtimeChunk: "single" },
  devServer: { static: "./dist", hot: true, historyApiFallback: true, port: 8080 },
  cache: { type: "filesystem" },
};

Practice Notes

Checklist for Webpack vs Vite
- Practice: production starter
- Scenario: admin portal
- Share a webpack.common.js and merge dev/prod with webpack-merge
- Prefer asset modules over file-loader/url-loader
- Use contenthash + splitChunks for long-term caching
- Measure with webpack-bundle-analyzer before optimizing
- Webpack 5: use output.clean instead of clean-webpack-plugin when possible

Useful commands:
  npm install -D webpack webpack-cli webpack-dev-server html-webpack-plugin
  npx webpack --mode development
  npx webpack serve --mode development
  NODE_ENV=production npx webpack --mode production
  npx webpack --json=stats.json
