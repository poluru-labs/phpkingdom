const CopyWebpackPlugin = require("copy-webpack-plugin");

module.exports = {
  plugins: [
    new CopyWebpackPlugin({
      patterns: [{ from: "public", to: ".", globOptions: { ignore: ["**/index.html"] } }],
    }),
  ],
};

Practice Notes

Checklist for Static Assets
- Practice: asset modules
- Scenario: CI production build
- Share a webpack.common.js and merge dev/prod with webpack-merge
- Prefer asset modules over file-loader/url-loader
- Use contenthash + splitChunks for long-term caching
- Measure with webpack-bundle-analyzer before optimizing
- Webpack 5: use output.clean instead of clean-webpack-plugin when possible

Useful commands:
  npm install -D webpack webpack-cli webpack-dev-server html-webpack-plugin
  npx webpack --mode development
  npx webpack serve --mode development
  NODE_ENV=production npx webpack --mode production
  npx webpack --json=stats.json
