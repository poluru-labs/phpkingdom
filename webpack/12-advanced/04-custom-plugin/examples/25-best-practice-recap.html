// Custom Plugin — Example 25
// Focus: best practice recap | Scenario: admin portal
class HelloPlugin {
  apply(compiler) {
    compiler.hooks.done.tap("HelloPlugin", (stats) => {
      console.log("build done", stats.hasErrors() ? "with errors" : "ok");
    });
  }
}
module.exports = { plugins: [new HelloPlugin()] };
