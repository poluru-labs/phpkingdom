// jest.fn() — Example 13
// Focus: mockReturnValue | Scenario: Vue app

test("jest.fn creates a mock function", () => {
  const fn = jest.fn();
  fn("a");
  fn("b", 2);

  expect(fn).toHaveBeenCalled();
  expect(fn).toHaveBeenCalledTimes(2);
  expect(fn).toHaveBeenNthCalledWith(2, "b", 2);
  expect(fn.mock.calls[0]).toEqual(["a"]);
});

test("jest.fn with initial implementation", () => {
  const add = jest.fn((a, b) => a + b);
  expect(add(2, 3)).toBe(5);
});
