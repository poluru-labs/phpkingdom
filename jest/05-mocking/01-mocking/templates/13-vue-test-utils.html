// Jest Mocking — Template 13
// Focus: vue test utils | Scenario: Vue app

// Mocking replaces real collaborators at test boundaries
const mailer = {
  send: jest.fn().mockResolvedValue({ queued: true }),
};

async function register(email) {
  await mailer.send({ to: email, template: "welcome" });
  return { email, status: "pending" };
}

test("register queues welcome email", async () => {
  await expect(register("ada@phpkingdom.com")).resolves.toMatchObject({
    status: "pending",
  });
  expect(mailer.send).toHaveBeenCalledWith(
    expect.objectContaining({ to: "ada@phpkingdom.com" })
  );
});

Practice Notes

Checklist for Jest Mocking
- Practice: vue test utils
- Scenario: Vue app
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating

Useful commands:
  npm install --save-dev jest
  npx jest
  npx jest --watch
  npx jest --coverage
  npx jest -u
