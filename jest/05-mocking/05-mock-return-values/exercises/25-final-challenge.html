test("mockReturnValue and Once variants", () => {
  const dice = jest.fn();
  dice.mockReturnValue(6);
  expect(dice()).toBe(6);
  expect(dice()).toBe(6);

  dice.mockReturnValueOnce(1).mockReturnValueOnce(2).mockReturnValue(3);
  expect(dice()).toBe(1);
  expect(dice()).toBe(2);
  expect(dice()).toBe(3);
});

test("mockResolvedValue for promises", async () => {
  const fetchUser = jest.fn().mockResolvedValue({ id: 7 });
  await expect(fetchUser()).resolves.toEqual({ id: 7 });
});
