Starter

// jest.config.ts
import type { Config } from "jest";

const config: Config = {
  preset: "ts-jest",
  testEnvironment: "node",
  roots: ["<rootDir>/src"],
};
export default config;

// src/greet.ts
export function greet(name: string): string {
  return `Hello, ${name}`;
}

// src/greet.test.ts
import { greet } from "./greet";

test("greets", () => {
  expect(greet("Ada")).toBe("Hello, Ada");
});
