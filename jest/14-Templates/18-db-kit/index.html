Starter

const users = {
  findById: jest.fn(),
  save: jest.fn(),
};

async function rename(id, name) {
  const row = await users.findById(id);
  if (!row) throw new Error("missing");
  row.name = name;
  return users.save(row);
}

test("repository mock", async () => {
  users.findById.mockResolvedValue({ id: 1, name: "Ada" });
  users.save.mockImplementation(async (u) => u);
  await expect(rename(1, "Grace")).resolves.toEqual({ id: 1, name: "Grace" });
});
