Starter

function card(title) {
  return { type: "card", props: { title, site: "phpkingdom.com" } };
}

test("card snapshot", () => {
  expect(card("Home")).toMatchSnapshot();
});

test("inline", () => {
  expect(card("Sale")).toMatchInlineSnapshot(`
    {
      "props": {
        "site": "phpkingdom.com",
        "title": "Sale",
      },
      "type": "card",
    }
  `);
});
