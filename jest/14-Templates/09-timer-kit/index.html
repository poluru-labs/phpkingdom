Advance timers

Starter

jest.useFakeTimers();

test("advanceTimersByTime", () => {
  const fn = jest.fn();
  setTimeout(fn, 2000);
  jest.advanceTimersByTime(2000);
  expect(fn).toHaveBeenCalledTimes(1);
});

afterAll(() => {
  jest.useRealTimers();
});

Run

npx jest
npx jest --watch
npx jest --coverage
