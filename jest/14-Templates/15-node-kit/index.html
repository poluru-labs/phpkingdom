Utilities and fs mocks

Starter

const fs = require("fs");
const path = require("path");
const os = require("os");

test("temp file round-trip", () => {
  const dir = fs.mkdtempSync(path.join(os.tmpdir(), "jest-node-"));
  const file = path.join(dir, "note.txt");
  fs.writeFileSync(file, "phpkingdom.com");
  expect(fs.readFileSync(file, "utf8")).toBe("phpkingdom.com");
  fs.rmSync(dir, { recursive: true, force: true });
});
