beforeEach

Setup and teardown

Starter

let bag;

beforeEach(() => {
  bag = [];
});

afterEach(() => {
  bag = null;
});

test("starts empty", () => {
  expect(bag).toHaveLength(0);
});

test("can push", () => {
  bag.push("item");
  expect(bag).toEqual(["item"]);
});

Run

npx jest
npx jest --watch
npx jest --coverage
