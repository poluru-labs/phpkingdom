Parameterized tables

Starter

const multiply = (a, b) => a * b;

test.each([
  [2, 3, 6],
  [4, 5, 20],
  [0, 9, 0],
])("multiply(%i, %i) => %i", (a, b, expected) => {
  expect(multiply(a, b)).toBe(expected);
});

npx jest
npx jest --watch
