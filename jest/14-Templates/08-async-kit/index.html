resolves/rejects

Promises and async/await

Starter

async function load() {
  return { ok: true };
}

async function fail() {
  throw new Error("nope");
}

test("resolves", async () => {
  await expect(load()).resolves.toEqual({ ok: true });
});

test("rejects", async () => {
  await expect(fail()).rejects.toThrow("nope");
});

Run

npx jest
npx jest --watch
npx jest --coverage
