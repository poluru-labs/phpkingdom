Starter

import { jest } from "@jest/globals";

jest.unstable_mockModule("./logger.js", () => ({
  info: jest.fn(),
}));

const { info } = await import("./logger.js");
const { runJob } = await import("./job.js");

test("esm mock", async () => {
  runJob("deploy");
  expect(info).toHaveBeenCalledWith("deploy");
});
