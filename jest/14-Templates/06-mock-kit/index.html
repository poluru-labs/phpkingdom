Starter

test("jest.fn + spyOn", () => {
  const send = jest.fn().mockReturnValue(true);
  expect(send("hi")).toBe(true);
  expect(send).toHaveBeenCalledWith("hi");

  const spy = jest.spyOn(Math, "abs").mockReturnValue(9);
  expect(Math.abs(-3)).toBe(9);
  spy.mockRestore();
});
