test("jsdom document", () => {
  document.body.innerHTML = `<button id="go">Go</button>`;
  const btn = document.getElementById("go");
  expect(btn.textContent).toBe("Go");
  btn.click();
  expect(document.querySelectorAll("button")).toHaveLength(1);
});
