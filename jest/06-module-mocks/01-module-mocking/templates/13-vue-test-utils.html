// Module Mocking — Template 13
// Focus: vue test utils | Scenario: Vue app

// userService.js depends on ./api
// jest.mock hoists and replaces the module

jest.mock("./api", () => ({
  getUser: jest.fn(() => Promise.resolve({ id: 1, name: "Ada" })),
}));

const { getUser } = require("./api");
const { loadProfile } = require("./userService");

test("loadProfile uses mocked api", async () => {
  getUser.mockResolvedValueOnce({ id: 2, name: "Grace" });
  await expect(loadProfile(2)).resolves.toMatchObject({ name: "Grace" });
  expect(getUser).toHaveBeenCalledWith(2);
});

Practice Notes

Checklist for Module Mocking
- Practice: vue test utils
- Scenario: Vue app
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating

Useful commands:
  npm install --save-dev jest
  npx jest
  npx jest --watch
  npx jest --coverage
  npx jest -u
