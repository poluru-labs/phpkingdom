// Mock ES Modules — Template 24
// Focus: esm mock | Scenario: Express API

// ESM mocking (experimental / node --experimental-vm-modules)
import { jest } from "@jest/globals";

jest.unstable_mockModule("./logger.js", () => ({
  log: jest.fn(),
}));

const { log } = await import("./logger.js");
const { run } = await import("./job.js");

test("esm module mock", async () => {
  run("deploy");
  expect(log).toHaveBeenCalledWith("deploy");
});
// Prefer CJS jest.mock when the codebase still uses require()
