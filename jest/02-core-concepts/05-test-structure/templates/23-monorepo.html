// Test Structure — Template 23
// Focus: monorepo | Scenario: Vue app

function parseId(raw) {
  const n = Number(raw);
  if (!Number.isInteger(n) || n <= 0) throw new Error("invalid id");
  return n;
}

describe("parseId", () => {
  // Arrange
  const raw = "42";

  test("happy path", () => {
    // Act
    const id = parseId(raw);
    // Assert
    expect(id).toBe(42);
  });

  test("rejects invalid", () => {
    expect(() => parseId("x")).toThrow("invalid id");
  });
});

Practice Notes

Checklist for Test Structure
- Practice: monorepo
- Scenario: Vue app
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating
