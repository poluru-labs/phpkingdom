// Mock Service Worker — Example 12
// Focus: spyOn method | Scenario: React app

// Mock Service Worker intercepts HTTP in Node/jsdom
import { setupServer } from "msw/node";
import { http, HttpResponse } from "msw";

const server = setupServer(
  http.get("/api/user", () => HttpResponse.json({ id: 1, name: "Ada" }))
);

beforeAll(() => server.listen());
afterEach(() => server.resetHandlers());
afterAll(() => server.close());

test("fetch hits MSW handler", async () => {
  const res = await fetch("/api/user");
  await expect(res.json()).resolves.toEqual({ id: 1, name: "Ada" });
});
// Great for testing clients without stubbing fetch itself
