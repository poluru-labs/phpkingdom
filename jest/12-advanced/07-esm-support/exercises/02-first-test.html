// Jest ESM Support — Exercise 02
// Focus: first test | Scenario: React app

// package.json: "type": "module"
// jest.config.js (or jest.config.cjs):
/** @type {import('jest').Config} */
export default {
  testEnvironment: "node",
  transform: {},
  // Node 18+: node --experimental-vm-modules node_modules/jest/bin/jest.js
};

// math.mjs
export function sum(a, b) {
  return a + b;
}

// math.test.mjs
import { sum } from "./math.mjs";

test("esm import works", () => {
  expect(sum(2, 3)).toBe(5);
});

Practice Notes

Checklist for Jest ESM Support
- Practice: first test
- Scenario: React app
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating

Useful commands:
  npm install --save-dev jest
  npx jest
  npx jest --watch
  npx jest --coverage
  npx jest -u
