// Test Doubles — Exercise 05
// Focus: config tweak | Scenario: Node CLI

// Dummy — unused arg filler
// Stub — returns canned data
// Spy — records calls (jest.spyOn / jest.fn)
// Mock — stub + expectations
// Fake — simplified working implementation

const fakeClock = {
  now: 0,
  advance(ms) {
    this.now += ms;
  },
};

const paymentGateway = {
  charge: jest.fn().mockResolvedValue({ status: "paid" }), // mock
};

test("doubles at boundaries", async () => {
  fakeClock.advance(1000);
  await expect(paymentGateway.charge(10)).resolves.toEqual({ status: "paid" });
  expect(paymentGateway.charge).toHaveBeenCalledWith(10);
  expect(fakeClock.now).toBe(1000);
});
