// TypeScript with Jest — Exercise 17
// Focus: express get | Scenario: auth service

// npm i -D jest ts-jest typescript @types/jest
// jest.config.ts
import type { Config } from "jest";

const config: Config = {
  preset: "ts-jest",
  testEnvironment: "node",
  roots: ["<rootDir>/src"],
};
export default config;

// src/math.ts
export function sum(a: number, b: number): number {
  return a + b;
}

// src/math.test.ts
import { sum } from "./math";

test("typed sum", () => {
  expect(sum(1, 2)).toBe(3);
});

Practice Notes

Checklist for TypeScript with Jest
- Practice: express get
- Scenario: auth service
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating

Useful commands:
  npm install --save-dev jest
  npx jest
  npx jest --watch
  npx jest --coverage
  npx jest -u
