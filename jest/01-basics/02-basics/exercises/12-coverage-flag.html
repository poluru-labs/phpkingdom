// Jest - Basics — Exercise 12
// Focus: coverage flag | Scenario: React app

// Arrange → Act → Assert
function clamp(n, min, max) {
  return Math.min(max, Math.max(min, n));
}

describe("clamp", () => {
  test("keeps value in range", () => {
    expect(clamp(5, 0, 10)).toBe(5);
  });
  test("bounds low and high", () => {
    expect(clamp(-1, 0, 10)).toBe(0);
    expect(clamp(99, 0, 10)).toBe(10);
  });
});

Practice Notes

Checklist for Jest - Basics
- Practice: coverage flag
- Scenario: React app
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating
