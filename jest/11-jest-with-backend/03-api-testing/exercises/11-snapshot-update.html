// API Testing — Exercise 11
// Focus: snapshot update | Scenario: utility library

const http = require("http");

function getJson(url) {
  return new Promise((resolve, reject) => {
    http
      .get(url, (res) => {
        let raw = "";
        res.on("data", (c) => (raw += c));
        res.on("end", () => resolve({ status: res.statusCode, body: JSON.parse(raw) }));
      })
      .on("error", reject);
  });
}

// Prefer mocking fetch/axios OR spinning a local server in beforeAll
test("api client shape", async () => {
  const client = { get: jest.fn().mockResolvedValue({ data: { id: 1 } }) };
  await expect(client.get("/users/1")).resolves.toEqual({ data: { id: 1 } });
});

Practice Notes

Checklist for API Testing
- Practice: snapshot update
- Scenario: utility library
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating

Useful commands:

  npx jest
  npx jest --watch
  npx jest --coverage
  npx jest -u
