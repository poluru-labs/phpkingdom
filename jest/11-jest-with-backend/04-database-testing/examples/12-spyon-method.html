// Database Testing — Example 12
// Focus: spyOn method | Scenario: React app

const repo = {
  findById: jest.fn(),
  save: jest.fn(),
};

async function renameUser(id, name) {
  const user = await repo.findById(id);
  if (!user) throw new Error("not found");
  user.name = name;
  return repo.save(user);
}

test("db boundary is mocked", async () => {
  repo.findById.mockResolvedValue({ id: 1, name: "Ada" });
  repo.save.mockImplementation(async (u) => u);
  await expect(renameUser(1, "Grace")).resolves.toEqual({
    id: 1,
    name: "Grace",
  });
  expect(repo.save).toHaveBeenCalled();
});
// Integration tests may use testcontainers; unit tests mock the repository

Practice Notes

Checklist for Database Testing
- Practice: spyOn method
- Scenario: React app
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating

Useful commands:
  npm install --save-dev jest
  npx jest
  npx jest --watch
  npx jest --coverage
  npx jest -u
