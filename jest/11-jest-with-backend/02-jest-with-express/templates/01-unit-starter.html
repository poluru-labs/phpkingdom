Jest

// Jest with Express.js — Template 01
// Focus: unit starter | Scenario: utility library

const express = require("express");
const request = require("supertest");

function buildApp() {
  const app = express();
  app.get("/health", (_req, res) => res.json({ ok: true }));
  return app;
}

test("GET /health", async () => {
  const res = await request(buildApp()).get("/health");
  expect(res.status).toBe(200);
  expect(res.body).toEqual({ ok: true });
});
// Keep app factory exportable without listening on a port

Practice Notes

Checklist for Jest with Express.js
- Practice: unit starter
- Scenario: utility library
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating
