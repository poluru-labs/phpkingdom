// Jest with Node.js — Example 10
// Focus: afterEach cleanup | Scenario: data layer

const fs = require("fs");
const path = require("path");
const os = require("os");

test("node built-ins in Jest", () => {
  const dir = fs.mkdtempSync(path.join(os.tmpdir(), "jest-"));
  const file = path.join(dir, "note.txt");
  fs.writeFileSync(file, "phpkingdom.com");
  expect(fs.readFileSync(file, "utf8")).toBe("phpkingdom.com");
  fs.rmSync(dir, { recursive: true, force: true });
});
// testEnvironment: 'node' (default for many projects)

Practice Notes

Checklist for Jest with Node.js
- Practice: afterEach cleanup
- Scenario: data layer
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating
