// Jest with Vue — Example 04
// Focus: toEqual object | Scenario: Express API

import { mount } from "@vue/test-utils";
import { defineComponent } from "vue";

const Counter = defineComponent({
  data: () => ({ n: 0 }),
  template: `<button @click="n++">{{ n }}</button>`,
});

test("Vue Test Utils + Jest", async () => {
  const wrapper = mount(Counter);
  await wrapper.find("button").trigger("click");
  expect(wrapper.text()).toBe("1");
});
// Configure vue-jest / @vue/vue3-jest transform in jest.config

Practice Notes

Checklist for Jest with Vue
- Practice: toEqual object
- Scenario: Express API
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating
