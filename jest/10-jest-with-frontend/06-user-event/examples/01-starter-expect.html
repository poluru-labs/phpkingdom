// user-event Testing — Example 01
// Focus: starter expect | Scenario: utility library

import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";

function Search({ onSearch }) {
  return (
    <input
      aria-label="Search"
      onChange={(e) => onSearch(e.target.value)}
    />
  );
}

test("user-event types into inputs", async () => {
  const user = userEvent.setup();
  const onSearch = jest.fn();
  render(<Search onSearch={onSearch} />);
  await user.type(screen.getByLabelText("Search"), "jest");
  expect(onSearch).toHaveBeenLastCalledWith("jest");
});
// Prefer userEvent over fireEvent for realistic interactions

Practice Notes

Checklist for user-event Testing
- Practice: starter expect
- Scenario: utility library
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating
