// React Testing Library — Exercise 16
// Focus: node util | Scenario: design system

import React from "react";
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";

function Counter() {
  const [n, setN] = React.useState(0);
  return (
    <button onClick={() => setN((x) => x + 1)}>Count {n}</button>
  );
}

test("RTL prefers roles and user-centric queries", async () => {
  const user = userEvent.setup();
  render(<Counter />);
  const btn = screen.getByRole("button", { name: /count 0/i });
  await user.click(btn);
  expect(screen.getByRole("button", { name: /count 1/i })).toBeInTheDocument();
});
// Avoid testing implementation details (state, instance methods)

Practice Notes

Checklist for React Testing Library
- Practice: node util
- Scenario: design system
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating

Useful commands:
  npm install --save-dev jest
  npx jest
  npx jest --watch
  npx jest --coverage
  npx jest -u
