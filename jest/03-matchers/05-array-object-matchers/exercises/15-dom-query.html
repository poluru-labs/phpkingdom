Jest

// Array and Object Matchers — Exercise 15
// Focus: dom query | Scenario: Node CLI

test("array and object matchers", () => {
  const tools = ["jest", "mocha", "vitest"];
  const user = { id: 1, profile: { city: "Austin" } };

  expect(tools).toContain("jest");
  expect(tools).toEqual(expect.arrayContaining(["mocha"]));
  expect(tools).toHaveLength(3);
  expect(user).toHaveProperty("profile.city", "Austin");
  expect(user).toEqual(expect.objectContaining({ id: 1 }));
});
