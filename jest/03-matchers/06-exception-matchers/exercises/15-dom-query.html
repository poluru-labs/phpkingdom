// Exception Matchers — Exercise 15
// Focus: dom query | Scenario: Node CLI

function explode(msg) {
  throw new Error(msg);
}

async function explodeAsync() {
  throw new TypeError("bad type");
}

test("exception matchers", () => {
  expect(() => explode("boom")).toThrow();
  expect(() => explode("boom")).toThrow("boom");
  expect(() => explode("boom")).toThrow(/bo+/);
  expect(() => explode("boom")).toThrow(Error);
});

test("async exceptions", async () => {
  await expect(explodeAsync()).rejects.toThrow(TypeError);
  await expect(explodeAsync()).rejects.toThrow("bad type");
});
