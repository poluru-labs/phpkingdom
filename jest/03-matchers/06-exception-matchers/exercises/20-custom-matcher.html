// Exception Matchers — Exercise 20
// Focus: custom matcher | Scenario: data layer

function explode(msg) {
  throw new Error(msg);
}

async function explodeAsync() {
  throw new TypeError("bad type");
}

test("exception matchers", () => {
  expect(() => explode("boom")).toThrow();
  expect(() => explode("boom")).toThrow("boom");
  expect(() => explode("boom")).toThrow(/bo+/);
  expect(() => explode("boom")).toThrow(Error);
});

test("async exceptions", async () => {
  await expect(explodeAsync()).rejects.toThrow(TypeError);
  await expect(explodeAsync()).rejects.toThrow("bad type");
});

Practice Notes

Checklist for Exception Matchers
- Practice: custom matcher
- Scenario: data layer
- Keep arrange / act / assert clear and deterministic
- Clear or restore mocks between tests (clearMocks / restoreMocks)
- Prefer async/await with resolves / rejects over done()
- Mock at boundaries; avoid over-mocking pure logic
- Review snapshot diffs before updating
