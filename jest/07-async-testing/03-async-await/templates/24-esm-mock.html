// Testing Async Await — Template 24
// Focus: esm mock | Scenario: Express API

async function getUser(id) {
  if (id <= 0) throw new Error("invalid");
  return { id, name: "Ada" };
}

test("await happy path", async () => {
  const user = await getUser(1);
  expect(user).toEqual({ id: 1, name: "Ada" });
});

test("await rejection", async () => {
  await expect(getUser(-1)).rejects.toThrow("invalid");
});
