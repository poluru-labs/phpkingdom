// Testing Callbacks — Template 21
// Focus: watch kit | Scenario: utility library

function fetchData(cb) {
  setTimeout(() => cb(null, { ok: true }), 10);
}

test("callback style with done", (done) => {
  fetchData((err, data) => {
    try {
      expect(err).toBeNull();
      expect(data).toEqual({ ok: true });
      done();
    } catch (e) {
      done(e);
    }
  });
});
// Prefer promisify + async/await when you control the API
