Fake Timers

Section: Async Testing

Focus: Jest.Fn Mock

Scenario: utility library

Study the Jest pattern and adapt matchers, mocks, or async flow to your project.

Jest

// Fake Timers — Example 11
// Focus: jest.fn mock | Scenario: utility library

jest.useFakeTimers();

function schedule(fn) {
  setTimeout(fn, 1000);
}

test("advanceTimersByTime runs deferred work", () => {
  const fn = jest.fn();
  schedule(fn);
  expect(fn).not.toHaveBeenCalled();
  jest.advanceTimersByTime(1000);
  expect(fn).toHaveBeenCalledTimes(1);
});

test("runAllTimers flushes the queue", () => {
  const fn = jest.fn();
  setTimeout(fn, 5000);
  jest.runAllTimers();
  expect(fn).toHaveBeenCalled();
});

afterAll(() => {
  jest.useRealTimers();
});
