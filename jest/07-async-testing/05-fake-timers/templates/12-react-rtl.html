Jest

// Fake Timers — Template 12
// Focus: react rtl | Scenario: React app

jest.useFakeTimers();

function schedule(fn) {
  setTimeout(fn, 1000);
}

test("advanceTimersByTime runs deferred work", () => {
  const fn = jest.fn();
  schedule(fn);
  expect(fn).not.toHaveBeenCalled();
  jest.advanceTimersByTime(1000);
  expect(fn).toHaveBeenCalledTimes(1);
});

test("runAllTimers flushes the queue", () => {
  const fn = jest.fn();
  setTimeout(fn, 5000);
  jest.runAllTimers();
  expect(fn).toHaveBeenCalled();
});

afterAll(() => {
  jest.useRealTimers();
});
