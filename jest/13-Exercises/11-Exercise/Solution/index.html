Solution 11 — Mock Module

Sketch jest.mock for a ./request module.

Solution

jest.mock('./request', () => ({
  fetchUser: jest.fn(() => Promise.resolve({ id: 1 })),
}));

const { fetchUser } = require('./request');

test('loads user', async () => {
  await expect(fetchUser()).resolves.toEqual({ id: 1 });
});
