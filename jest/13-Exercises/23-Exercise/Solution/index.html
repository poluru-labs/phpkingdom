Extend expect with toBeWithinRange.

Solution

expect.extend({
  toBeWithinRange(received, floor, ceiling) {
    const pass = received >= floor && received <= ceiling;
    return {
      pass,
      message: () => `expected ${received} to be within ${floor}-${ceiling}`,
    };
  },
});

expect(100).toBeWithinRange(90, 110);
