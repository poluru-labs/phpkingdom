// components/Card.tsx — import from Server or Client parents
export function Card({
  title,
  children,
}: {
  title: string;
  children: React.ReactNode;
}) {
  return (
    <article className="card">
      <h2>{title}</h2>
      {children}
    </article>
  );
}

// app/page.tsx
import { Card } from "@/components/Card";
export default function Home() {
  return <Card title="Welcome"><p>Shared UI</p></Card>;
}

Practice Notes

Checklist for Shared Components
- Practice: parallel route
- Scenario: SaaS dashboard
- Prefer App Router: app/, layout.tsx, page.tsx, loading.tsx, error.tsx
- Default to Server Components; add "use client" only when needed
- Cache intentionally: fetch options, revalidate, tags, no-store
- Use Route Handlers / Server Actions for backend work in the same app
- Optimize with next/image, next/font, and Metadata API

Useful commands:
  npx create-next-app@latest
  npm run dev
  npm run build && npm start
  npx next lint
  ANALYZE=true npm run build  # with @next/bundle-analyzer
