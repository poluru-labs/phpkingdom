import { Suspense } from "react";

async function Feed() {
  const posts = await fetch("https://api.example.com/feed", {
    cache: "no-store",
  }).then((r) => r.json());
  return (
    <ul>
      {posts.map((p: { id: number; title: string }) => (
        <li key={p.id}>{p.title}</li>
      ))}
    </ul>
  );
}

export default function Page() {
  return (
    <main>
      <h1>Feed</h1>
      <Suspense fallback={<p>Loading feed…</p>}>
        <Feed />
      </Suspense>
    </main>
  );
}
