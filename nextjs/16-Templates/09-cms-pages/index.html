export default async function CmsPage({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;
  const page = await fetch(`https://cms.example.com/pages/${slug}`, {
    next: { tags: [`page-${slug}`] },
  }).then((r) => r.json());

  return (
    <article>
      <h1>{page.title}</h1>
      <div dangerouslySetInnerHTML={{ __html: page.body }} />
    </article>
  );
}
