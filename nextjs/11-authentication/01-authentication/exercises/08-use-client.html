// Next.js Authentication — Exercise 08
// Focus: use client | Scenario: news feed

// Auth.js (NextAuth v5) — auth.ts
import NextAuth from "next-auth";
import GitHub from "next-auth/providers/github";

export const { handlers, auth, signIn, signOut } = NextAuth({
  providers: [GitHub],
});

// app/api/auth/[...nextauth]/route.ts
export { handlers as GET, handlers as POST } from "@/auth";

// app/page.tsx
import { auth, signIn } from "@/auth";

export default async function Home() {
  const session = await auth();
  if (!session) {
    return (
      <form
        action={async () => {
          "use server";
          await signIn("github");
        }}
      >
        <button type="submit">Sign in with GitHub</button>
      </form>
    );
  }
  return <p>Hello {session.user?.name}</p>;
}

Practice Notes

Checklist for Next.js Authentication
- Practice: use client
- Scenario: news feed
- Prefer App Router: app/, layout.tsx, page.tsx, loading.tsx, error.tsx
- Default to Server Components; add "use client" only when needed
- Cache intentionally: fetch options, revalidate, tags, no-store
- Use Route Handlers / Server Actions for backend work in the same app
- Optimize with next/image, next/font, and Metadata API
