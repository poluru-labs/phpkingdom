// Static Rendering — Template 13
// Focus: modal intercept | Scenario: blog platform

// Static by default when data is cacheable
export default async function MarketingPage() {
  const res = await fetch("https://api.example.com/hero", {
    cache: "force-cache",
  });
  const hero = await res.json();
  return (
    <main>
      <h1>{hero.title}</h1>
      <p>{hero.subtitle}</p>
    </main>
  );
}
// Built once; served as static HTML/RSC payload

Practice Notes

Checklist for Static Rendering
- Practice: modal intercept
- Scenario: blog platform
- Prefer App Router: app/, layout.tsx, page.tsx, loading.tsx, error.tsx
- Default to Server Components; add "use client" only when needed
- Cache intentionally: fetch options, revalidate, tags, no-store
- Use Route Handlers / Server Actions for backend work in the same app
- Optimize with next/image, next/font, and Metadata API
