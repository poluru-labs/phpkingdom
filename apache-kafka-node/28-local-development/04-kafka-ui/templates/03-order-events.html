KafkaJS (Node.js)

// Kafka UI Tools — Template 03
// Focus: order events | Scenario: activity tracker

// KafkaJS topic: Kafka UI Tools
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-kafka-ui",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_kafka_ui(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "kafka-ui-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "kafka-ui", label: "Kafka UI Tools", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_KAFKA_UI = "kafka-ui";

Practice Notes

Checklist for Kafka UI Tools
- Practice: order events
- Scenario: activity tracker
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

  npm install kafkajs
