// Local Kafka Cluster — Template 15
// Focus: sasl secure | Scenario: payment bus

// KafkaJS topic: Local Kafka Cluster
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-local-cluster",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_local_cluster(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "local-cluster-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "local-cluster", label: "Local Kafka Cluster", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_LOCAL_CLUSTER = "local-cluster";

Practice Notes

Checklist for Local Kafka Cluster
- Practice: sasl secure
- Scenario: payment bus
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
