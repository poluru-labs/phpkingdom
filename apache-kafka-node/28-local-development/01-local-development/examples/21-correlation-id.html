KafkaJS (Node.js)

// Local Kafka Development — Example 21
// Focus: correlation id | Scenario: order service

// KafkaJS topic: Local Kafka Development
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-local-development",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_local_development(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "local-development-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "local-development", label: "Local Kafka Development", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_LOCAL_DEVELOPMENT = "local-development";

Practice Notes

Checklist for Local Kafka Development
- Practice: correlation id
- Scenario: order service
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
