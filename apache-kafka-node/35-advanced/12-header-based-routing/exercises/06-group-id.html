KafkaJS (Node.js)

// Header-Based Routing — Exercise 06
// Focus: group id | Scenario: inventory sync

// KafkaJS topic: Header-Based Routing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-header-based-routing",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_header_based_routing(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "header-based-routing-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "header-based-routing", label: "Header-Based Routing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_HEADER_BASED_ROUTING = "header-based-routing";

Practice Notes

Checklist for Header-Based Routing
- Practice: group id
- Scenario: inventory sync
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
