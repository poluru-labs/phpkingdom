KafkaJS (Node.js)

// Kafka MirrorMaker — Exercise 05
// Focus: consumer run | Scenario: payment bus

// KafkaJS topic: Kafka MirrorMaker
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-mirror-maker",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_mirror_maker(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "mirror-maker-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "mirror-maker", label: "Kafka MirrorMaker", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_MIRROR_MAKER = "mirror-maker";

Practice Notes

Checklist for Kafka MirrorMaker
- Practice: consumer run
- Scenario: payment bus
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
