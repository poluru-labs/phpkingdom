export async function consumeWithBackpressure(consumer, topic) {
  await consumer.subscribe({ topic });
  await consumer.run({
    eachBatch: async ({ batch, heartbeat, isRunning, isStale, pause }) => {
      if (await queueAlmostFull()) {
        const resume = pause();
        setTimeout(() => resume(), 1000);
        return;
      }
      for (const message of batch.messages) {
        if (!isRunning() || isStale()) break;
        await handle(message);
        await heartbeat();
      }
    },
  });
}
async function queueAlmostFull() { return false; }
async function handle(_) {}
