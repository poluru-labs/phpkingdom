KafkaJS (Node.js)

// Kafka Streams (Node) — Example 25
// Focus: best practice recap | Scenario: payment bus

// Node stream processing sketch with KafkaJS (not JVM Kafka Streams)
import { Kafka } from "kafkajs";

const kafka = new Kafka({ clientId: "stream-worker", brokers: ["localhost:9092"] });

export async function run() {
  const consumer = kafka.consumer({ groupId: "stream-workers" });
  const producer = kafka.producer({ idempotent: true });
  await consumer.connect();
  await producer.connect();
  await consumer.subscribe({ topic: "orders", fromBeginning: false });
  await consumer.run({
    eachMessage: async ({ message }) => {
      const order = JSON.parse(message.value.toString());
      if (order.total > 100) {
        await producer.send({
          topic: "orders.high-value",
          messages: [{ key: order.id, value: JSON.stringify(order) }],
        });
      }
    },
  });
}

export const TOPIC_ID = "kafka-streams-node";

Practice Notes

Checklist for Kafka Streams (Node)
- Practice: best practice recap
- Scenario: payment bus
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
