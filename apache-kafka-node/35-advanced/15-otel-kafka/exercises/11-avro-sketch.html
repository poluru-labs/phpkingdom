// Propagate W3C traceparent in Kafka headers
export function injectTrace(headers = {}) {
  const span = getActiveSpan(); // from your OTel API
  if (!span) return headers;
  return { ...headers, traceparent: Buffer.from(spanToTraceparent(span)) };
}

export function extractTrace(headers) {
  const raw = headers?.traceparent?.toString();
  return raw ? contextFromTraceparent(raw) : undefined;
}

Practice Notes

Checklist for OpenTelemetry Kafka
- Practice: avro sketch
- Scenario: order service
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
