// ksqlDB (run against ksqlDB server; Node usually posts SQL via REST)
export async function createStream(ksqlUrl) {
  const sql = `
    CREATE STREAM orders_s (id VARCHAR KEY, total DOUBLE)
    WITH (KAFKA_TOPIC='orders', VALUE_FORMAT='JSON');
  `;
  await fetch(`${ksqlUrl}/ksql`, {
    method: "POST",
    headers: { "content-type": "application/vnd.ksql.v1+json" },
    body: JSON.stringify({ ksql: sql, streamsProperties: {} }),
  });
}
