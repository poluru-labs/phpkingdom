Section: Advanced

Study the KafkaJS pattern and adapt it to your event-driven service.

KafkaJS (Node.js)

// Cooperative Sticky Assignor — Example 20
// Focus: consumer lag | Scenario: audit trail

import { Kafka, PartitionAssignors } from "kafkajs";

const kafka = new Kafka({ clientId: "app", brokers: ["localhost:9092"] });
const consumer = kafka.consumer({
  groupId: "orders",
  // Prefer cooperative sticky to reduce stop-the-world rebalances
  partitionAssignors: [PartitionAssignors.cooperativeSticky],
});

export const TOPIC_ID = "cooperative-sticky";
