KafkaJS (Node.js)

// Production Deployment — Exercise 10
// Focus: json parse | Scenario: audit trail

// KafkaJS topic: Production Deployment
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-production-deployment",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_production_deployment(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "production-deployment-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "production-deployment", label: "Production Deployment", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PRODUCTION_DEPLOYMENT = "production-deployment";

Practice Notes

Checklist for Production Deployment
- Practice: json parse
- Scenario: audit trail
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
