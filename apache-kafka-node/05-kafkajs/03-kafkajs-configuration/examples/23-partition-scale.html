KafkaJS (Node.js)

// KafkaJS Configuration — Example 23
// Focus: partition scale | Scenario: activity tracker

// KafkaJS topic: KafkaJS Configuration
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-kafkajs-configuration",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_kafkajs_configuration(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "kafkajs-configuration-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "kafkajs-configuration", label: "KafkaJS Configuration", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_KAFKAJS_CONFIGURATION = "kafkajs-configuration";

Practice Notes

Checklist for KafkaJS Configuration
- Practice: partition scale
- Scenario: activity tracker
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
