import { Kafka } from "kafkajs";

export function secureClient() {
  return new Kafka({
    clientId: "secure-app",
    brokers: process.env.KAFKA_BROKERS.split(","),
    ssl: true,
    sasl: {
      mechanism: "scram-sha-512",
      username: process.env.KAFKA_USER,
      password: process.env.KAFKA_PASS,
    },
  });
}
