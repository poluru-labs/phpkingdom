KafkaJS (Node.js)

// SASL Authentication — Exercise 08
// Focus: commit offsets | Scenario: saga workflow

import { Kafka } from "kafkajs";

export function secureClient() {
  return new Kafka({
    clientId: "secure-app",
    brokers: process.env.KAFKA_BROKERS.split(","),
    ssl: true,
    sasl: {
      mechanism: "scram-sha-512",
      username: process.env.KAFKA_USER,
      password: process.env.KAFKA_PASS,
    },
  });
}

export const TOPIC_ID = "sasl";

Practice Notes

Checklist for SASL Authentication
- Practice: commit offsets

- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
