// Partition Leaders — Exercise 21

// KafkaJS topic: Partition Leaders
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-partition-leaders",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_partition_leaders(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "partition-leaders-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "partition-leaders", label: "Partition Leaders", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PARTITION_LEADERS = "partition-leaders";
