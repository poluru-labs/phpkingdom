// Message Retention — Example 25
// Focus: best practice recap | Scenario: payment bus

// KafkaJS topic: Message Retention
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-retention",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_retention(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "retention-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "retention", label: "Message Retention", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_RETENTION = "retention";
