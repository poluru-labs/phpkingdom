// Confluent Cloud — Exercise 19
// Focus: compose ui | Scenario: analytics stream

// KafkaJS topic: Confluent Cloud
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-confluent-cloud",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_confluent_cloud(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "confluent-cloud-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "confluent-cloud", label: "Confluent Cloud", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CONFLUENT_CLOUD = "confluent-cloud";

Practice Notes

Checklist for Confluent Cloud
- Practice: compose ui
- Scenario: analytics stream
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:

  npm install kafkajs
