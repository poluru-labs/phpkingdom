// Amazon MSK — Exercise 05
// Focus: consumer run | Scenario: payment bus

import { Kafka } from "kafkajs";

export function mskClient() {
  return new Kafka({
    clientId: "msk-app",
    brokers: process.env.MSK_BROKERS.split(","),
    ssl: true,
    sasl: {
      mechanism: "aws", // or scram depending on MSK auth
      authorizationIdentity: process.env.AWS_ROLE_ARN,
    },
  });
}

export const TOPIC_ID = "aws-msk";

Practice Notes

Checklist for Amazon MSK
- Practice: consumer run
- Scenario: payment bus
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
