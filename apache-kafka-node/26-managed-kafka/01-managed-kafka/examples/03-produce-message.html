KafkaJS (Node.js)

// Managed Apache Kafka — Example 03
// Focus: produce message | Scenario: activity tracker

// KafkaJS topic: Managed Apache Kafka
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-managed-kafka",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_managed_kafka(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "managed-kafka-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "managed-kafka", label: "Managed Apache Kafka", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_MANAGED_KAFKA = "managed-kafka";

Practice Notes

Checklist for Managed Apache Kafka
- Practice: produce message
- Scenario: activity tracker
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
