KafkaJS (Node.js)

// KafkaJS Setup — Exercise 18
// Focus: jest mock | Scenario: saga workflow

// KafkaJS topic: KafkaJS Setup
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-kafkajs-setup",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_kafkajs_setup(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "kafkajs-setup-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "kafkajs-setup", label: "KafkaJS Setup", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_KAFKAJS_SETUP = "kafkajs-setup";

Practice Notes

Checklist for KafkaJS Setup
- Practice: jest mock
- Scenario: saga workflow
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
