// Sink Connectors — Template 21
// Focus: mirror maker | Scenario: order service

// KafkaJS topic: Sink Connectors
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-sink-connectors",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_sink_connectors(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "sink-connectors-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "sink-connectors", label: "Sink Connectors", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_SINK_CONNECTORS = "sink-connectors";

Practice Notes

Checklist for Sink Connectors
- Practice: mirror maker
- Scenario: order service
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
