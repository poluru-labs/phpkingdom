// Connector Configuration — Template 04
// Focus: notification fanout | Scenario: log pipeline

// KafkaJS topic: Connector Configuration
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-connect-configuration",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_connect_configuration(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "connect-configuration-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "connect-configuration", label: "Connector Configuration", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CONNECT_CONFIGURATION = "connect-configuration";

Practice Notes

Checklist for Connector Configuration
- Practice: notification fanout
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
