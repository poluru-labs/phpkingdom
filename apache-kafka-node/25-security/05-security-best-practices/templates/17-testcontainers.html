// Kafka Security Best Practices — Template 17
// Focus: testcontainers | Scenario: CDC bridge

// KafkaJS topic: Kafka Security Best Practices
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-security-best-practices",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_security_best_practices(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "security-best-practices-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "security-best-practices", label: "Kafka Security Best Practices", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_SECURITY_BEST_PRACTICES = "security-best-practices";

Practice Notes

Checklist for Kafka Security Best Practices
- Practice: testcontainers
- Scenario: CDC bridge
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
