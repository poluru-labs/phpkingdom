KafkaJS (Node.js)

// Apache Kafka Security — Exercise 04
// Focus: keyed produce | Scenario: log pipeline

// KafkaJS topic: Apache Kafka Security
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-security",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_security(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "security-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "security", label: "Apache Kafka Security", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_SECURITY = "security";

Practice Notes

Checklist for Apache Kafka Security
- Practice: keyed produce
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
