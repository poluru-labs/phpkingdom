// Kafka Performance Tuning — Template 13
// Focus: choreography saga | Scenario: activity tracker

// KafkaJS topic: Kafka Performance Tuning
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-performance-tuning",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_performance_tuning(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "performance-tuning-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "performance-tuning", label: "Kafka Performance Tuning", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PERFORMANCE_TUNING = "performance-tuning";

Practice Notes

Checklist for Kafka Performance Tuning
- Practice: choreography saga
- Scenario: activity tracker
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
