KafkaJS (Node.js)

// Kafka Message Keys — Exercise 24
// Focus: k8s deploy | Scenario: log pipeline

// KafkaJS topic: Kafka Message Keys
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-message-keys",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_message_keys(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "message-keys-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "message-keys", label: "Kafka Message Keys", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_MESSAGE_KEYS = "message-keys";

Practice Notes

Checklist for Kafka Message Keys
- Practice: k8s deploy
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
