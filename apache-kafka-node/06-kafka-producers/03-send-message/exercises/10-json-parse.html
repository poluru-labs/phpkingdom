// Send Kafka Messages — Exercise 10

// KafkaJS topic: Send Kafka Messages
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-send-message",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_send_message(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "send-message-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "send-message", label: "Send Kafka Messages", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_SEND_MESSAGE = "send-message";

Practice Notes

Checklist for Send Kafka Messages

- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
