KafkaJS (Node.js)

// Producer Configuration — Exercise 12
// Focus: dlq publish | Scenario: notification hub

// KafkaJS topic: Producer Configuration
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-producer-configuration",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_producer_configuration(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "producer-configuration-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "producer-configuration", label: "Producer Configuration", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PRODUCER_CONFIGURATION = "producer-configuration";

Practice Notes

Checklist for Producer Configuration
- Practice: dlq publish
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
