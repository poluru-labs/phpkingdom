// KafkaJS topic: Create a Node.js Producer
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-create-producer",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_create_producer(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "create-producer-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "create-producer", label: "Create a Node.js Producer", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CREATE_PRODUCER = "create-producer";
