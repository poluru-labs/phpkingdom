// Transactional outbox: write DB row + outbox in one TX, relay to Kafka
export async function placeOrder(db, relay, order) {
  await db.tx(async (trx) => {
    await trx.insert("orders", order);
    await trx.insert("outbox", {
      aggregate_type: "order",
      aggregate_id: order.id,
      type: "order.created",
      payload: JSON.stringify(order),
    });
  });
  // separate publisher polls outbox → producer.send
}

Practice Notes

Checklist for Debezium Outbox
- Practice: jest mock

- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
