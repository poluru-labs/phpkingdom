KafkaJS (Node.js)

// Kafka Event Design — Template 24
// Focus: backpressure | Scenario: log pipeline

// KafkaJS topic: Kafka Event Design
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-event-design",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_event_design(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "event-design-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "event-design", label: "Kafka Event Design", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_EVENT_DESIGN = "event-design";

Practice Notes

Checklist for Kafka Event Design
- Practice: backpressure
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
