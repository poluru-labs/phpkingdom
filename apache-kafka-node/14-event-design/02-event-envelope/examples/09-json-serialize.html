KafkaJS (Node.js)

// Event Envelope — Example 09
// Focus: json serialize | Scenario: analytics stream

// KafkaJS topic: Event Envelope
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-event-envelope",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_event_envelope(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "event-envelope-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "event-envelope", label: "Event Envelope", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_EVENT_ENVELOPE = "event-envelope";

Practice Notes

Checklist for Event Envelope
- Practice: json serialize
- Scenario: analytics stream
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
