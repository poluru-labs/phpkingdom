// Consumer Rebalance Errors — Exercise 21
// Focus: trace headers | Scenario: order service

// KafkaJS topic: Consumer Rebalance Errors
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-rebalance-errors",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_rebalance_errors(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "rebalance-errors-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "rebalance-errors", label: "Consumer Rebalance Errors", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_REBALANCE_ERRORS = "rebalance-errors";

Practice Notes

Checklist for Consumer Rebalance Errors
