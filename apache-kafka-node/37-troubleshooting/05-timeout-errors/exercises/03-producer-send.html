KafkaJS (Node.js)

// Kafka Timeout Errors — Exercise 03
// Focus: producer send | Scenario: activity tracker

// KafkaJS topic: Kafka Timeout Errors
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-timeout-errors",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_timeout_errors(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "timeout-errors-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "timeout-errors", label: "Kafka Timeout Errors", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_TIMEOUT_ERRORS = "timeout-errors";

Practice Notes

Checklist for Kafka Timeout Errors
- Practice: producer send
- Scenario: activity tracker
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
