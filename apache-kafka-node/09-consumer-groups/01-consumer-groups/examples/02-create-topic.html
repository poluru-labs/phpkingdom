KafkaJS (Node.js)

// Kafka Consumer Groups — Example 02
// Focus: create topic | Scenario: notification hub

import { Kafka } from "kafkajs";
const kafka = new Kafka({ clientId: "app", brokers: ["localhost:9092"] });

export async function runGroup(groupId = "orders-workers") {
  const consumer = kafka.consumer({ groupId });
  await consumer.connect();
  await consumer.subscribe({ topic: "orders", fromBeginning: false });
  await consumer.run({
    eachBatch: async ({ batch, resolveOffset, heartbeat, commitOffsetsIfNecessary }) => {
      for (const message of batch.messages) {
        await process(message);
        resolveOffset(message.offset);
        await heartbeat();
      }
      await commitOffsetsIfNecessary();
    },
  });
}
async function process(message) {}

export const TOPIC_ID = "consumer-groups";

Practice Notes

Checklist for Kafka Consumer Groups
- Practice: create topic
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
