// Static Group Membership — Example 04
// Focus: message key | Scenario: log pipeline

// KafkaJS topic: Static Group Membership
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-static-membership",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_static_membership(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "static-membership-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "static-membership", label: "Static Group Membership", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_STATIC_MEMBERSHIP = "static-membership";

Practice Notes

Checklist for Static Group Membership
