// Group Coordinator — Exercise 17

// KafkaJS topic: Group Coordinator
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-group-coordinator",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_group_coordinator(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "group-coordinator-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "group-coordinator", label: "Group Coordinator", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_GROUP_COORDINATOR = "group-coordinator";
