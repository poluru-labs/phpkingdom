KafkaJS (Node.js)

// Apache Kafka - Introduction — Exercise 03
// Focus: producer send | Scenario: activity tracker

import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-intro",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function hello() {
  const producer = kafka.producer();
  await producer.connect();
  await producer.send({
    topic: "hello",
    messages: [{ value: JSON.stringify({ hello: "kafka" }) }],
  });
  await producer.disconnect();
}

export const TOPIC_ID = "introduction";

Practice Notes

Checklist for Apache Kafka - Introduction
- Practice: producer send
- Scenario: activity tracker
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
