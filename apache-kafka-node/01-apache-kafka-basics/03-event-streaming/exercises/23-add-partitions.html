// Event Streaming — Exercise 23
// Focus: add partitions | Scenario: activity tracker

// KafkaJS topic: Event Streaming
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-event-streaming",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_event_streaming(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "event-streaming-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "event-streaming", label: "Event Streaming", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_EVENT_STREAMING = "event-streaming";

Practice Notes

Checklist for Event Streaming
- Practice: add partitions

- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
