KafkaJS (Node.js)

// Event Notification — Example 02
// Focus: create topic | Scenario: notification hub

// KafkaJS topic: Event Notification
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-event-notification",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_event_notification(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "event-notification-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "event-notification", label: "Event Notification", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_EVENT_NOTIFICATION = "event-notification";

Practice Notes

Checklist for Event Notification
- Practice: create topic
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
