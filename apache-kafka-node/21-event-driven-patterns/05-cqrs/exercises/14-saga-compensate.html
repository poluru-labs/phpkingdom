KafkaJS (Node.js)

// CQRS with Kafka — Exercise 14
// Focus: saga compensate | Scenario: log pipeline

// KafkaJS topic: CQRS with Kafka
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-cqrs",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_cqrs(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "cqrs-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "cqrs", label: "CQRS with Kafka", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CQRS = "cqrs";

Practice Notes

Checklist for CQRS with Kafka
- Practice: saga compensate
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

  npm install kafkajs
