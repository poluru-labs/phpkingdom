// Event-Carried State Transfer — Template 18
// Focus: docker local | Scenario: saga workflow

// KafkaJS topic: Event-Carried State Transfer
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-event-carried-state",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_event_carried_state(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "event-carried-state-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "event-carried-state", label: "Event-Carried State Transfer", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_EVENT_CARRIED_STATE = "event-carried-state";

Practice Notes

Checklist for Event-Carried State Transfer
- Practice: docker local
- Scenario: saga workflow
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
