// Event Sourcing — Template 02
// Focus: hello consumer | Scenario: notification hub

// KafkaJS topic: Event Sourcing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-event-sourcing",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_event_sourcing(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "event-sourcing-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "event-sourcing", label: "Event Sourcing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_EVENT_SOURCING = "event-sourcing";

Practice Notes

Checklist for Event Sourcing
- Practice: hello consumer
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

  npm install kafkajs
