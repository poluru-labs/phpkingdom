// Real-Time Log Processing — Template 25

// KafkaJS topic: Real-Time Log Processing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-log-processing-project",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_log_processing_project(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "log-processing-project-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "log-processing-project", label: "Real-Time Log Processing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_LOG_PROCESSING_PROJECT = "log-processing-project";

Practice Notes

Checklist for Real-Time Log Processing
- Practice: production starter
- Scenario: payment bus
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
