// User Activity Tracking — Exercise 13
// Focus: exactly once | Scenario: activity tracker

// KafkaJS topic: User Activity Tracking
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-activity-tracking-project",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_activity_tracking_project(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "activity-tracking-project-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "activity-tracking-project", label: "User Activity Tracking", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_ACTIVITY_TRACKING_PROJECT = "activity-tracking-project";

Practice Notes

Checklist for User Activity Tracking

- Use DLQ/retry topics for poison messages
