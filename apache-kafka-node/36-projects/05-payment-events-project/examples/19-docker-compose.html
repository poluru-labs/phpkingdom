// Payment Events Pipeline — Example 19
// Focus: docker compose | Scenario: analytics stream

// KafkaJS topic: Payment Events Pipeline
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-payment-events-project",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_payment_events_project(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "payment-events-project-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "payment-events-project", label: "Payment Events Pipeline", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PAYMENT_EVENTS_PROJECT = "payment-events-project";

Checklist for Payment Events Pipeline
