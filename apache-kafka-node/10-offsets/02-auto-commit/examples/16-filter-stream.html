// Automatic Offset Commit — Example 16
// Focus: filter stream | Scenario: inventory sync

// KafkaJS topic: Automatic Offset Commit
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-auto-commit",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_auto_commit(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "auto-commit-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "auto-commit", label: "Automatic Offset Commit", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_AUTO_COMMIT = "auto-commit";

Practice Notes

Checklist for Automatic Offset Commit
- Practice: filter stream

- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
