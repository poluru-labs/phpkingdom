KafkaJS (Node.js)

// Seek to an Offset — Exercise 16
// Focus: ssl config | Scenario: inventory sync

// KafkaJS topic: Seek to an Offset
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-seek-offset",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_seek_offset(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "seek-offset-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "seek-offset", label: "Seek to an Offset", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_SEEK_OFFSET = "seek-offset";
