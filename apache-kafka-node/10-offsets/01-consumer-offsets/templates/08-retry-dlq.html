// Kafka Consumer Offsets — Template 08
// Focus: retry dlq | Scenario: saga workflow

// KafkaJS topic: Kafka Consumer Offsets
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-consumer-offsets",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_consumer_offsets(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "consumer-offsets-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "consumer-offsets", label: "Kafka Consumer Offsets", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CONSUMER_OFFSETS = "consumer-offsets";

Practice Notes

Checklist for Kafka Consumer Offsets
- Practice: retry dlq
- Scenario: saga workflow
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
