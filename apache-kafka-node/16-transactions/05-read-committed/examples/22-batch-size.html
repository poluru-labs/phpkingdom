KafkaJS (Node.js)

// Read Committed Messages — Example 22
// Focus: batch size | Scenario: notification hub

// KafkaJS topic: Read Committed Messages
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-read-committed",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_read_committed(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "read-committed-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "read-committed", label: "Read Committed Messages", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_READ_COMMITTED = "read-committed";

Practice Notes

Checklist for Read Committed Messages
- Practice: batch size
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
