KafkaJS (Node.js)

// Transactional Producer — Example 01
// Focus: client connect | Scenario: order service

// KafkaJS topic: Transactional Producer
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-transactional-producer",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_transactional_producer(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "transactional-producer-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "transactional-producer", label: "Transactional Producer", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_TRANSACTIONAL_PRODUCER = "transactional-producer";

Practice Notes

Checklist for Transactional Producer
- Practice: client connect
- Scenario: order service
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
