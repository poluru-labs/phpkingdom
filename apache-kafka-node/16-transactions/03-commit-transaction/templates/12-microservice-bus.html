// Commit a Transaction — Template 12
// Focus: microservice bus | Scenario: notification hub

// KafkaJS topic: Commit a Transaction
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-commit-transaction",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_commit_transaction(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "commit-transaction-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "commit-transaction", label: "Commit a Transaction", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_COMMIT_TRANSACTION = "commit-transaction";

Practice Notes

Checklist for Commit a Transaction
- Practice: microservice bus
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
