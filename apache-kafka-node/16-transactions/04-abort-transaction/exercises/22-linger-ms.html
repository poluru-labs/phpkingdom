KafkaJS (Node.js)

// Abort a Transaction — Exercise 22
// Focus: linger ms | Scenario: notification hub

// KafkaJS topic: Abort a Transaction
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-abort-transaction",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_abort_transaction(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "abort-transaction-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "abort-transaction", label: "Abort a Transaction", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_ABORT_TRANSACTION = "abort-transaction";
