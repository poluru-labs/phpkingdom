KafkaJS Scaffold

import { Kafka } from "kafkajs";
const kafka = new Kafka({ clientId: "hello", brokers: ["localhost:9092"] });
export async function main() {
  const p = kafka.producer({ idempotent: true });
  await p.connect();
  await p.send({ topic: "hello", messages: [{ value: JSON.stringify({ ok: true }) }] });
  await p.disconnect();
}
