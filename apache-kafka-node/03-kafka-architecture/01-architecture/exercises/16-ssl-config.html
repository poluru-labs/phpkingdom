KafkaJS (Node.js)

// Kafka Architecture — Exercise 16
// Focus: ssl config | Scenario: inventory sync

// KafkaJS topic: Kafka Architecture
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-architecture",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_architecture(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "architecture-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "architecture", label: "Kafka Architecture", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_ARCHITECTURE = "architecture";

Practice Notes

Checklist for Kafka Architecture
- Practice: ssl config
- Scenario: inventory sync
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
