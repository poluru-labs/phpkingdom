KafkaJS (Node.js)

// Schema Evolution — Example 12
// Focus: acks all | Scenario: notification hub

// KafkaJS topic: Schema Evolution
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-schema-evolution",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_schema_evolution(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "schema-evolution-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "schema-evolution", label: "Schema Evolution", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_SCHEMA_EVOLUTION = "schema-evolution";

Practice Notes

Checklist for Schema Evolution
- Practice: acks all
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
