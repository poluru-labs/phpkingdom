// JSON Schema / Avro via Confluent Schema Registry (sketch)
export async function registerSchema(registryUrl, subject, schema) {
  const res = await fetch(`${registryUrl}/subjects/${subject}/versions`, {
    method: "POST",
    headers: { "content-type": "application/vnd.schemaregistry.v1+json" },
    body: JSON.stringify({ schema: JSON.stringify(schema) }),
  });
  return res.json(); // { id }
}

export const TOPIC_ID = "schema-registry";
