KafkaJS (Node.js)

// Production Releases — Example 20
// Focus: consumer lag | Scenario: audit trail

// KafkaJS topic: Production Releases
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-production-releases",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_production_releases(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "production-releases-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "production-releases", label: "Production Releases", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PRODUCTION_RELEASES = "production-releases";

Practice Notes

Checklist for Production Releases
- Practice: consumer lag
- Scenario: audit trail
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
