KafkaJS (Node.js)

// Kafka Cluster Health — Example 24
// Focus: outbox publish | Scenario: log pipeline

// KafkaJS topic: Kafka Cluster Health
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-cluster-health",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_cluster_health(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "cluster-health-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "cluster-health", label: "Kafka Cluster Health", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CLUSTER_HEALTH = "cluster-health";

Practice Notes

Checklist for Kafka Cluster Health
- Practice: outbox publish
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
