KafkaJS (Node.js)

// Scaling Kafka Applications — Template 24
// Focus: backpressure | Scenario: log pipeline

// KafkaJS topic: Scaling Kafka Applications
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-scaling",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_scaling(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "scaling-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "scaling", label: "Scaling Kafka Applications", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_SCALING = "scaling";

Practice Notes

Checklist for Scaling Kafka Applications
- Practice: backpressure
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
