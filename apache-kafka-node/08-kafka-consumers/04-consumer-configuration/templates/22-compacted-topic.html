// Consumer Configuration — Template 22

// KafkaJS topic: Consumer Configuration
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-consumer-configuration",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_consumer_configuration(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "consumer-configuration-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "consumer-configuration", label: "Consumer Configuration", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CONSUMER_CONFIGURATION = "consumer-configuration";

Practice Notes

Checklist for Consumer Configuration
- Practice: compacted topic
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
