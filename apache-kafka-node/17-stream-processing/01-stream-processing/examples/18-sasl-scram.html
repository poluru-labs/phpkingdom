KafkaJS (Node.js)

// Kafka Stream Processing — Example 18
// Focus: sasl scram | Scenario: saga workflow

// KafkaJS topic: Kafka Stream Processing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-stream-processing",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_stream_processing(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "stream-processing-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "stream-processing", label: "Kafka Stream Processing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_STREAM_PROCESSING = "stream-processing";

Practice Notes

Checklist for Kafka Stream Processing
- Practice: sasl scram
- Scenario: saga workflow
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
