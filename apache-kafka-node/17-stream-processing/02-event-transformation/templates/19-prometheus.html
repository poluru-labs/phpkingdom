// Event Transformation — Template 19
// Focus: prometheus | Scenario: analytics stream

// KafkaJS topic: Event Transformation
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-event-transformation",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_event_transformation(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "event-transformation-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "event-transformation", label: "Event Transformation", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_EVENT_TRANSFORMATION = "event-transformation";

Practice Notes

Checklist for Event Transformation
- Practice: prometheus
- Scenario: analytics stream
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
