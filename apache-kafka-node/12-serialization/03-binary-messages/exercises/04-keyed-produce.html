KafkaJS (Node.js)

// Binary Messages — Exercise 04
// Focus: keyed produce | Scenario: log pipeline

// KafkaJS topic: Binary Messages
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-binary-messages",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_binary_messages(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "binary-messages-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "binary-messages", label: "Binary Messages", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_BINARY_MESSAGES = "binary-messages";

Practice Notes

Checklist for Binary Messages
- Practice: keyed produce
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
