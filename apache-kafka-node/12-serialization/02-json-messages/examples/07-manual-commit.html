KafkaJS (Node.js)

// JSON Messages — Example 07
// Focus: manual commit | Scenario: CDC bridge

// KafkaJS topic: JSON Messages
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-json-messages",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_json_messages(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "json-messages-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "json-messages", label: "JSON Messages", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_JSON_MESSAGES = "json-messages";

Practice Notes

Checklist for JSON Messages
- Practice: manual commit
- Scenario: CDC bridge
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
