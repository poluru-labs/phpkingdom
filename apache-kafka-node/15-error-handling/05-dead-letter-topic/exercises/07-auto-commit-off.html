export async function withDlq({ consumer, producer, mainTopic, dlqTopic }) {
  await consumer.run({
    eachMessage: async ({ topic, partition, message }) => {
      try {
        await handle(JSON.parse(message.value.toString()));
      } catch (err) {
        await producer.send({
          topic: dlqTopic,
          messages: [{
            key: message.key,
            value: message.value,
            headers: {
              ...message.headers,
              "x-original-topic": topic,
              "x-error": String(err.message),
            },
          }],
        });
      }
    },
  });
}
async function handle(_) {}

Practice Notes

Checklist for Dead-Letter Topics
- Practice: auto commit off
- Scenario: CDC bridge
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
