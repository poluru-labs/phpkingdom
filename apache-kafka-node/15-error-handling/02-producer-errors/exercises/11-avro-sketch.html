// Producer Errors — Exercise 11

// KafkaJS topic: Producer Errors
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-producer-errors",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_producer_errors(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "producer-errors-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "producer-errors", label: "Producer Errors", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PRODUCER_ERRORS = "producer-errors";
