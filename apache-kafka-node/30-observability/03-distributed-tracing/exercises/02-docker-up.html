KafkaJS (Node.js)

// Distributed Tracing — Exercise 02
// Focus: docker up | Scenario: notification hub

// KafkaJS topic: Distributed Tracing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-distributed-tracing",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_distributed_tracing(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "distributed-tracing-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "distributed-tracing", label: "Distributed Tracing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_DISTRIBUTED_TRACING = "distributed-tracing";

Practice Notes

Checklist for Distributed Tracing
- Practice: docker up
- Scenario: notification hub
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
