// Correlation IDs — Example 15
// Focus: transaction begin | Scenario: payment bus

// KafkaJS topic: Correlation IDs
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-correlation-ids",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_correlation_ids(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "correlation-ids-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "correlation-ids", label: "Correlation IDs", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CORRELATION_IDS = "correlation-ids";
