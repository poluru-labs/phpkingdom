// Producer Acknowledgments — Template 01

// KafkaJS topic: Producer Acknowledgments
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-producer-acknowledgments",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_producer_acknowledgments(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "producer-acknowledgments-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "producer-acknowledgments", label: "Producer Acknowledgments", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_PRODUCER_ACKNOWLEDGMENTS = "producer-acknowledgments";

Practice Notes

Checklist for Producer Acknowledgments

- Use DLQ/retry topics for poison messages
