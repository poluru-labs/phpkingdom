KafkaJS (Node.js)

// Choreography-Based Saga — Example 08
// Focus: seek offset | Scenario: saga workflow

// KafkaJS topic: Choreography-Based Saga
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-choreography-saga",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_choreography_saga(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "choreography-saga-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "choreography-saga", label: "Choreography-Based Saga", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_CHOREOGRAPHY_SAGA = "choreography-saga";

Practice Notes

Checklist for Choreography-Based Saga
- Practice: seek offset
- Scenario: saga workflow
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
