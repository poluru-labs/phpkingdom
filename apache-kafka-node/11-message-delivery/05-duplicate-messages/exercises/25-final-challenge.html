// Handle Duplicate Messages — Exercise 25

// KafkaJS topic: Handle Duplicate Messages
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-duplicate-messages",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_duplicate_messages(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "duplicate-messages-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "duplicate-messages", label: "Handle Duplicate Messages", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_DUPLICATE_MESSAGES = "duplicate-messages";
