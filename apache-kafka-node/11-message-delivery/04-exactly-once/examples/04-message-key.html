KafkaJS (Node.js)

// Exactly-Once Semantics — Example 04
// Focus: message key | Scenario: log pipeline

import { Kafka } from "kafkajs";
const kafka = new Kafka({ clientId: "eos", brokers: ["localhost:9092"] });

export async function transactionalSend() {
  const producer = kafka.producer({
    transactionalId: "eos-orders-1",
    idempotent: true,
    maxInFlightRequests: 1,
  });
  await producer.connect();
  const tx = await producer.transaction();
  try {
    await tx.send({ topic: "orders", messages: [{ value: "1" }] });
    await tx.commit();
  } catch (e) {
    await tx.abort();
    throw e;
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_ID = "exactly-once";

Practice Notes

Checklist for Exactly-Once Semantics
- Practice: message key
- Scenario: log pipeline
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
