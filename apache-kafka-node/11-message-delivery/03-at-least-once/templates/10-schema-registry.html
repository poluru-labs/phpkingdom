// At-Least-Once Delivery — Template 10

// KafkaJS topic: At-Least-Once Delivery
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-at-least-once",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_at_least_once(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "at-least-once-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "at-least-once", label: "At-Least-Once Delivery", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_AT_LEAST_ONCE = "at-least-once";

Practice Notes

Checklist for At-Least-Once Delivery
- Practice: schema registry
- Scenario: audit trail
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
