KafkaJS (Node.js)

// Kafka Application Testing — Exercise 15
// Focus: outbox poll | Scenario: payment bus

// KafkaJS topic: Kafka Application Testing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-testing",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_testing(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "testing-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "testing", label: "Kafka Application Testing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_TESTING = "testing";

Practice Notes

Checklist for Kafka Application Testing
- Practice: outbox poll
- Scenario: payment bus
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
