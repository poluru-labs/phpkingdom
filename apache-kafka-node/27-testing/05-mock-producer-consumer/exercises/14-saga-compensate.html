// Mock Producers and Consumers — Exercise 14
// Focus: saga compensate | Scenario: log pipeline

// KafkaJS topic: Mock Producers and Consumers
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-mock-producer-consumer",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_mock_producer_consumer(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "mock-producer-consumer-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "mock-producer-consumer", label: "Mock Producers and Consumers", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_MOCK_PRODUCER_CONSUMER = "mock-producer-consumer";
