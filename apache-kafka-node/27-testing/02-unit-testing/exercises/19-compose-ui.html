KafkaJS (Node.js)

// Kafka Unit Testing — Exercise 19
// Focus: compose ui | Scenario: analytics stream

// KafkaJS topic: Kafka Unit Testing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-unit-testing",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_unit_testing(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "unit-testing-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "unit-testing", label: "Kafka Unit Testing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_UNIT_TESTING = "unit-testing";

Practice Notes

Checklist for Kafka Unit Testing
- Practice: compose ui
- Scenario: analytics stream
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers

Useful commands:
  docker compose up -d
  npm install kafkajs
  node producer.js
  node consumer.js
  kafka-topics.sh --bootstrap-server localhost:9092 --list
