// Kafka Integration Testing — Template 23

// KafkaJS topic: Kafka Integration Testing
import { Kafka, logLevel } from "kafkajs";

const kafka = new Kafka({
  clientId: "phpkingdom-integration-testing",
  brokers: [process.env.KAFKA_BROKERS || "localhost:9092"],
  logLevel: logLevel.INFO,
});

export async function demo_integration_testing(payload = {}) {
  const producer = kafka.producer({ idempotent: true });
  await producer.connect();
  try {
    await producer.send({
      topic: process.env.KAFKA_TOPIC || "integration-testing-events",
      messages: [{
        key: payload.key || "demo",
        value: JSON.stringify({ topic: "integration-testing", label: "Kafka Integration Testing", ...payload }),
      }],
    });
  } finally {
    await producer.disconnect();
  }
}

export const TOPIC_INTEGRATION_TESTING = "integration-testing";

Practice Notes

Checklist for Kafka Integration Testing
- Practice: custom partitioner
- Scenario: activity tracker
- Prefer idempotent producers and explicit keys for ordering
- Commit offsets after successful processing (at-least-once)
- Use DLQ/retry topics for poison messages
- Never log secrets; include correlation IDs in headers
