// Files and Patterns — Template 13
// Focus: html reporter kit | Scenario: component library

// files + exclude patterns control what Karma serves
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: [
      "src/vendor/shim.js",
      { pattern: "src/assets/**/*", watched: false, included: false, served: true },
      "src/**/*.js",
      "src/**/*.spec.js",
    ],
    exclude: [
      "src/**/*.e2e.js",
      "src/**/*.stories.js",
    ],
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
// included:false + served:true for images/fixtures loaded via /base/...
