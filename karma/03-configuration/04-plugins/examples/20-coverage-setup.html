// Karma Plugins — Example 20
// Focus: coverage setup | Scenario: documentation site

// Explicit plugins list (recommended in monorepos / CI)
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    plugins: [
      require("karma-jasmine"),
      require("karma-chrome-launcher"),
      require("karma-coverage"),
      require("karma-spec-reporter"),
      require("karma-junit-reporter"),
    ],
    files: ["src/**/*.spec.js"],
    reporters: ["spec", "coverage", "junit"],
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
// If plugins is omitted, Karma auto-loads karma-* from node_modules

Practice Notes

Checklist for Karma Plugins
- Practice: coverage setup
- Scenario: documentation site
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
