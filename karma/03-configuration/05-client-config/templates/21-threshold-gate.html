// Client Config — Template 21
// Focus: threshold gate | Scenario: Angular app

// client: options passed into the browser test iframe
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["Chrome"],
    singleRun: false,
    client: {
      captureConsole: true,
      clearContext: false,
      jasmine: {
        random: false,
        timeoutInterval: 10000,
        failFast: false,
      },
      args: ["--grep", "smoke"],
    },
  });
};

Practice Notes

Checklist for Client Config

Useful commands:

  npx karma init
