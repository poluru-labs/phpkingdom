// Client Config — Example 12
// Focus: headless chrome | Scenario: legacy SPA

// client: options passed into the browser test iframe
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["Chrome"],
    singleRun: false,
    client: {
      captureConsole: true,
      clearContext: false,
      jasmine: {
        random: false,
        timeoutInterval: 10000,
        failFast: false,
      },
      args: ["--grep", "smoke"],
    },
  });
};

Practice Notes

Checklist for Client Config
- Practice: headless chrome
- Scenario: legacy SPA

- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates

Useful commands:

  npx karma init
