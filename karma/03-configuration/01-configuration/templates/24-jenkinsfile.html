// Karma Configuration — Template 24
// Focus: jenkinsfile | Scenario: admin dashboard

// Deep configuration tour — logging, colors, timeouts
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["ChromeHeadless"],
    colors: true,
    logLevel: config.LOG_INFO,
    browserConsoleLogOptions: { level: "log", format: "%b %T: %m", terminal: true },
    captureTimeout: 60000,
    browserDisconnectTolerance: 2,
    reportSlowerThan: 500,
    singleRun: true,
  });
};

Practice Notes

Checklist for Karma Configuration
- Practice: jenkinsfile
- Scenario: admin dashboard
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
