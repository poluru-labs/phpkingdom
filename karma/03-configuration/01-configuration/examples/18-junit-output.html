// Karma Configuration — Example 18
// Focus: junit output | Scenario: TypeScript SDK

// Deep configuration tour — logging, colors, timeouts
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["ChromeHeadless"],
    colors: true,
    logLevel: config.LOG_INFO,
    browserConsoleLogOptions: { level: "log", format: "%b %T: %m", terminal: true },
    captureTimeout: 60000,
    browserDisconnectTolerance: 2,
    reportSlowerThan: 500,
    singleRun: true,
  });
};

Checklist for Karma Configuration

- Scenario: TypeScript SDK

- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:

  npx karma init
