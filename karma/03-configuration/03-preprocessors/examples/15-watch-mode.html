// Preprocessors — Example 15
// Focus: watch mode | Scenario: design system

// preprocessors transform files before the browser sees them
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/test.ts"],
    preprocessors: {
      "src/test.ts": ["webpack", "sourcemap"],
      "src/**/*.js": ["coverage"],
    },
    // npm i -D karma-webpack karma-sourcemap-loader karma-coverage
    webpack: require("./webpack.test.js"),
    webpackMiddleware: { stats: "errors-only" },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};

Checklist for Preprocessors

- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:

  npx karma init
  npx karma start --single-run
