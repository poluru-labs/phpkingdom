// Preprocessors — Exercise 11
// Focus: coverage report | Scenario: Angular app

// preprocessors transform files before the browser sees them
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/test.ts"],
    preprocessors: {
      "src/test.ts": ["webpack", "sourcemap"],
      "src/**/*.js": ["coverage"],
    },
    // npm i -D karma-webpack karma-sourcemap-loader karma-coverage
    webpack: require("./webpack.test.js"),
    webpackMiddleware: { stats: "errors-only" },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
