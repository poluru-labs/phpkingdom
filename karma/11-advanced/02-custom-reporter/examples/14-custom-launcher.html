// Custom reporter plugin sketch
function DotReporter(baseReporterDecorator) {
  baseReporterDecorator(this);
  this.onSpecComplete = function (browser, result) {
    process.stdout.write(result.success ? "." : "F");
  };
  this.onRunComplete = function () {
    process.stdout.write("\n");
  };
}
DotReporter.$inject = ["baseReporterDecorator"];

module.exports = {
  "reporter:dot-lite": ["type", DotReporter],
};

// reporters: ["dot-lite"], plugins: [require("./dot-lite-reporter")]
