// Custom launcher plugin sketch (karma-my-browser-launcher)
function MyBrowserLauncher(baseBrowserDecorator, args) {
  baseBrowserDecorator(this);
  this.name = "MyBrowser";
  this._start = function (url) {
    require("child_process").execFile(args.binary, [url]);
  };
}
MyBrowserLauncher.$inject = ["baseBrowserDecorator", "args"];

module.exports = {
  "launcher:MyBrowser": ["type", MyBrowserLauncher],
};

// karma.conf.js → browsers: ["MyBrowser"], plugins: [require("./my-launcher")]
