// Custom preprocessor — e.g. strip debug flags
function StripDebugPreprocessor(logger) {
  const log = logger.create("preprocessor.strip-debug");
  return function (content, file, done) {
    log.debug("stripping debug from %s", file.originalPath);
    done(content.replace(/debugger;/g, ""));
  };
}
StripDebugPreprocessor.$inject = ["logger"];

module.exports = {
  "preprocessor:strip-debug": ["factory", StripDebugPreprocessor],
};

// preprocessors: { "src/**/*.js": ["strip-debug"] }
// plugins: [require("./strip-debug")]
