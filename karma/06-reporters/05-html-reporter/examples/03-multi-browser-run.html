// HTML Reporter — Example 03
// Focus: multi browser run | Scenario: component library

// HTML report of suite results (browse after the run)
// npm i -D karma-html-reporter

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    reporters: ["progress", "html"],
    htmlReporter: {
      outputFile: "reports/units.html",
      pageTitle: "Karma Unit Tests",
      groupSuites: true,
      useCompactStyle: true,
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
// Open reports/units.html after CI artifacts are published

Practice Notes

Checklist for HTML Reporter
- Practice: multi browser run
- Scenario: component library
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
