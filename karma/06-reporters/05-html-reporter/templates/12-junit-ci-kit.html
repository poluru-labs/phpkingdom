// HTML Reporter — Template 12
// Focus: junit ci kit | Scenario: legacy SPA

// HTML report of suite results (browse after the run)
// npm i -D karma-html-reporter

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    reporters: ["progress", "html"],
    htmlReporter: {
      outputFile: "reports/units.html",
      pageTitle: "Karma Unit Tests",
      groupSuites: true,
      useCompactStyle: true,
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
// Open reports/units.html after CI artifacts are published
