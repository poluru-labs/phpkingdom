// Spec Reporter — Exercise 14
// Focus: angular service | Scenario: admin dashboard

// karma-spec-reporter — readable nested suite output
// npm i -D karma-spec-reporter

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    reporters: ["spec"],
    specReporter: {
      maxLogLines: 5,
      suppressErrorSummary: false,
      suppressFailed: false,
      suppressPassed: false,
      suppressSkipped: true,
      showSpecTiming: true,
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
