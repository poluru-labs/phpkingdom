// JUnit Reporter — Example 21
// Focus: coverage thresholds | Scenario: Angular app

// karma-junit-reporter — XML for CI dashboards
// npm i -D karma-junit-reporter

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    reporters: ["progress", "junit"],
    junitReporter: {
      outputDir: "reports/junit",
      outputFile: "karma-results.xml",
      suite: "karma",
      useBrowserName: false,
      nameFormatter: function (browser, result) { return result.description; },
      classNameFormatter: function (browser, result) { return result.suite.join("."); },
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
