Solution 02 — Minimal karma.conf.js

Create a minimal config with frameworks, files, browsers, and singleRun.

Solution

module.exports = function (config) {
  config.set({
    frameworks: ['jasmine'],
    files: ['src/**/*.js', 'src/**/*.spec.js'],
    browsers: ['ChromeHeadless'],
    singleRun: true
  });
};
