Solution 08 — Coverage Reporter

Add karma-coverage preprocessor and html + text-summary reporters.

Solution

preprocessors: {
  'src/**/*.js': ['coverage']
},
reporters: ['progress', 'coverage'],
coverageReporter: {
  dir: 'coverage/',
  reporters: [
    { type: 'html', subdir: 'html' },
    { type: 'text-summary' }
  ]
}
