Solution 20 — Docker Headless

Give a Dockerfile snippet installing Chrome for Karma.

Solution

FROM node:20-bookworm
RUN apt-get update && apt-get install -y chromium
ENV CHROME_BIN=/usr/bin/chromium
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
CMD ["npm", "test"]
