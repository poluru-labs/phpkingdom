// src/greet.js
function greet(name) {
  return "Hello, " + name;
}

// src/greet.spec.js
describe("greet", function () {
  it("returns a greeting", function () {
    expect(greet("Karma")).toBe("Hello, Karma");
  });
});

// Run: npx karma start --single-run --browsers=ChromeHeadless
