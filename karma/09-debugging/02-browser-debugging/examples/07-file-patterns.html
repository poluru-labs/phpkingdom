// Browser Debugging — Example 07
// Focus: file patterns | Scenario: monorepo package

// Browser DevTools workflow for Karma
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["Chrome"],
    customLaunchers: {
      ChromeDebug: {
        base: "Chrome",
        flags: ["--auto-open-devtools-for-tabs"],
      },
    },
    singleRun: false,
    client: { clearContext: false, captureConsole: true },
  });
};
// Use debugger; inside a spec or break on failed expectations

Practice Notes

Checklist for Browser Debugging
- Practice: file patterns
- Scenario: monorepo package
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
