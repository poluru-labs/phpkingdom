// Debug Test Failures — Example 03
// Focus: multi browser run | Scenario: component library

// Make failures deterministic and actionable
describe("failure hygiene", function () {
  it("avoids flaky randomness", function () {
    expect(2 + 2).toBe(4);
  });

  it("uses precise matchers", function () {
    expect({ id: 1, name: "Ada" }).toEqual(jasmine.objectContaining({ id: 1 }));
  });
});

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    client: { jasmine: { random: false, stopOnFailure: false } },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
