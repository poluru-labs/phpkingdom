// Angular Testing with Karma — Template 23
// Focus: github workflow | Scenario: component library

// Angular CLI already wires Karma — inspect angular.json test target
// builder: @angular-devkit/build-angular:karma

module.exports = function (config) {
  config.set({
    basePath: "",
    frameworks: ["jasmine", "@angular-devkit/build-angular"],
    plugins: [
      require("karma-jasmine"),
      require("karma-chrome-launcher"),
      require("karma-jasmine-html-reporter"),
      require("karma-coverage"),
      require("@angular-devkit/build-angular/plugins/karma"),
    ],
    client: { jasmine: { random: false }, clearContext: false },
    reporters: ["progress", "kjhtml"],
    browsers: ["Chrome"],
    singleRun: false,
    restartOnFileChange: true,
  });
};
// ng test

Practice Notes

Checklist for Angular Testing with Karma
- Practice: github workflow
- Scenario: component library
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
