// Pipe and Directive Testing — Template 08
// Focus: ci headless | Scenario: TypeScript SDK

import { Pipe, PipeTransform, Directive, ElementRef, Input } from "@angular/core";
import { TestBed } from "@angular/core/testing";

@Pipe({ name: "pkUpper" })
class UpperPipe implements PipeTransform {
  transform(v: string) {
    return (v || "").toUpperCase();
  }
}

@Directive({ selector: "[pkHighlight]" })
class HighlightDirective {
  @Input() pkHighlight = "yellow";
  constructor(private el: ElementRef) {}
  ngOnChanges() {
    this.el.nativeElement.style.backgroundColor = this.pkHighlight;
  }
}

describe("Pipe + Directive", () => {
  it("UpperPipe transforms", () => {
    expect(new UpperPipe().transform("karma")).toBe("KARMA");
  });

  it("HighlightDirective is declared", () => {
    TestBed.configureTestingModule({ declarations: [HighlightDirective] });
    expect(HighlightDirective).toBeTruthy();
  });
});
