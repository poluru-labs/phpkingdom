// Service Testing — Template 07
// Focus: coverage project | Scenario: monorepo package

import { TestBed } from "@angular/core/testing";
import { Injectable } from "@angular/core";

@Injectable({ providedIn: "root" })
class ValueService {
  getValue() {
    return "real";
  }
}

describe("ValueService", () => {
  let service: ValueService;

  beforeEach(() => {
    TestBed.configureTestingModule({});
    service = TestBed.inject(ValueService);
  });

  it("returns real value", () => {
    expect(service.getValue()).toBe("real");
  });

  it("can be overridden with a stub", () => {
    TestBed.resetTestingModule();
    TestBed.configureTestingModule({
      providers: [{ provide: ValueService, useValue: { getValue: () => "stub" } }],
    });
    expect(TestBed.inject(ValueService).getValue()).toBe("stub");
  });
});
