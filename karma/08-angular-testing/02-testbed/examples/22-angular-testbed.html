// Angular TestBed — Example 22
// Focus: angular testbed | Scenario: legacy SPA

import { TestBed } from "@angular/core/testing";
import { Component } from "@angular/core";

@Component({ selector: "pk-hello", template: `<p>Hello {{ name }}</p>` })
class HelloComponent {
  name = "Angular";
}

describe("TestBed", () => {
  beforeEach(async () => {
    await TestBed.configureTestingModule({
      declarations: [HelloComponent],
    }).compileComponents();
  });

  it("creates the component", () => {
    const fixture = TestBed.createComponent(HelloComponent);
    expect(fixture.componentInstance).toBeTruthy();
    fixture.detectChanges();
    expect(fixture.nativeElement.textContent).toContain("Hello Angular");
  });
});
// Run via Angular Karma builder: ng test

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
