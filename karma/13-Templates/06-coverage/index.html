Starter Scaffold

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.js", "src/**/*.spec.js"],
    preprocessors: { "src/**/*.js": ["coverage"] },
    reporters: ["progress", "coverage"],
    coverageReporter: {
      dir: "coverage/",
      reporters: [
        { type: "html", subdir: "html" },
        { type: "text-summary" },
      ],
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};

Run

npx karma start --single-run
