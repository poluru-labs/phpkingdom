    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    customLaunchers: {
      ChromeHeadlessCI: {
        base: "ChromeHeadless",
        flags: ["--no-sandbox", "--disable-gpu", "--disable-dev-shm-usage"],
      },
    },
    browsers: ["ChromeHeadlessCI"],
    singleRun: true,
    autoWatch: false,
  });
};
