module.exports = function (config) {
  config.set({
    frameworks: ["mocha", "chai"],
    files: ["src/**/*.spec.js"],
    browsers: ["ChromeHeadless"],
    client: { mocha: { timeout: 5000 } },
    singleRun: true,
  });
};

describe("mocha+chai", function () {
  it("asserts with expect", function () {
    expect([1, 2]).to.have.lengthOf(2);
  });
});
