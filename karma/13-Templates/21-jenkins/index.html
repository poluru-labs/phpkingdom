Starter Scaffold

pipeline {
  agent any
  environment { CI = "true" }
  stages {
    stage("Install") { steps { sh "npm ci" } }
    stage("Test") {
      steps { sh "npm test" }
      post {
        always { junit "reports/junit/*.xml" }
      }
    }
  }
}

Run

npx karma start --single-run
