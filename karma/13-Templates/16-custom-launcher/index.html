module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    customLaunchers: {
      ChromeHeadlessWide: {
        base: "ChromeHeadless",
        flags: ["--window-size=1920,1080", "--no-sandbox"],
      },
    },
    browsers: ["ChromeHeadlessWide"],
    singleRun: true,
  });
};
