// Coverage Thresholds — Template 03
// Focus: mocha project | Scenario: component library

// Fail the build when coverage drops
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.js", "src/**/*.spec.js"],
    preprocessors: { "src/**/*.js": ["coverage"] },
    reporters: ["progress", "coverage"],
    coverageReporter: {
      check: {
        global: {
          statements: 80,
          branches: 70,
          functions: 80,
          lines: 80,
        },
        each: {
          statements: 50,
          branches: 50,
          functions: 50,
          lines: 50,
          overrides: { "src/legacy/**/*.js": { statements: 30 } },
        },
      },
      reporters: [{ type: "text-summary" }],
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};

Practice Notes

Checklist for Coverage Thresholds
- Practice: mocha project
- Scenario: component library
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
