// Coverage Thresholds — Exercise 22
// Focus: custom launcher | Scenario: legacy SPA

// Fail the build when coverage drops
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.js", "src/**/*.spec.js"],
    preprocessors: { "src/**/*.js": ["coverage"] },
    reporters: ["progress", "coverage"],
    coverageReporter: {
      check: {
        global: {
          statements: 80,
          branches: 70,
          functions: 80,
          lines: 80,
        },
        each: {
          statements: 50,
          branches: 50,
          functions: 50,
          lines: 50,
          overrides: { "src/legacy/**/*.js": { statements: 30 } },
        },
      },
      reporters: [{ type: "text-summary" }],
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
