// karma-coverage — Example 12
// Focus: headless chrome | Scenario: legacy SPA

// karma-coverage plugin wiring details
// npm i -D karma-coverage

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    plugins: ["karma-jasmine", "karma-chrome-launcher", "karma-coverage"],
    files: ["src/app.js", "src/app.spec.js"],
    preprocessors: { "src/app.js": ["coverage"] },
    reporters: ["progress", "coverage"],
    coverageReporter: {
      includeAllSources: true,
      instrumenterOptions: { istanbul: { noCompact: true } },
      reporters: [{ type: "lcov", subdir: "lcov" }, { type: "text" }],
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};
