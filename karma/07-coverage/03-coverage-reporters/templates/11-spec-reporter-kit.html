// Coverage Reporters — Template 11
// Focus: spec reporter kit | Scenario: Angular app

// Choose coverage output formats for humans vs CI
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.js", "src/**/*.spec.js"],
    preprocessors: { "src/**/*.js": ["coverage"] },
    reporters: ["coverage"],
    coverageReporter: {
      dir: "coverage/",
      reporters: [
        { type: "html", subdir: "html" },
        { type: "lcovonly", subdir: "lcov" },
        { type: "cobertura", subdir: ".", file: "cobertura.xml" },
        { type: "json-summary", subdir: ".", file: "summary.json" },
        { type: "text-summary" },
      ],
    },
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};

Practice Notes

Checklist for Coverage Reporters
- Practice: spec reporter kit
- Scenario: Angular app
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
