// ChromeHeadless — Example 24
// Focus: service spec | Scenario: admin dashboard

// ChromeHeadless — default CI browser for Karma
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.js", "src/**/*.spec.js"],
    browsers: ["ChromeHeadless"],
    customLaunchers: {
      ChromeHeadlessCI: {
        base: "ChromeHeadless",
        flags: [
          "--no-sandbox",
          "--disable-gpu",
          "--disable-dev-shm-usage",
        ],
      },
    },
    // In CI: browsers: ["ChromeHeadlessCI"]
    singleRun: true,
    autoWatch: false,
  });
};
