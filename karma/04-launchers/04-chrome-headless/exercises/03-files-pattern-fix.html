// ChromeHeadless — Exercise 03
// Focus: files pattern fix | Scenario: component library

// ChromeHeadless — default CI browser for Karma
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.js", "src/**/*.spec.js"],
    browsers: ["ChromeHeadless"],
    customLaunchers: {
      ChromeHeadlessCI: {
        base: "ChromeHeadless",
        flags: [
          "--no-sandbox",
          "--disable-gpu",
          "--disable-dev-shm-usage",
        ],
      },
    },
    // In CI: browsers: ["ChromeHeadlessCI"]
    singleRun: true,
    autoWatch: false,
  });
};
