// Firefox Launcher — Exercise 21
// Focus: jenkins stage | Scenario: Angular app

// karma-firefox-launcher
// npm i -D karma-firefox-launcher

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["FirefoxHeadless"],
    customLaunchers: {
      FirefoxHeadlessCI: {
        base: "FirefoxHeadless",
        prefs: {
          "media.navigator.permission.disabled": true,
        },
      },
    },
    singleRun: true,
  });
};
// Set FIREFOX_BIN if Firefox is not on PATH
