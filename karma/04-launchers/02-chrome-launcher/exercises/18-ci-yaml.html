// Chrome Launcher — Exercise 18
// Focus: ci yaml | Scenario: TypeScript SDK

// karma-chrome-launcher — headed Chrome for local debugging
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["Chrome"],
    customLaunchers: {
      ChromeDebug: {
        base: "Chrome",
        flags: ["--auto-open-devtools-for-tabs", "--window-size=1280,800"],
      },
    },
    singleRun: false,
    autoWatch: true,
  });
};
