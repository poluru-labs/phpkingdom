// Chrome Launcher — Template 23
// Focus: github workflow | Scenario: component library

// karma-chrome-launcher — headed Chrome for local debugging
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["Chrome"],
    customLaunchers: {
      ChromeDebug: {
        base: "Chrome",
        flags: ["--auto-open-devtools-for-tabs", "--window-size=1280,800"],
      },
    },
    singleRun: false,
    autoWatch: true,
  });
};
