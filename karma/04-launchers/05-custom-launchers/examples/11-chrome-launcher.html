// Custom Launchers — Example 11
// Focus: chrome launcher | Scenario: Angular app

// customLaunchers: clone a base launcher and add flags/prefs
module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    customLaunchers: {
      ChromeHeadlessWide: {
        base: "ChromeHeadless",
        flags: ["--window-size=1920,1080", "--no-sandbox"],
      },
      ChromeWithoutSecurity: {
        base: "Chrome",
        flags: ["--disable-web-security", "--user-data-dir=/tmp/karma-chrome"],
      },
    },
    browsers: ["ChromeHeadlessWide"],
    singleRun: true,
  });
};

Practice Notes

Checklist for Custom Launchers
- Practice: chrome launcher
- Scenario: Angular app
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
