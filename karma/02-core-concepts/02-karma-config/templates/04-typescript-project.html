// karma.conf.js — Template 04
// Focus: typescript project | Scenario: admin dashboard

// karma.conf.js — full starter with common options
module.exports = function (config) {
  config.set({
    basePath: "",
    frameworks: ["jasmine"],
    plugins: [
      "karma-jasmine",
      "karma-chrome-launcher",
      "karma-coverage",
    ],
    files: [
      { pattern: "src/**/*.js", watched: true },
      { pattern: "src/**/*.spec.js", watched: true },
    ],
    exclude: ["src/**/*.e2e.js"],
    preprocessors: { "src/**/*.js": ["coverage"] },
    reporters: ["progress", "coverage"],
    coverageReporter: { type: "text-summary" },
    browsers: ["ChromeHeadless"],
    singleRun: true,
    autoWatch: false,
    client: { jasmine: { random: false }, clearContext: false },
  });
};

Practice Notes

Checklist for karma.conf.js
- Practice: typescript project
- Scenario: admin dashboard
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
