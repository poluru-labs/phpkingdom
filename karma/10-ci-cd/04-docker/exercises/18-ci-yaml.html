# Dockerfile — Node + Chromium for Karma CI
FROM node:20-bookworm

RUN apt-get update && apt-get install -y chromium \
  && rm -rf /var/lib/apt/lists/*

ENV CHROME_BIN=/usr/bin/chromium
ENV CI=true

WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
CMD ["npm", "test"]

# docker build -t karma-ci .
# docker run --rm karma-ci
