// Karma CI/CD — Template 16
// Focus: chrome only | Scenario: CI pipeline

// CI checklist for Karma
// - browsers: ChromeHeadless (+ CI flags)
// - singleRun: true, autoWatch: false
// - reporters: progress + junit (+ coverage)

module.exports = function (config) {
  const isCI = !!process.env.CI;
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: [isCI ? "ChromeHeadlessCI" : "Chrome"],
    customLaunchers: {
      ChromeHeadlessCI: {
        base: "ChromeHeadless",
        flags: ["--no-sandbox", "--disable-gpu", "--disable-dev-shm-usage"],
      },
    },
    singleRun: isCI,
    autoWatch: !isCI,
    reporters: isCI ? ["progress", "junit"] : ["progress"],
  });
};

Practice Notes

Checklist for Karma CI/CD
- Practice: chrome only
- Scenario: CI pipeline
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
