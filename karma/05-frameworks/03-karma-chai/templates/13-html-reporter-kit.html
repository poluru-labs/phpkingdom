// Karma with Chai — Template 13
// Focus: html reporter kit | Scenario: component library

// karma-chai pairs with mocha
// npm i -D karma-mocha mocha karma-chai chai

module.exports = function (config) {
  config.set({
    frameworks: ["mocha", "chai"],
    files: ["src/**/*.spec.js"],
    browsers: ["ChromeHeadless"],
    singleRun: true,
  });
};

describe("chai expect", function () {
  it("supports BDD assertions", function () {
    expect([1, 2]).to.include(2).and.to.have.lengthOf(2);
    expect({ a: 1 }).to.have.property("a", 1);
  });
  it("supports assert style", function () {
    assert.strictEqual(2 + 2, 4);
  });
});

Practice Notes

Checklist for Karma with Chai
- Practice: html reporter kit
- Scenario: component library
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
