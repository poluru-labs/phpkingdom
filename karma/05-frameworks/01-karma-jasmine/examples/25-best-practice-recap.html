// Karma with Jasmine — Example 25
// Focus: best practice recap | Scenario: design system

// karma-jasmine + jasmine-core
// npm i -D karma-jasmine jasmine-core

module.exports = function (config) {
  config.set({
    frameworks: ["jasmine"],
    files: ["src/**/*.spec.js"],
    browsers: ["ChromeHeadless"],
    client: { jasmine: { random: false } },
    singleRun: true,
  });
};

describe("cart", function () {
  beforeEach(function () {
    this.items = [];
  });
  it("starts empty", function () {
    expect(this.items.length).toBe(0);
  });
  it("pushes items", function () {
    this.items.push("sku-1");
    expect(this.items).toEqual(["sku-1"]);
  });
});

Practice Notes

Checklist for Karma with Jasmine
- Practice: best practice recap
- Scenario: design system
- Prefer ChromeHeadless (+ CI flags) for automation
- Keep singleRun/autoWatch gated on process.env.CI
- Pair frameworks (Jasmine/Mocha) with explicit plugins[]
- Add junit + coverage reporters for CI gates
- Open /debug.html with headed Chrome when diagnosing flakes

Useful commands:
  npm i -D karma karma-jasmine jasmine-core karma-chrome-launcher
  npx karma init
  npx karma start --single-run
  npx karma start --browsers=Chrome
