import { addErrorHandler, getAppStatus } from "single-spa";

addErrorHandler((err) => {
  console.error("single-spa error", err.appOrParcelName, err);
  const el = document.getElementById("mfe-error");
  if (el) {
    el.hidden = false;
    el.textContent = "Failed to load " + err.appOrParcelName;
  }
  console.log("status", getAppStatus(err.appOrParcelName));
});
