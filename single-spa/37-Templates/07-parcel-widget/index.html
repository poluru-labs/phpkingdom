import { mountRootParcel } from "single-spa";

const parcelConfig = {
  async bootstrap() {},
  async mount(props) {
    props.domElement.textContent = "Parcel: " + (props.label || "widget");
  },
  async unmount(props) {
    props.domElement.textContent = "";
  },
};

export function mountWidget(domElement, label) {
  return mountRootParcel(parcelConfig, { domElement, label });
}
