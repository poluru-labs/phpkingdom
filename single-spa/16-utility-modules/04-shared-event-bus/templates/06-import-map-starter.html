// Shared Event Bus — Template 06
// Focus: import map starter | Scenario: design system CDN

const bus = {
  emit(type, detail) {
    window.dispatchEvent(new CustomEvent(type, { detail }));
  },
  on(type, handler) {
    window.addEventListener(type, handler);
    return () => window.removeEventListener(type, handler);
  },
};

bus.emit("notify", { message: "Hello from MFE" });
