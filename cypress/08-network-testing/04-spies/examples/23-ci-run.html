// Spies — Example 23
// Focus: ci run | Scenario: checkout flow

describe("cy.spy", () => {
  it("spies on console.error during a flow", () => {
    cy.visit("/checkout", {
      onBeforeLoad(win) {
        cy.spy(win.console, "error").as("consoleError");
      },
    });
    cy.get("[data-cy=pay]").click();
    cy.get("@consoleError").should("not.have.been.called");
  });
});
