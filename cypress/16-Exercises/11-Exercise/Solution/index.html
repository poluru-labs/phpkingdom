Solution 11 — cy.session Login

Cache login with cy.session.

Solution

beforeEach(() => {
  cy.session('user', () => {
    cy.visit('/login');
    cy.login('user@phpkingdom.com', 'secret');
    cy.url().should('include', '/app');
  });
  cy.visit('/app');
});
