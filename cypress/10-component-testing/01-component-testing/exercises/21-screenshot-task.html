// Component Testing — Exercise 21
// Focus: screenshot task | Scenario: SaaS dashboard

// cypress.config.js — component: { devServer: { framework: "react", bundler: "vite" } }
// cypress/support/component.js → import { mount } from "cypress/react"

import Button from "./Button";

describe("Button component", () => {
  it("renders the label and emits click", () => {
    const onClick = cy.stub().as("onClick");
    cy.mount(<Button onClick={onClick}>Save</Button>);
    cy.contains("button", "Save").click();
    cy.get("@onClick").should("have.been.calledOnce");
  });
});

Practice Notes

Checklist for Component Testing
- Practice: screenshot task
- Scenario: SaaS dashboard
- Prefer data-cy / role-based selectors over brittle CSS
- Lean on Cypress retry-ability; avoid cy.wait(ms)
- Isolate tests; use cy.session for auth speed
- Stub flaky network with cy.intercept; assert with cy.wait('@alias')
- Keep e2e for journeys; use component testing for UI units

Useful commands:
  npm install --save-dev cypress
  npx cypress open
  npx cypress run
  npx cypress run --record --parallel
  npx cypress run --browser chrome
