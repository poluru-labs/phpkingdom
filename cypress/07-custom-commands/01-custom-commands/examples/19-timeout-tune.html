// Custom Commands — Example 19
// Focus: timeout tune | Scenario: docs site

// cypress/support/commands.js
Cypress.Commands.add("login", (email, password) => {
  cy.session([email], () => {
    cy.visit("/login");
    cy.get("[data-cy=email]").type(email);
    cy.get("[data-cy=password]").type(password, { log: false });
    cy.get("[data-cy=submit]").click();
    cy.url().should("include", "/app");
  });
});

// usage
describe("custom commands", () => {
  it("logs in via cy.login", () => {
    cy.login("ada@phpkingdom.com", "secret");
    cy.visit("/app");
    cy.get("[data-cy=user-menu]").should("contain", "Ada");
  });
});

Practice Notes

Checklist for Custom Commands
- Practice: timeout tune
- Scenario: docs site
- Prefer data-cy / role-based selectors over brittle CSS
- Lean on Cypress retry-ability; avoid cy.wait(ms)
- Isolate tests; use cy.session for auth speed
- Stub flaky network with cy.intercept; assert with cy.wait('@alias')
- Keep e2e for journeys; use component testing for UI units
