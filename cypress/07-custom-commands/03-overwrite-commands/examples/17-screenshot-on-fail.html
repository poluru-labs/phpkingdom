// Overwrite Commands — Example 17
// Focus: screenshot on fail | Scenario: API-backed SPA

// Soften noisy visit logs / add default qs
Cypress.Commands.overwrite("visit", (originalFn, url, options = {}) => {
  const next = {
    ...options,
    qs: { utm_source: "cypress", ...(options.qs || {}) },
  };
  return originalFn(url, next);
});

describe("overwrite visit", () => {
  it("still visits as usual", () => {
    cy.visit("/campaigns");
    cy.location("search").should("include", "utm_source=cypress");
  });
});
