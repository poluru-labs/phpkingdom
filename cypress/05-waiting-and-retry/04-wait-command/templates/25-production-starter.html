// cy.wait() — Template 25
// Focus: production starter | Scenario: auth portal

describe("cy.wait", () => {
  it("waits on an aliased network call", () => {
    cy.intercept("GET", "/api/orders*").as("orders");
    cy.visit("/orders");
    cy.wait("@orders").its("response.statusCode").should("eq", 200);
    cy.get("[data-cy=order-row]").should("have.length.greaterThan", 0);
  });
});
// Prefer cy.wait("@alias") over cy.wait(5000)
