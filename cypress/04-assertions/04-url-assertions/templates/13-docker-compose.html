// URL Assertions — Template 13

describe("URL assertions", () => {
  it("asserts path, query, and hash", () => {
    cy.visit("/products?sort=price#featured");
    cy.location("pathname").should("eq", "/products");
    cy.location("search").should("include", "sort=price");
    cy.hash().should("eq", "#featured");
    cy.url().should("include", "/products");
  });
});
