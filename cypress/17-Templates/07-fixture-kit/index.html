Starter Spec

// cypress/fixtures/users.json → [{ "email": "ada@phpkingdom.com" }]

describe("fixture-driven login", () => {
  it("types email from users.json", () => {
    cy.fixture("users").then((users) => {
      cy.visit("/login");
      cy.get("[data-cy=email]").type(users[0].email);
      cy.get("[data-cy=password]").type("secret", { log: false });
      cy.get("[data-cy=submit]").click();
    });
  });
});
