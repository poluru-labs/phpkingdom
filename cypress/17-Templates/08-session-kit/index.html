Starter Spec

describe("session kit", () => {
  beforeEach(() => {
    cy.session("ada", () => {
      cy.visit("/login");
      cy.get("[data-cy=email]").type("ada@phpkingdom.com");
      cy.get("[data-cy=password]").type("secret", { log: false });
      cy.get("[data-cy=submit]").click();
      cy.url().should("include", "/app");
    });
  });

  it("opens billing authenticated", () => {
    cy.visit("/billing");
    cy.get("[data-cy=plan]").should("be.visible");
  });
});
