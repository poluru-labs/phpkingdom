Starter Spec

import { Counter } from "./Counter";

describe("<Counter />", () => {
  it("increments", () => {
    cy.mount(<Counter initial={0} />);
    cy.get("[data-cy=inc]").click();
    cy.contains("[data-cy=count]", "1");
  });
});
