Environment switching

Starter Spec

// cypress.config.js
// env: { apiUrl: "http://localhost:4000" }
// CLI: CYPRESS_baseUrl=https://staging.example.com \
//      CYPRESS_apiUrl=https://api.staging.example.com \
//      npx cypress run

describe("staging env", () => {
  it("uses env apiUrl", () => {
    cy.request(`${Cypress.env("apiUrl")}/health`).its("status").should("eq", 200);
  });
});

Run

npx cypress open
npx cypress run
