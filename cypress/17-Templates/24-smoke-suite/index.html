Starter Spec

describe("smoke", { tags: ["@smoke"] }, () => {
  it("home loads", () => {
    cy.visit("/");
    cy.get("[data-cy=app]").should("be.visible");
  });

  it("login page reachable", () => {
    cy.visit("/login");
    cy.get("[data-cy=submit]").should("be.visible");
  });

  it("health API ok", () => {
    cy.request("/api/health").its("status").should("eq", 200);
  });
});
