Starter Spec

describe("production checklist", () => {
  beforeEach(() => {
    cy.intercept("GET", "/api/bootstrap", { fixture: "bootstrap.json" }).as("boot");
    cy.session("prod-user", () => {
      cy.visit("/login");
      cy.get("[data-cy=email]").type(Cypress.env("userEmail"));
      cy.get("[data-cy=password]").type(Cypress.env("userPassword"), { log: false });
      cy.get("[data-cy=submit]").click();
    });
  });

  it("dashboard + a11y smoke", () => {
    cy.visit("/dashboard");
    cy.wait("@boot");
    cy.get("[data-cy=dashboard]").should("be.visible");
    cy.injectAxe();
    cy.checkA11y("[data-cy=dashboard]", { includedImpacts: ["critical"] });
  });
});
