Form login with data-cy

Starter Spec

describe("login", () => {
  it("signs in with valid credentials", () => {
    cy.visit("/login");
    cy.get("[data-cy=email]").type("ada@phpkingdom.com");
    cy.get("[data-cy=password]").type("secret", { log: false });
    cy.get("[data-cy=submit]").click();
    cy.location("pathname").should("eq", "/app");
    cy.get("[data-cy=user-menu]").should("contain", "Ada");
  });
});

Run

npx cypress open
npx cypress run
