// cypress/support/factories/user.js
export function buildUser(overrides = {}) {
  const n = Date.now();
  return {
    name: "Test User",
    email: `user+${n}@phpkingdom.com`,
    password: "secret",
    ...overrides,
  };
}

describe("data factory", () => {
  it("registers a unique user", () => {
    const user = buildUser({ name: "Ada" });
    cy.visit("/register");
    cy.get("[data-cy=name]").type(user.name);
    cy.get("[data-cy=email]").type(user.email);
    cy.get("[data-cy=password]").type(user.password, { log: false });
    cy.get("[data-cy=submit]").click();
  });
});
