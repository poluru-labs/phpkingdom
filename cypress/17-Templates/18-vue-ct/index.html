Starter Spec

import Toggle from "./Toggle.vue";

describe("Toggle.vue", () => {
  it("toggles pressed state", () => {
    cy.mount(Toggle, { props: { label: "Dark" } });
    cy.get("[data-cy=toggle]")
      .should("have.attr", "aria-pressed", "false")
      .click()
      .should("have.attr", "aria-pressed", "true");
  });
});
