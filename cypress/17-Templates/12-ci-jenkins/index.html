Starter Spec

pipeline {
  agent any
  stages {
    stage("Install") { steps { sh "npm ci" } }
    stage("Build") { steps { sh "npm run build" } }
    stage("Cypress") {
      steps { sh "npx cypress run --browser chrome" }
      post {
        always {
          archiveArtifacts artifacts: "cypress/videos/**,cypress/screenshots/**", allowEmptyArchive: true
        }
      }
    }
  }
}

Run

npx cypress open
npx cypress run
