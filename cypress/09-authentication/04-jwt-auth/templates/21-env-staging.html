// JWT Auth Testing — Template 21
// Focus: env staging | Scenario: SaaS dashboard

describe("JWT auth testing", () => {
  it("injects a bearer token for API + UI", () => {
    cy.request("POST", "/api/auth/login", {
      email: "ada@phpkingdom.com",
      password: "secret",
    }).then(({ body }) => {
      window.localStorage.setItem("token", body.accessToken);
    });
    cy.visit("/app");
    cy.request({
      url: "/api/me",
      headers: { Authorization: `Bearer ${window.localStorage.getItem("token")}` },
    }).its("status").should("eq", 200);
  });
});

Checklist for JWT Auth Testing

- Stub flaky network with cy.intercept; assert with cy.wait('@alias')
