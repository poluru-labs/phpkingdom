// cy.session() — Example 04
// Focus: navigation | Scenario: admin panel

describe("cy.session", () => {
  beforeEach(() => {
    cy.session("ada", () => {
      cy.visit("/login");
      cy.get("[data-cy=email]").type("ada@phpkingdom.com");
      cy.get("[data-cy=password]").type("secret", { log: false });
      cy.get("[data-cy=submit]").click();
      cy.url().should("include", "/app");
    });
  });

  it("opens billing already authenticated", () => {
    cy.visit("/billing");
    cy.get("[data-cy=plan]").should("be.visible");
  });

  it("opens settings already authenticated", () => {
    cy.visit("/settings");
    cy.get("[data-cy=profile-form]").should("exist");
  });
});
