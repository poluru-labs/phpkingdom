// Test Data — Template 07
// Focus: fixture kit | Scenario: API-backed SPA

const userFactory = (overrides = {}) => ({
  email: `user+${Date.now()}@phpkingdom.com`,
  name: "Test User",
  ...overrides,
});

describe("test data", () => {
  it("creates unique users per run", () => {
    const user = userFactory({ name: "Ada" });
    cy.visit("/register");
    cy.get("[data-cy=name]").type(user.name);
    cy.get("[data-cy=email]").type(user.email);
    cy.get("[data-cy=submit]").click();
    cy.contains(user.email).should("be.visible");
  });
});

Practice Notes

Checklist for Test Data
- Practice: fixture kit
- Scenario: API-backed SPA
- Prefer data-cy / role-based selectors over brittle CSS
- Lean on Cypress retry-ability; avoid cy.wait(ms)
- Isolate tests; use cy.session for auth speed
- Stub flaky network with cy.intercept; assert with cy.wait('@alias')
- Keep e2e for journeys; use component testing for UI units
