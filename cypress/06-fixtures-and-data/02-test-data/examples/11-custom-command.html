// Test Data — Example 11
// Focus: custom command | Scenario: SaaS dashboard

const userFactory = (overrides = {}) => ({
  email: `user+${Date.now()}@phpkingdom.com`,
  name: "Test User",
  ...overrides,
});

describe("test data", () => {
  it("creates unique users per run", () => {
    const user = userFactory({ name: "Ada" });
    cy.visit("/register");
    cy.get("[data-cy=name]").type(user.name);
    cy.get("[data-cy=email]").type(user.email);
    cy.get("[data-cy=submit]").click();
    cy.contains(user.email).should("be.visible");
  });
});
