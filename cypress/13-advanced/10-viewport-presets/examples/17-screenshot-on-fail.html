// Viewport Presets — Example 17
// Focus: screenshot on fail | Scenario: API-backed SPA

describe("viewport presets", () => {
  const presets = [
    ["iphone-6", "mobile"],
    ["ipad-2", "tablet"],
    ["macbook-15", "desktop"],
  ];

  presets.forEach(([preset, label]) => {
    it(`renders nav on ${label} (${preset})`, () => {
      cy.viewport(preset);
      cy.visit("/");
      cy.get("[data-cy=nav]").should("be.visible");
    });
  });

  it("custom dimensions", () => {
    cy.viewport(360, 800);
    cy.visit("/");
    cy.get("[data-cy=nav-toggle]").should("be.visible");
  });
});
