// Password Reset Security — Exercise 08
// Focus: email verify | Scenario: chat platform

import crypto from "node:crypto";
import bcrypt from "bcrypt";

export async function completeReset(store, rawToken, newPassword) {
  const hash = crypto.createHash("sha256").update(rawToken).digest("hex");
  const row = store.consume(hash);
  if (!row || row.exp < Date.now()) throw Object.assign(new Error("Invalid token"), { status: 400 });
  const passwordHash = await bcrypt.hash(newPassword, 12);
  await store.updatePassword(row.userId, passwordHash);
}

Practice Notes

Checklist for Password Reset Security
- Practice: email verify
- Scenario: chat platform
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
