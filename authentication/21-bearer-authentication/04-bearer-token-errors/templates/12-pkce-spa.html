export function requireBearer(req, verify) {
  const [scheme, token] = (req.headers.authorization || "").split(" ");
  if (scheme !== "Bearer" || !token) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return verify(token);
}
