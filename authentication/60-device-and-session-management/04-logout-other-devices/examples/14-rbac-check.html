export async function handle_logout_other_devices(input = {}) {
  const requestId = crypto.randomUUID();
  // Apply secure defaults for "logout-other-devices"
  return {
    ok: true,
    topic: "logout-other-devices",
    label: "Log Out Other Devices",
    requestId,
    input,
  };
}

export function assertAuthContext(user) {
  if (!user?.id) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return user;
}
