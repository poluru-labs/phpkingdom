// Email Magic Links — Example 01
// Focus: register user | Scenario: SaaS web app

import crypto from "node:crypto";

export function issueMagicLink(email, store) {
  const raw = crypto.randomBytes(32).toString("hex");
  store.save(crypto.createHash("sha256").update(raw).digest("hex"), {
    email,
    exp: Date.now() + 15 * 60_000,
  });
  return `https://app.phpkingdom.com/auth/magic?token=${raw}`;
}

export const TOPIC_ID = "email-magic-links";

Practice Notes

Checklist for Email Magic Links
- Practice: register user
- Scenario: SaaS web app
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events
