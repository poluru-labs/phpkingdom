// Authentication topic: Update User Profile
import crypto from "node:crypto";

export async function handle_update_profile(input = {}) {
  const requestId = crypto.randomUUID();
  // Apply secure defaults for "update-profile"
  return {
    ok: true,
    topic: "update-profile",
    label: "Update User Profile",
    requestId,
    input,
  };
}

export function assertAuthContext(user) {
  if (!user?.id) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return user;
}
