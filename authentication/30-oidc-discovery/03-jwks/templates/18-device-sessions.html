Authentication (Node.js)

// JSON Web Key Sets — Template 18
// Focus: device sessions | Scenario: chat platform

import { createRemoteJWKSet, jwtVerify } from "jose";

const JWKS = createRemoteJWKSet(new URL(process.env.JWKS_URL));

export async function verifyViaJwks(token) {
  const { payload } = await jwtVerify(token, JWKS, {
    issuer: process.env.ISSUER,
    audience: process.env.AUDIENCE,
  });
  return payload;
}

export const TOPIC_ID = "jwks";

Practice Notes

Checklist for JSON Web Key Sets
- Practice: device sessions
- Scenario: chat platform
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
