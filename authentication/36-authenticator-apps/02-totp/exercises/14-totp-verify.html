// Time-Based One-Time Passwords — Exercise 14
// Focus: totp verify | Scenario: admin console

import { authenticator } from "otplib";

export function setupTotp(email) {
  const secret = authenticator.generateSecret();
  const otpauth = authenticator.keyuri(email, "phpkingdom.com", secret);
  return { secret, otpauth };
}

export function verifyTotp(token, secret) {
  return authenticator.check(token, secret);
}

export const TOPIC_ID = "totp";

Practice Notes

Checklist for Time-Based One-Time Passwords
- Practice: totp verify
- Scenario: admin console
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
