import crypto from "node:crypto";

export async function handle_nestjs_passport(input = {}) {
  const requestId = crypto.randomUUID();
  // Apply secure defaults for "nestjs-passport"
  return {
    ok: true,
    topic: "nestjs-passport",
    label: "NestJS Passport Integration",
    requestId,
    input,
  };
}

export function assertAuthContext(user) {
  if (!user?.id) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return user;
}

Practice Notes

Checklist for NestJS Passport Integration
- Practice: zero trust
- Scenario: healthcare portal
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
