// Authentication proves identity; authorization decides what that identity may do.
export async function register({ email, password }, users, hash) {
  if (users.has(email)) throw Object.assign(new Error("exists"), { status: 409 });
  const user = { id: cryptoRandom(), email, passwordHash: await hash(password) };
  users.set(email, user);
  return { id: user.id, email };
}

function cryptoRandom() {
  return require("node:crypto").randomUUID();
}

Practice Notes

Checklist for Authentication - Introduction
- Practice: oauth redirect
- Scenario: chat platform
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
