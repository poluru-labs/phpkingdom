// WebAuthn authentication options
import { generateAuthenticationOptions } from "@simplewebauthn/server";

export async function beginPasskeyLogin(allowCredentials) {
  return generateAuthenticationOptions({
    rpID: "phpkingdom.com",
    allowCredentials,
    userVerification: "preferred",
  });
}

Practice Notes

Checklist for WebAuthn Authentication
- Practice: sms otp
- Scenario: fintech portal
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me

Notes
