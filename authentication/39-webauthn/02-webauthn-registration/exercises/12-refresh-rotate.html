// WebAuthn registration options (server sketch)
import { generateRegistrationOptions } from "@simplewebauthn/server";

export async function beginPasskeyRegistration(user) {
  return generateRegistrationOptions({
    rpName: "phpkingdom.com",
    rpID: "phpkingdom.com",
    userID: Buffer.from(user.id),
    userName: user.email,
    attestationType: "none",
    authenticatorSelection: { residentKey: "preferred", userVerification: "preferred" },
  });
}

export const TOPIC_ID = "webauthn-registration";

Practice Notes

Checklist for WebAuthn Registration
- Practice: refresh rotate
- Scenario: mobile BFF
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
