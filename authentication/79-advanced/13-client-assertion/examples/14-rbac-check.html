// Client Assertion — Example 14
// Focus: rbac check | Scenario: admin console

// private_key_jwt client authentication
import jwt from "jsonwebtoken";
import crypto from "node:crypto";

export function clientAssertion({ clientId, tokenEndpoint, privateKey }) {
  const now = Math.floor(Date.now() / 1000);
  return jwt.sign(
    { iss: clientId, sub: clientId, aud: tokenEndpoint, jti: crypto.randomUUID(), iat: now, exp: now + 60 },
    privateKey,
    { algorithm: "RS256" }
  );
}

export const TOPIC_ID = "client-assertion";

Practice Notes

Checklist for Client Assertion
- Practice: rbac check
- Scenario: admin console
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
