// OAuth 2.1 — Example 17
// Focus: rotate keys | Scenario: developer API

// OAuth 2.1 mental model: authorization code + PKCE for public clients;
// no implicit / password grants; refresh rotation recommended.

export function authorizeUrl({ clientId, redirectUri, state, codeChallenge }) {
  const u = new URL("https://auth.example/authorize");
  u.searchParams.set("response_type", "code");
  u.searchParams.set("client_id", clientId);
  u.searchParams.set("redirect_uri", redirectUri);
  u.searchParams.set("scope", "openid profile");
  u.searchParams.set("state", state);
  u.searchParams.set("code_challenge", codeChallenge);
  u.searchParams.set("code_challenge_method", "S256");
  return u.toString();
}

Practice Notes

Checklist for OAuth 2.1
- Practice: rotate keys
- Scenario: developer API
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
