// OAuth mTLS — bind access tokens to client certificate (sketch)
export function tokenRequestWithMtls({ tokenUrl, clientCert, clientKey, code, redirectUri }) {
  // Use undici/fetch agent with cert+key; Authorization server returns
  // token bound to certificate thumbprint (cnf.x5t#S256).
  return {
    url: tokenUrl,
    method: "POST",
    body: new URLSearchParams({
      grant_type: "authorization_code",
      code,
      redirect_uri: redirectUri,
    }),
    mtls: { cert: clientCert, key: clientKey },
  };
}

Practice Notes

Checklist for OAuth mTLS
- Practice: session cookie
- Scenario: enterprise SSO
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
