// JWT Best Practices — Exercise 18
// Focus: csrf token | Scenario: chat platform

// JWT BCP (RFC 8725) — short-lived access, explicit aud/iss, reject "none"
import jwt from "jsonwebtoken";

export function signAccess(user, secret) {
  return jwt.sign(
    { sub: user.id, roles: user.roles },
    secret,
    { algorithm: "HS256", expiresIn: "15m", issuer: "phpkingdom-auth", audience: "phpkingdom-api" }
  );
}

export function verifyAccess(token, secret) {
  return jwt.verify(token, secret, {
    algorithms: ["HS256"], // never allow "none"
    issuer: "phpkingdom-auth",
    audience: "phpkingdom-api",
  });
}

Practice Notes

Checklist for JWT Best Practices
- Practice: csrf token
- Scenario: chat platform
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
