// Token Introspection — Example 13
// Focus: webauthn register | Scenario: REST microservice

// RFC 7662 token introspection
export async function introspect(token, { introspectionUrl, clientId, clientSecret }) {
  const res = await fetch(introspectionUrl, {
    method: "POST",
    headers: {
      "content-type": "application/x-www-form-urlencoded",
      authorization: "Basic " + Buffer.from(`${clientId}:${clientSecret}`).toString("base64"),
    },
    body: new URLSearchParams({ token }),
  });
  return res.json(); // { active, scope, sub, exp, ... }
}

Practice Notes

Checklist for Token Introspection
- Practice: webauthn register
- Scenario: REST microservice
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
