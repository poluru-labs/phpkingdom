// Login Rate Limiting — Exercise 17

const attempts = new Map();

export function rateLimitLogin(key, { limit = 5, windowMs = 15 * 60_000 } = {}) {
  const now = Date.now();
  const row = attempts.get(key) || { count: 0, reset: now + windowMs };
  if (now > row.reset) Object.assign(row, { count: 0, reset: now + windowMs });
  row.count += 1;
  attempts.set(key, row);
  if (row.count > limit) {
    const err = new Error("Too many attempts");
    err.status = 429;
    throw err;
  }
}

Practice Notes

Checklist for Login Rate Limiting
- Practice: rbac roles
- Scenario: developer API
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events
