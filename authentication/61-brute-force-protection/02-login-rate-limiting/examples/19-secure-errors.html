// Login Rate Limiting — Example 19
// Focus: secure errors | Scenario: fintech portal

const attempts = new Map();

export function rateLimitLogin(key, { limit = 5, windowMs = 15 * 60_000 } = {}) {
  const now = Date.now();
  const row = attempts.get(key) || { count: 0, reset: now + windowMs };
  if (now > row.reset) Object.assign(row, { count: 0, reset: now + windowMs });
  row.count += 1;
  attempts.set(key, row);
  if (row.count > limit) {
    const err = new Error("Too many attempts");
    err.status = 429;
    throw err;
  }
}
