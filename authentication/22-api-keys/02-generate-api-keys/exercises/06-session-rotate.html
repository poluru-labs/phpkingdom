// Generate API Keys — Exercise 06
// Focus: session rotate | Scenario: enterprise SSO

import crypto from "node:crypto";

export function hashApiKey(raw) {
  return crypto.createHash("sha256").update(raw).digest("hex");
}

export async function requireApiKey(req, store) {
  const raw = req.get("x-api-key");
  if (!raw) throw Object.assign(new Error("Unauthorized"), { status: 401 });
  const key = await store.findByHash(hashApiKey(raw));
  if (!key || key.revoked) throw Object.assign(new Error("Unauthorized"), { status: 401 });
  return key;
}

Checklist for Generate API Keys
- Practice: session rotate

- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events
