// SAML 2.0 SSO — validate assertion signature + conditions (sketch)
export function assertSamlConditions(assertion, { audience, now = Date.now() }) {
  if (assertion.audience !== audience) throw new Error("bad audience");
  if (now < assertion.notBefore || now > assertion.notOnOrAfter) throw new Error("expired");
  return { nameId: assertion.nameId, attributes: assertion.attributes };
}

Practice Notes

Checklist for SAML Metadata
- Practice: email verify
- Scenario: chat platform
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
