// OIDC: OAuth 2.0 + id_token (JWT) + UserInfo
export async function exchangeCodeForTokens({ tokenUrl, code, redirectUri, clientId, clientSecret }) {
  const res = await fetch(tokenUrl, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body: new URLSearchParams({
      grant_type: "authorization_code",
      code,
      redirect_uri: redirectUri,
      client_id: clientId,
      client_secret: clientSecret,
    }),
  });
  return res.json(); // access_token, id_token, refresh_token?
}

Practice Notes

Checklist for SAML vs OpenID Connect
- Practice: session cookie
- Scenario: enterprise SSO
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
