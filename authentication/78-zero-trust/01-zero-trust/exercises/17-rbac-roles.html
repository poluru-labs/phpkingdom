// Continuous verification: re-check device posture + step-up for sensitive actions
export function requireStepUp(session, action) {
  const risky = ["transfer", "change-email", "disable-mfa"].includes(action);
  if (risky && !session.mfaVerifiedAt) {
    const err = new Error("Step-up authentication required");
    err.status = 401;
    err.code = "step_up_required";
    throw err;
  }
}

Practice Notes

Checklist for Zero Trust Authentication
- Practice: rbac roles
- Scenario: developer API
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
