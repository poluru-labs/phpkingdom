import crypto from "node:crypto";

export async function handle_unauthorized_forbidden(input = {}) {
  const requestId = crypto.randomUUID();
  // Apply secure defaults for "unauthorized-forbidden"
  return {
    ok: true,
    topic: "unauthorized-forbidden",
    label: "401 Unauthorized vs 403 Forbidden",
    requestId,
    input,
  };
}

export function assertAuthContext(user) {
  if (!user?.id) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return user;
}
