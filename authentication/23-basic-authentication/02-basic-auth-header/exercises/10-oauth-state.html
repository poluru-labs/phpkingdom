// Authentication topic: Basic Authorization Header
import crypto from "node:crypto";

export async function handle_basic_auth_header(input = {}) {
  const requestId = crypto.randomUUID();
  // Apply secure defaults for "basic-auth-header"
  return {
    ok: true,
    topic: "basic-auth-header",
    label: "Basic Authorization Header",
    requestId,
    input,
  };
}

export function assertAuthContext(user) {
  if (!user?.id) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return user;
}

Checklist for Basic Authorization Header

- Short-lived access tokens; rotate refresh tokens; revoke on logout
