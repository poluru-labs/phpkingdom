// Authentication topic: Voice OTP Verification
import crypto from "node:crypto";

export async function handle_voice_otp(input = {}) {
  const requestId = crypto.randomUUID();
  // Apply secure defaults for "voice-otp"
  return {
    ok: true,
    topic: "voice-otp",
    label: "Voice OTP Verification",
    requestId,
    input,
  };
}

export function assertAuthContext(user) {
  if (!user?.id) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return user;
}

Checklist for Voice OTP Verification

- Short-lived access tokens; rotate refresh tokens; revoke on logout
