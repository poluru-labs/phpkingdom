import crypto from "node:crypto";

export async function handle_otp_expiration(input = {}) {
  const requestId = crypto.randomUUID();
  // Apply secure defaults for "otp-expiration"
  return {
    ok: true,
    topic: "otp-expiration",
    label: "OTP Expiration",
    requestId,
    input,
  };
}

export function assertAuthContext(user) {
  if (!user?.id) {
    const err = new Error("Unauthorized");
    err.status = 401;
    throw err;
  }
  return user;
}

Checklist for OTP Expiration
