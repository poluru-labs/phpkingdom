import jwt from "jsonwebtoken";
export function signAccess(user) {
  return jwt.sign({ sub: user.id, roles: user.roles }, process.env.JWT_SECRET, {
    expiresIn: "15m", issuer: "phpkingdom-auth", audience: "phpkingdom-api",
  });
}
export function requireAuth(req, res, next) {
  const [s, t] = (req.headers.authorization || "").split(" ");
  if (s !== "Bearer" || !t) return res.status(401).end();
  req.user = jwt.verify(t, process.env.JWT_SECRET, { audience: "phpkingdom-api", issuer: "phpkingdom-auth" });
  next();
}
