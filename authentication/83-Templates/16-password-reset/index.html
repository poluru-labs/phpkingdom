import crypto from "node:crypto";
import bcrypt from "bcrypt";
export function issueReset(userId, store) {
  const raw = crypto.randomBytes(32).toString("hex");
  store.save(sha256(raw), { userId, exp: Date.now() + 3600_000 });
  return raw; // email once
}
export async function resetPassword(store, raw, password) {
  const row = store.consume(sha256(raw));
  if (!row || row.exp < Date.now()) throw Object.assign(new Error("invalid"), { status: 400 });
  await store.update(row.userId, { passwordHash: await bcrypt.hash(password, 12) });
}
function sha256(s) { return crypto.createHash("sha256").update(s).digest("hex"); }
