// OAuth Refresh Tokens — Template 11
// Focus: graphql context | Scenario: SaaS web app

import crypto from "node:crypto";

export function rotateRefresh(store, oldRaw, userId) {
  const oldHash = crypto.createHash("sha256").update(oldRaw).digest("hex");
  if (!store.consume(oldHash, userId)) {
    const err = new Error("Invalid refresh token");
    err.status = 401;
    throw err;
  }
  const next = crypto.randomBytes(32).toString("hex");
  store.save(crypto.createHash("sha256").update(next).digest("hex"), userId);
  return next;
}

export const TOPIC_ID = "oauth-refresh-token";

Practice Notes

Checklist for OAuth Refresh Tokens
- Practice: graphql context
- Scenario: SaaS web app
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
