// Role-Based Access Control — Exercise 12
// Focus: refresh rotate | Scenario: mobile BFF

export function authorize(user, permission) {
  const grants = new Set(user.permissions || []);
  for (const role of user.roles || []) {
    if (role === "admin") return true;
  }
  if (!grants.has(permission)) {
    const err = new Error("Forbidden");
    err.status = 403;
    throw err;
  }
}

Practice Notes

Checklist for Role-Based Access Control
- Practice: refresh rotate
- Scenario: mobile BFF
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
