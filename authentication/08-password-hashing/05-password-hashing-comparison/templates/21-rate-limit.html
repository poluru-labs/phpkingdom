// Compare bcrypt / argon2 / scrypt / pbkdf2 for password storage
import bcrypt from "bcrypt";
import argon2 from "argon2";

export async function hashPassword(password, algo = "argon2") {
  if (algo === "bcrypt") return bcrypt.hash(password, 12);
  return argon2.hash(password, { type: argon2.argon2id });
}

export async function verifyPassword(password, hash, algo = "argon2") {
  if (algo === "bcrypt") return bcrypt.compare(password, hash);
  return argon2.verify(hash, password);
}
