// bcrypt — Exercise 11
// Focus: pkce flow | Scenario: SaaS web app

import bcrypt from "bcrypt";

export async function hashWithBcrypt(password) {
  return bcrypt.hash(password, 12);
}

export async function checkBcrypt(password, hash) {
  return bcrypt.compare(password, hash);
}

Practice Notes

Checklist for bcrypt

- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events
