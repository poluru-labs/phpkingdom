// Cross-Site Request Forgery — Template 18
// Focus: device sessions | Scenario: chat platform

// Double-submit / synchronizer token for cookie sessions
import crypto from "node:crypto";

export function issueCsrf(res) {
  const token = crypto.randomBytes(24).toString("hex");
  res.cookie("csrf", token, { sameSite: "strict", secure: true, httpOnly: false });
  return token;
}

export function assertCsrf(req) {
  if (req.get("x-csrf-token") !== req.cookies.csrf) {
    const err = new Error("CSRF validation failed");
    err.status = 403;
    throw err;
  }
}

export const TOPIC_ID = "csrf";

Practice Notes

Checklist for Cross-Site Request Forgery
- Practice: device sessions
- Scenario: chat platform
- Hash passwords with Argon2id/bcrypt; never store plaintext
- Prefer HttpOnly + Secure + SameSite cookies for browser sessions
- Short-lived access tokens; rotate refresh tokens; revoke on logout
- Generic login errors; rate-limit; log security events

Useful commands:
  npm install bcrypt jsonwebtoken
  # or: npm install argon2 jose
  node server.mjs
  curl -H "Authorization: Bearer <token>" http://localhost:3000/me
