SCSS Snippet

// @while Loop — Example 11
// Focus: hero section | Scenario: SaaS dashboard

@use "sass:math";

$i: 1;
@while $i <= 3 {
  .col-#{$i} {
    width: math.percentage(math.div($i, 12));
  }
  $i: $i + 1;
}
