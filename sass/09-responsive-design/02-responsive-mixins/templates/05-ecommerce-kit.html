SCSS Snippet

// Responsive Mixins — Template 05
// Focus: ecommerce kit | Scenario: portfolio

@use "sass:map";
$bp: (sm: 576px, md: 768px, lg: 992px);

@mixin respond-to($name) {
  @media (min-width: map.get($bp, $name)) {
    @content;
  }
}

.grid {
  display: grid;
  gap: 1rem;
  grid-template-columns: 1fr;
  @include respond-to(md) { grid-template-columns: repeat(3, 1fr); }
}
