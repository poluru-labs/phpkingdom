// Fluid Typography — Exercise 15
// Focus: performance pass | Scenario: portfolio

@use "sass:math";

@function fluid-type($min, $max, $min-vw: 320, $max-vw: 1200) {
  $slope: math.div($max - $min, $max-vw - $min-vw);
  $intercept: $min - $slope * $min-vw;
  @return clamp(#{$min}px, #{$intercept}px + #{$slope * 100}vw, #{$max}px);
}

h1 { font-size: fluid-type(28, 48); }
p  { font-size: fluid-type(16, 18); }
