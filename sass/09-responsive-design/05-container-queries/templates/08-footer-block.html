SCSS Snippet

// Container Queries with Sass — Template 08
// Focus: footer block | Scenario: analytics page

// Pair Sass tokens with native container queries
.card-grid {
  container-type: inline-size;
  container-name: cards;
}

@mixin container-up($name, $min) {
  @container #{$name} (min-width: #{$min}) {
    @content;
  }
}

.card {
  padding: 1rem;
  @include container-up(cards, 40rem) {
    display: grid;
    grid-template-columns: 1fr 1fr;
  }
}
