SCSS Snippet

// Sass Breakpoints — Template 05
// Focus: ecommerce kit | Scenario: portfolio

@use "sass:map";

$breakpoints: (
  sm: 576px,
  md: 768px,
  lg: 992px,
  xl: 1200px,
);

@function bp($name) {
  @return map.get($breakpoints, $name);
}

@media (min-width: bp(lg)) {
  .container { max-width: 960px; }
}
