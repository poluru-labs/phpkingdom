DynamoDB (Node.js SDK v3)

// Encryption at Rest — Exercise 11
// Focus: query gsi | Scenario: user profile API

import { UpdateTableCommand, DynamoDBClient } from "@aws-sdk/client-dynamodb";
const client = new DynamoDBClient({ region: "us-east-1" });
export async function enableKms(TableName, kmsKeyArn) {
  await client.send(new UpdateTableCommand({
    TableName,
    SSESpecification: { Enabled: true, SSEType: "KMS", KMSMasterKeyId: kmsKeyArn },
  }));
}

Checklist for Encryption at Rest
- Practice: query gsi

- Use ConditionExpression for idempotency / optimistic locking
