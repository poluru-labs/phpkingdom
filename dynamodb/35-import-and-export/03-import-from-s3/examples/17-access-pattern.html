DynamoDB (Node.js SDK v3)

// Import DynamoDB from S3 — Example 17
// Focus: access pattern | Scenario: multi-tenant SaaS

import { ImportTableCommand, DynamoDBClient } from "@aws-sdk/client-dynamodb";
const client = new DynamoDBClient({ region: "us-east-1" });
export async function importFromS3(bucket, prefix) {
  await client.send(new ImportTableCommand({
    S3BucketSource: { S3Bucket: bucket, S3KeyPrefix: prefix },
    InputFormat: "DYNAMODB_JSON",
    TableCreationParameters: {
      TableName: "ImportedTable",
      AttributeDefinitions: [
        { AttributeName: "pk", AttributeType: "S" },
        { AttributeName: "sk", AttributeType: "S" },
      ],
      KeySchema: [
        { AttributeName: "pk", KeyType: "HASH" },
        { AttributeName: "sk", KeyType: "RANGE" },
      ],
      BillingMode: "PAY_PER_REQUEST",
    },
  }));
}
