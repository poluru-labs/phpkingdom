DynamoDB (Node.js SDK v3)

// Order Service — Example 01
// Focus: create client | Scenario: user profile API

import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import {
  DynamoDBDocumentClient,
  PutCommand,
  GetCommand,
  UpdateCommand,
  QueryCommand,
} from "@aws-sdk/lib-dynamodb";

const raw = new DynamoDBClient({
  region: process.env.AWS_REGION || "us-east-1",
  ...(process.env.DDB_ENDPOINT
    ? { endpoint: process.env.DDB_ENDPOINT, credentials: { accessKeyId: "local", secretAccessKey: "local" } }
    : {}),
});
const doc = DynamoDBDocumentClient.from(raw);
const TableName = process.env.TABLE_NAME || "AppTable";

export async function placeOrder(userId, orderId, lines) {
  const total = lines.reduce((s, l) => s + l.price * l.qty, 0);
  await doc.send(new PutCommand({
    TableName,
    Item: {
      pk: `USER#${userId}`,
      sk: `ORDER#${orderId}`,
      entityType: "Order",
      lines,
      total,
      status: "NEW",
    },
  }));
}

Practice Notes

Checklist for Order Service
- Practice: create client
- Scenario: user profile API
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey
