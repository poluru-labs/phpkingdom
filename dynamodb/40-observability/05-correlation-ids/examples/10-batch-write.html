DynamoDB (Node.js SDK v3)

// Correlation IDs — Example 10
// Focus: batch write | Scenario: IoT telemetry

export function ensureCorrelationId(event) {
  return (
    event.headers?.["x-correlation-id"] ||
    event.requestContext?.requestId ||
    crypto.randomUUID()
  );
}
