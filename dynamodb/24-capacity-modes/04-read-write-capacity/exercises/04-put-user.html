// Read and Write Capacity Units — Exercise 04
// Focus: put user | Scenario: URL shortener

// 1 RCU ≈ 1 strongly consistent read of 4KB / sec (2 eventual)
// 1 WCU ≈ 1 write of 1KB / sec
export function estimateRcu(itemKb, strong, readsPerSec) {
  const units = Math.ceil(itemKb / 4) * readsPerSec;
  return strong ? units : Math.ceil(units / 2);
}
