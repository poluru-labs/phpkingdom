DynamoDB (Node.js SDK v3)

// Many-to-Many Relationships — Exercise 17
// Focus: entity prefix | Scenario: multi-tenant SaaS

import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import {
  DynamoDBDocumentClient,
  PutCommand,
  GetCommand,
  UpdateCommand,
  DeleteCommand,
  QueryCommand,
  ScanCommand,
  BatchGetCommand,
  BatchWriteCommand,
  TransactWriteCommand,
} from "@aws-sdk/lib-dynamodb";

const raw = new DynamoDBClient({
  region: process.env.AWS_REGION || "us-east-1",
  ...(process.env.DDB_ENDPOINT
    ? { endpoint: process.env.DDB_ENDPOINT, credentials: { accessKeyId: "local", secretAccessKey: "local" } }
    : {}),
});
const doc = DynamoDBDocumentClient.from(raw);
const TableName = process.env.TABLE_NAME || "AppTable";

export async function linkManyToMany(a, b, rel = "FOLLOWS") {
  await doc.send(new TransactWriteCommand({
    TransactItems: [
      { Put: { TableName, Item: { pk: a, sk: `${rel}#${b}`, entityType: "Edge" } } },
      { Put: { TableName, Item: { pk: b, sk: `FOLLOWED_BY#${a}`, entityType: "Edge" } } },
    ],
  }));
}

Practice Notes

Checklist for Many-to-Many Relationships
- Practice: entity prefix
- Scenario: multi-tenant SaaS
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey

Useful commands:
  npm install @aws-sdk/client-dynamodb @aws-sdk/lib-dynamodb
  docker run -p 8000:8000 amazon/dynamodb-local
  aws dynamodb list-tables --endpoint-url http://localhost:8000
  node index.mjs

Notes

This exercise shows one way to apply Many-to-Many Relationships in a multi-tenant saas context.

Model for queries, keep partition keys well distributed, and use GSIs only when access patterns require them.
