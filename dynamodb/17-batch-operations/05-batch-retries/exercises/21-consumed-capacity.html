export async function withBackoff(fn, attempts = 5) {
  let delay = 50;
  for (let i = 0; i < attempts; i++) {
    try {
      return await fn();
    } catch (err) {
      if (i === attempts - 1) throw err;
      await new Promise((r) => setTimeout(r, delay));
      delay *= 2;
    }
  }
}

Practice Notes

Checklist for Batch Retry Strategies
- Practice: consumed capacity
- Scenario: user profile API
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey
