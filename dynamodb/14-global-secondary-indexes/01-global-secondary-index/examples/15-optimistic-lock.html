DynamoDB (Node.js SDK v3)

// Global Secondary Index — Example 15
// Focus: optimistic lock | Scenario: session store

import { UpdateTableCommand, DynamoDBClient } from "@aws-sdk/client-dynamodb";
const client = new DynamoDBClient({ region: "us-east-1" });
export async function addEmailGsi() {
  await client.send(new UpdateTableCommand({
    TableName: "AppTable",
    AttributeDefinitions: [{ AttributeName: "email", AttributeType: "S" }],
    GlobalSecondaryIndexUpdates: [{
      Create: {
        IndexName: "EmailIndex",
        KeySchema: [{ AttributeName: "email", KeyType: "HASH" }],
        Projection: { ProjectionType: "ALL" },
      },
    }],
  }));
}

Practice Notes

Checklist for Global Secondary Index
- Practice: optimistic lock
- Scenario: session store
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey
