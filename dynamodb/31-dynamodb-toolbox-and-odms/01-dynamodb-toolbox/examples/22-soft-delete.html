DynamoDB (Node.js SDK v3)

// DynamoDB Toolbox — Example 22
// Focus: soft delete | Scenario: e-commerce orders

import { Entity, Table } from "dynamodb-toolbox";
import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient } from "@aws-sdk/lib-dynamodb";
const DocumentClient = DynamoDBDocumentClient.from(new DynamoDBClient({ region: "us-east-1" }));
export const MyTable = new Table({ name: "AppTable", partitionKey: "pk", sortKey: "sk", DocumentClient });
export const User = new Entity({
  name: "User",
  attributes: { pk: { partitionKey: true }, sk: { sortKey: true }, email: { type: "string" } },
  table: MyTable,
});

Practice Notes

Checklist for DynamoDB Toolbox
- Practice: soft delete
- Scenario: e-commerce orders
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey

Useful commands:
  npm install @aws-sdk/client-dynamodb @aws-sdk/lib-dynamodb
  docker run -p 8000:8000 amazon/dynamodb-local
  aws dynamodb list-tables --endpoint-url http://localhost:8000
  node index.mjs
