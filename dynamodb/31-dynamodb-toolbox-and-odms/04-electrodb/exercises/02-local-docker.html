// ElectroDB — Exercise 02
// Focus: local docker | Scenario: e-commerce orders

import { Entity } from "electrodb";
import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
export const UserEntity = new Entity({
  model: { entity: "user", version: "1", service: "app" },
  attributes: {
    userId: { type: "string", required: true },
    email: { type: "string", required: true },
  },
  indexes: {
    primary: { pk: { field: "pk", composite: ["userId"] }, sk: { field: "sk", composite: [] } },
  },
}, { client: new DynamoDBClient({}), table: "AppTable" });

Practice Notes

Checklist for ElectroDB
- Practice: local docker
- Scenario: e-commerce orders
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey

Useful commands:
  npm install @aws-sdk/client-dynamodb @aws-sdk/lib-dynamodb
  docker run -p 8000:8000 amazon/dynamodb-local
  aws dynamodb list-tables --endpoint-url http://localhost:8000
  node index.mjs

Notes

This exercise shows one way to apply ElectroDB in a e-commerce orders context.

Model for queries, keep partition keys well distributed, and use GSIs only when access patterns require them.
