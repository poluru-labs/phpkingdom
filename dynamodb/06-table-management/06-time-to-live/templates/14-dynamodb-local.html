// Time to Live (TTL) — Template 14
// Focus: dynamodb local | Scenario: URL shortener

import { UpdateTimeToLiveCommand, DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient, PutCommand } from "@aws-sdk/lib-dynamodb";
const client = new DynamoDBClient({ region: "us-east-1" });
const doc = DynamoDBDocumentClient.from(client);
// TTL attribute must be epoch seconds (Number)
export async function enableTtl(TableName = "AppTable") {
  await client.send(new UpdateTimeToLiveCommand({
    TableName,
    TimeToLiveSpecification: { AttributeName: "expiresAt", Enabled: true },
  }));
}
export async function putEphemeral(id, ttlSeconds = 3600) {
  const expiresAt = Math.floor(Date.now() / 1000) + ttlSeconds;
  await doc.send(new PutCommand({
    TableName: "AppTable",
    Item: { pk: `SESSION#${id}`, sk: "META", expiresAt },
  }));
}

Practice Notes

Checklist for Time to Live (TTL)
- Practice: dynamodb local
- Scenario: URL shortener
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey
