DynamoDB (Node.js SDK v3)

// Describe a Table — Template 04
// Focus: gsi email | Scenario: URL shortener

import { DescribeTableCommand, DynamoDBClient } from "@aws-sdk/client-dynamodb";
const client = new DynamoDBClient({ region: "us-east-1" });
export async function describe(TableName = "AppTable") {
  const { Table } = await client.send(new DescribeTableCommand({ TableName }));
  return { status: Table.TableStatus, keys: Table.KeySchema, gsis: Table.GlobalSecondaryIndexes };
}
