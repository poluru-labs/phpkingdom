import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient, UpdateCommand, QueryCommand } from "@aws-sdk/lib-dynamodb";

const doc = DynamoDBDocumentClient.from(new DynamoDBClient({ region: "us-east-1" }));
const TableName = process.env.TABLE_NAME || "AppTable";

export async function incrementSharded(counterId, shardCount, delta = 1) {
  const shard = Math.floor(Math.random() * shardCount);
  await doc.send(new UpdateCommand({
    TableName,
    Key: { pk: `COUNTER#${counterId}`, sk: `SHARD#${shard}` },
    UpdateExpression: "ADD #c :d",
    ExpressionAttributeNames: { "#c": "count" },
    ExpressionAttributeValues: { ":d": delta },
  }));
}

export async function sumShards(counterId) {
  const { Items } = await doc.send(new QueryCommand({
    TableName,
    KeyConditionExpression: "pk = :pk AND begins_with(sk, :sk)",
    ExpressionAttributeValues: { ":pk": `COUNTER#${counterId}`, ":sk": "SHARD#" },
  }));
  return (Items || []).reduce((n, i) => n + (i.count || 0), 0);
}
