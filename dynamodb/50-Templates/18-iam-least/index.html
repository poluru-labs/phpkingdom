LeadingKeys

Fine-grained IAM LeadingKeys condition

SDK Scaffold

export const userRowPolicy = {
  Version: "2012-10-17",
  Statement: [
    {
      Effect: "Allow",
      Action: ["dynamodb:GetItem", "dynamodb:PutItem", "dynamodb:UpdateItem", "dynamodb:Query"],
      Resource: ["arn:aws:dynamodb:*:*:table/AppTable"],
      Condition: {
        "ForAllValues:StringLike": {
          "dynamodb:LeadingKeys": ["USER#${cognito-identity.amazonaws.com:sub}"],
        },
      },
    },
  ],
};
