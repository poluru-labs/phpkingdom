SDK Scaffold

import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient, PutCommand } from "@aws-sdk/lib-dynamodb";

// Point the client at the nearest replica region
export function regionalDoc(region = process.env.AWS_REGION || "us-east-1") {
  return DynamoDBDocumentClient.from(new DynamoDBClient({ region }));
}

export async function putInRegion(doc, Item) {
  await doc.send(new PutCommand({
    TableName: process.env.TABLE_NAME || "AppTable",
    Item: { ...Item, updatedAt: new Date().toISOString() },
  }));
}

export const replicas = ["us-east-1", "eu-west-1"];
