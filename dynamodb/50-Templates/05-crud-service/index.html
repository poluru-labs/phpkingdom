import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import {
  DynamoDBDocumentClient,
  PutCommand,
  GetCommand,
  UpdateCommand,
  DeleteCommand,
} from "@aws-sdk/lib-dynamodb";

const doc = DynamoDBDocumentClient.from(new DynamoDBClient({ region: "us-east-1" }));
const TableName = process.env.TABLE_NAME || "AppTable";

export const usersRepo = {
  create: (Item) => doc.send(new PutCommand({ TableName, Item, ConditionExpression: "attribute_not_exists(pk)" })),
  read: async (Key) => (await doc.send(new GetCommand({ TableName, Key }))).Item,
  updateEmail: (Key, email) =>
    doc.send(new UpdateCommand({
      TableName,
      Key,
      UpdateExpression: "SET email = :e",
      ExpressionAttributeValues: { ":e": email },
    })),
  remove: (Key) => doc.send(new DeleteCommand({ TableName, Key })),
};
