import { mockClient } from "aws-sdk-client-mock";
import { DynamoDBDocumentClient, GetCommand, PutCommand } from "@aws-sdk/lib-dynamodb";

const ddbMock = mockClient(DynamoDBDocumentClient);

beforeEach(() => ddbMock.reset());

test("get profile", async () => {
  ddbMock.on(GetCommand).resolves({ Item: { pk: "USER#1", sk: "PROFILE", email: "a@b.com" } });
  const res = await DynamoDBDocumentClient.prototype.send.call(
    { send: ddbMock },
    new GetCommand({ TableName: "AppTable", Key: { pk: "USER#1", sk: "PROFILE" } })
  );
  expect(res.Item.email).toBe("a@b.com");
});

test("put profile", async () => {
  ddbMock.on(PutCommand).resolves({});
  await expect(
    DynamoDBDocumentClient.prototype.send.call(
      { send: ddbMock },
      new PutCommand({ TableName: "AppTable", Item: { pk: "USER#2", sk: "PROFILE" } })
    )
  ).resolves.toBeDefined();
});
