export async function processStreamEvents(event) {
  for (const record of event.Records || []) {
    if (record.eventName === "INSERT") {
      console.log("created", record.dynamodb.Keys);
    } else if (record.eventName === "MODIFY") {
      console.log("updated", record.dynamodb.Keys);
    } else if (record.eventName === "REMOVE") {
      console.log("removed", record.dynamodb.Keys);
    }
  }
}
