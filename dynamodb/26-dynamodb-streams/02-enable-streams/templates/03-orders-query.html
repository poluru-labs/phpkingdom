DynamoDB (Node.js SDK v3)

// Enable DynamoDB Streams — Template 03
// Focus: orders query | Scenario: chat messages

import { UpdateTableCommand, DynamoDBClient } from "@aws-sdk/client-dynamodb";
const client = new DynamoDBClient({ region: "us-east-1" });
export async function enableStreams(TableName = "AppTable") {
  await client.send(new UpdateTableCommand({
    TableName,
    StreamSpecification: { StreamEnabled: true, StreamViewType: "NEW_AND_OLD_IMAGES" },
  }));
}
