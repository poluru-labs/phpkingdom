DynamoDB (Node.js SDK v3)

// Build a CRUD API — Example 06
// Focus: query pk sk | Scenario: inventory ledger

import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import {
  DynamoDBDocumentClient,
  PutCommand,
  GetCommand,
  UpdateCommand,
  DeleteCommand,
  QueryCommand,
  ScanCommand,
  BatchGetCommand,
  BatchWriteCommand,
  TransactWriteCommand,
} from "@aws-sdk/lib-dynamodb";

const raw = new DynamoDBClient({
  region: process.env.AWS_REGION || "us-east-1",
  ...(process.env.DDB_ENDPOINT
    ? { endpoint: process.env.DDB_ENDPOINT, credentials: { accessKeyId: "local", secretAccessKey: "local" } }
    : {}),
});
const doc = DynamoDBDocumentClient.from(raw);
const TableName = process.env.TABLE_NAME || "AppTable";

export const crud = {
  async create(Item) {
    await doc.send(new PutCommand({ TableName, Item, ConditionExpression: "attribute_not_exists(pk)" }));
  },
  async read(Key) {
    return (await doc.send(new GetCommand({ TableName, Key }))).Item;
  },
  async update(Key, UpdateExpression, ExpressionAttributeValues) {
    await doc.send(new UpdateCommand({ TableName, Key, UpdateExpression, ExpressionAttributeValues }));
  },
  async remove(Key) {
    await doc.send(new DeleteCommand({ TableName, Key }));
  },
};

Practice Notes

Checklist for Build a CRUD API
- Practice: query pk sk
- Scenario: inventory ledger
- Design for access patterns first (Query over Scan)
- Prefer Document Client for app code; raw client for admin
- Use ConditionExpression for idempotency / optimistic locking
- Paginate with ExclusiveStartKey / LastEvaluatedKey
