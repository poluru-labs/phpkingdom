Express

// Express Application — Template 07
// Focus: mongoose app | Scenario: realtime chat

const express = require("express");
const app = express();

// Application settings
app.set("trust proxy", 1);
app.disable("x-powered-by");

app.locals.site = "phpkingdom.com";

app.get("/", (req, res) => {
  res.json({ site: app.locals.site, env: app.get("env") });
});

module.exports = app;
// Separate app.js from listen() for easier testing
