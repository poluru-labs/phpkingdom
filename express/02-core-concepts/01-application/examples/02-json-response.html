Express

// Express Application — Example 02
// Focus: json response | Scenario: admin backend

const express = require("express");
const app = express();

// Application settings
app.set("trust proxy", 1);
app.disable("x-powered-by");

app.locals.site = "phpkingdom.com";

app.get("/", (req, res) => {
  res.json({ site: app.locals.site, env: app.get("env") });
});

module.exports = app;
// Separate app.js from listen() for easier testing

Practice Notes

Checklist for Express Application
