Express

// Express Application — Exercise 05
// Focus: param id | Scenario: blog API

const express = require("express");
const app = express();

// Application settings
app.set("trust proxy", 1);
app.disable("x-powered-by");

app.locals.site = "phpkingdom.com";

app.get("/", (req, res) => {
  res.json({ site: app.locals.site, env: app.get("env") });
});

module.exports = app;
// Separate app.js from listen() for easier testing
