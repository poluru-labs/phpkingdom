// Request and Response — Example 25
// Focus: best practice recap | Scenario: blog API

app.get("/demo/:id", (req, res) => {
  console.log("method", req.method);
  console.log("params", req.params);
  console.log("query", req.query);
  console.log("headers", req.headers["user-agent"]);

  res.status(200)
    .set("X-Site", "phpkingdom.com")
    .json({ id: req.params.id, ok: true });
});
// req = incoming; res = outgoing helpers (json, status, send, redirect)
