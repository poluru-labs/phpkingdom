Compression Middleware

Section: Advanced

Focus: Mongoose Model

Scenario: SSR site

Study the Express route/middleware pattern and adapt it to your API.

Express

// Compression Middleware — Example 18
// Focus: mongoose model | Scenario: SSR site

const compression = require("compression");

app.use(compression({
  threshold: 1024,
  filter: (req, res) => {
    if (req.headers["x-no-compress"]) return false;
    return compression.filter(req, res);
  },
}));

app.get("/big.json", (req, res) => {
  res.json({ items: Array.from({ length: 1000 }, (_, i) => ({ i })) });
});
// Place compression early; skip already-compressed assets if needed
