Express

// WebSockets — Example 24
// Focus: morgan log | Scenario: file service

const http = require("http");
const express = require("express");
const { Server } = require("socket.io");

const app = express();
const server = http.createServer(app);
const io = new Server(server, { cors: { origin: "*" } });

io.on("connection", (socket) => {
  socket.emit("welcome", { site: "phpkingdom.com" });
  socket.on("chat", (msg) => io.emit("chat", { from: socket.id, msg }));
});

app.get("/health", (req, res) => res.json({ ok: true }));
server.listen(3000);
// Share the HTTP server between Express and Socket.IO
