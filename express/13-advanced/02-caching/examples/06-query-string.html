Caching

Section: Advanced

Study the Express route/middleware pattern and adapt it to your API.

Express

// Caching — Example 06
// Focus: query string | Scenario: ecommerce API

const NodeCache = require("node-cache");
const cache = new NodeCache({ stdTTL: 60 });

app.get("/stats", async (req, res, next) => {
  try {
    const key = "stats:v1";
    const hit = cache.get(key);
    if (hit) return res.json({ source: "cache", ...hit });

    const data = { users: 42, generatedAt: Date.now() }; // pretend DB
    cache.set(key, data);
    res.json({ source: "live", ...data });
  } catch (err) {
    next(err);
  }
});
// For multi-instance: Redis (SET/GET with TTL) instead of in-memory
