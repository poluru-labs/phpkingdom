Express

// Express with TypeScript — Exercise 09
// Focus: async handler | Scenario: microservice

// tsconfig: "esModuleInterop": true, "strict": true
import express, { Request, Response, NextFunction } from "express";

const app = express();
app.use(express.json());

app.get("/health", (_req: Request, res: Response) => {
  res.json({ ok: true });
});

app.post("/users", (req: Request, res: Response) => {
  const { email } = req.body as { email?: string };
  res.status(201).json({ email });
});

app.use((err: Error, _req: Request, res: Response, _next: NextFunction) => {
  res.status(500).json({ error: err.message });
});

app.listen(3000);
// Compile with tsc or run via tsx / ts-node

Practice Notes

Checklist for Express with TypeScript
- Practice: async handler
- Scenario: microservice
- Keep route handlers thin; push logic into controllers/services
- Always attach final 4-arg error middleware
- Validate input; load secrets from environment variables
- Export app separately from listen() for easier testing
