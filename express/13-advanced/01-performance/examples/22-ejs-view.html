Express

// Express Performance — Example 22
// Focus: ejs view | Scenario: admin backend

const compression = require("compression");
app.use(compression());

// Avoid sync fs in request path; reuse DB pools; cache hot reads
app.get("/report", async (req, res, next) => {
  try {
    // stream large responses when possible
    res.json({ cached: true, generatedAt: Date.now() });
  } catch (err) {
    next(err);
  }
});

// CLUSTER tip: run multiple workers behind a process manager (PM2)
// Keep middleware lean — every app.use runs on every request
