// Express = minimal HTTP framework on Node.js
// Install: npm install express

const express = require("express");
const app = express();

app.get("/", (req, res) => {
  res.send("Welcome to Express — phpkingdom.com");
});

app.listen(3000, () => console.log("http://localhost:3000"));
