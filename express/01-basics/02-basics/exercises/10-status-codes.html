Express

// Express.js - Basics — Exercise 10
// Focus: status codes | Scenario: public docs API

const express = require("express");
const app = express();

app.use(express.json());

app.get("/ping", (req, res) => res.json({ pong: true }));
app.post("/echo", (req, res) => res.json({ youSent: req.body }));

app.listen(3000);
// Basics: create app → middleware → routes → listen
