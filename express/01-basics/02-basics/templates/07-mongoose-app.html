Express

// Express.js - Basics — Template 07
// Focus: mongoose app | Scenario: realtime chat

const express = require("express");
const app = express();

app.use(express.json());

app.get("/ping", (req, res) => res.json({ pong: true }));
app.post("/echo", (req, res) => res.json({ youSent: req.body }));

app.listen(3000);
// Basics: create app → middleware → routes → listen
