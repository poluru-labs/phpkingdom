Express

// First Express Server — Example 13
// Focus: body parser | Scenario: auth service

const express = require("express");
const app = express();

app.get("/", (req, res) => {
  res.type("html").send("<h1>Hello from Express</h1>");
});

app.listen(3000, () => {
  console.log("First server listening on 3000");
});

Practice Notes

Checklist for First Express Server
