// Export app without listening for tests
// app.js → module.exports = app;
// server.js → app.listen(PORT)

const request = require("supertest");
const app = require("../app");

describe("API", () => {
  it("GET /health", async () => {
    const res = await request(app).get("/health");
    expect(res.status).toBe(200);
    expect(res.body.ok).toBe(true);
  });
});

Practice Notes

Checklist for Express Testing
- Practice: static site
- Scenario: blog API
- Keep route handlers thin; push logic into controllers/services
- Always attach final 4-arg error middleware
- Validate input; load secrets from environment variables
- Export app separately from listen() for easier testing
