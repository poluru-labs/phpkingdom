// Integration Testing — Exercise 21
// Focus: static mount | Scenario: REST API

const request = require("supertest");
const app = require("../app");

describe("users CRUD", () => {
  it("creates and lists users", async () => {
    const created = await request(app)
      .post("/users")
      .send({ name: "Ada", email: "ada@example.com" })
      .expect(201);
    expect(created.body.email).toBe("ada@example.com");

    const list = await request(app).get("/users").expect(200);
    expect(list.body.some((u) => u.email === "ada@example.com")).toBe(true);
  });
});

Practice Notes

Checklist for Integration Testing
- Practice: static mount
- Scenario: REST API
- Keep route handlers thin; push logic into controllers/services
- Always attach final 4-arg error middleware
- Validate input; load secrets from environment variables
- Export app separately from listen() for easier testing
