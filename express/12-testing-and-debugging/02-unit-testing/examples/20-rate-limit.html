// Unit Testing — Example 20
// Focus: rate limit | Scenario: public docs API

// services/math.js
exports.add = (a, b) => a + b;

// services/math.test.js (node:test or jest)
const { test } = require("node:test");
const assert = require("node:assert/strict");
const { add } = require("./math");

test("add", () => {
  assert.equal(add(2, 3), 5);
});
// Unit-test pure services without spinning up Express

Practice Notes

Checklist for Unit Testing
- Practice: rate limit
- Scenario: public docs API
- Keep route handlers thin; push logic into controllers/services
- Always attach final 4-arg error middleware
- Validate input; load secrets from environment variables
- Export app separately from listen() for easier testing
