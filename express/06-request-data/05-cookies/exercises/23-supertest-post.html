// Cookies — Exercise 23
// Focus: supertest post | Scenario: auth service

const cookieParser = require("cookie-parser");
app.use(cookieParser(process.env.COOKIE_SECRET));

app.get("/login", (req, res) => {
  res.cookie("sid", "abc123", {
    httpOnly: true,
    sameSite: "lax",
    secure: process.env.NODE_ENV === "production",
    maxAge: 60 * 60 * 1000,
  });
  res.json({ ok: true });
});

app.get("/me", (req, res) => {
  res.json({ sid: req.cookies.sid, signed: req.signedCookies });
});
