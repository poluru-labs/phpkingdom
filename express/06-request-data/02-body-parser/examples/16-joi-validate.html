Express

// Body Parser — Example 16
// Focus: joi validate | Scenario: ecommerce API

const express = require("express");
const app = express();

// Built-in (Express 4.16+)
app.use(express.json({ limit: "100kb" }));
app.use(express.urlencoded({ extended: true, limit: "100kb" }));
app.use(express.text({ type: "text/*" }));
app.use(express.raw({ type: "application/octet-stream", limit: "2mb" }));

app.post("/hooks", (req, res) => {
  res.json({ typeofBody: typeof req.body, body: req.body });
});
// Prefer built-ins over the legacy body-parser package
