// Form Data — Exercise 21
// Focus: static mount | Scenario: REST API

// URL-encoded HTML forms
app.use(express.urlencoded({ extended: true }));

app.post("/contact", (req, res) => {
  const { name, message } = req.body;
  res.send(`Thanks ${name}: ${message}`);
});

// Multipart (files + fields) needs multer — see file-upload / multer-uploads
