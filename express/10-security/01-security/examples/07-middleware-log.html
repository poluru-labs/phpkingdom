Express

// Express Security — Example 07
// Focus: middleware log | Scenario: realtime chat

const helmet = require("helmet");
const cors = require("cors");
const rateLimit = require("express-rate-limit");

app.disable("x-powered-by");
app.use(helmet());
app.use(cors({ origin: process.env.CORS_ORIGIN, credentials: true }));
app.use("/api", rateLimit({ windowMs: 15 * 60 * 1000, max: 100 }));

app.use(express.json({ limit: "100kb" }));
// Stack: headers + CORS + throttling + small bodies + validated input

Practice Notes

Checklist for Express Security
- Practice: middleware log

- Keep route handlers thin; push logic into controllers/services
- Always attach final 4-arg error middleware
- Validate input; load secrets from environment variables
