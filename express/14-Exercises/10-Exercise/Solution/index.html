Solution 10 — Custom Error

Throw/create an error with status 404 for missing resources.

Solution

function notFound(message = 'Not found') {
  const err = new Error(message);
  err.status = 404;
  return err;
}

app.get('/api/items/:id', (req, res, next) => {
  next(notFound('Item not found'));
});
