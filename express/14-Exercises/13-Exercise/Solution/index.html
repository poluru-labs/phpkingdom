Validate body email is an email.

Solution

app.post(
  '/register',
  body('email').isEmail(),
  (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) return res.status(400).json({ errors: errors.array() });
    res.status(201).json({ ok: true });
  }
);
