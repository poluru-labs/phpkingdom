Solution 02 — Hello Server

Create an Express app that listens on 3000 and returns Hello.

Solution

const express = require('express');
const app = express();

app.get('/', (req, res) => {
  res.send('Hello phpkingdom.com');
});

app.listen(3000, () => console.log('http://localhost:3000'));
