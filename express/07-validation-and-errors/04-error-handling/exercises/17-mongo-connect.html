// Error Handling — Exercise 17
// Focus: mongo connect | Scenario: realtime chat

class AppError extends Error {
  constructor(message, status = 500) {
    super(message);
    this.status = status;
  }
}

app.get("/items/:id", (req, res, next) => {
  try {
    throw new AppError("Item missing", 404);
  } catch (err) {
    next(err);
  }
});

app.use((err, req, res, next) => {
  res.status(err.status || 500).json({ error: err.message });
});
// Always forward errors with next(err); never leave them unhandled
