// Express Validation — Exercise 19
// Focus: cors setup | Scenario: microservice

function validateUser(req, res, next) {
  const errors = [];
  if (!req.body?.email?.includes("@")) errors.push("email");
  if (!req.body?.name || req.body.name.length < 2) errors.push("name");
  if (errors.length) {
    return res.status(400).json({ error: "validation failed", fields: errors });
  }
  next();
}

app.post("/users", validateUser, (req, res) => {
  res.status(201).json(req.body);
});
// Manual checks work; prefer a library for complex rules
