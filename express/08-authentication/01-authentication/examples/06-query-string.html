// Express Authentication — Example 06
// Focus: query string | Scenario: ecommerce API

function auth(req, res, next) {
  const header = req.headers.authorization || "";
  if (!header.startsWith("Bearer ")) {
    return res.status(401).json({ error: "Unauthorized" });
  }
  // Replace with real token verify (JWT / session lookup)
  req.user = { id: 1, role: "user" };
  next();
}

app.post("/login", (req, res) => {
  res.json({ token: "demo-token", user: { id: 1 } });
});
app.get("/profile", auth, (req, res) => res.json(req.user));
// Auth = prove identity; authorize = check permissions
