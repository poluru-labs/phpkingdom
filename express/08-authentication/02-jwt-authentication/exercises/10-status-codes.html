// JWT Authentication — Exercise 10
// Focus: status codes | Scenario: public docs API

const jwt = require("jsonwebtoken");
const secret = process.env.JWT_SECRET || "dev-secret";

app.post("/auth/login", (req, res) => {
  const { email, password } = req.body || {};
  if (password !== "secret") return res.status(401).json({ error: "bad credentials" });
  const token = jwt.sign({ sub: 1, email }, secret, { expiresIn: "1h" });
  res.json({ accessToken: token });
});

function requireJwt(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) return res.status(401).json({ error: "missing token" });
  try {
    req.user = jwt.verify(token, secret);
    next();
  } catch {
    res.status(401).json({ error: "invalid token" });
  }
}

app.get("/auth/me", requireJwt, (req, res) => res.json(req.user));
