// API Response Format — Exercise 15
// Focus: auth header | Scenario: blog API

function ok(res, data, meta = {}) {
  return res.json({ success: true, data, meta });
}
function fail(res, status, message, details) {
  return res.status(status).json({ success: false, error: { message, details } });
}

app.get("/api/users", (req, res) => {
  ok(res, [{ id: 1, name: "Ada" }], { page: 1 });
});
app.get("/api/users/:id", (req, res) => {
  fail(res, 404, "User not found", { id: req.params.id });
});
// Consistent envelopes help clients and API gateways
