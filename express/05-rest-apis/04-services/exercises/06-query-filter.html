// Services — Exercise 06
// Focus: query filter | Scenario: ecommerce API

// services/usersService.js
const users = [{ id: 1, name: "Ada", email: "ada@example.com" }];

exports.findAll = () => users;
exports.findById = (id) => users.find((u) => u.id === Number(id));
exports.create = (dto) => {
  const user = { id: users.length + 1, ...dto };
  users.push(user);
  return user;
};

// controllers/usersController.js
const usersService = require("../services/usersService");
exports.list = (req, res) => res.json(usersService.findAll());
// Business logic lives in services, not route handlers
