// MongoDB with Express — Exercise 08
// Focus: router mount | Scenario: SSR site

const { MongoClient } = require("mongodb");

const client = new MongoClient(process.env.MONGODB_URI);
let db;

async function connect() {
  await client.connect();
  db = client.db("app");
}

app.get("/notes", async (req, res, next) => {
  try {
    const notes = await db.collection("notes").find().limit(20).toArray();
    res.json(notes);
  } catch (err) {
    next(err);
  }
});

connect().then(() => app.listen(3000));
// Official driver; for schemas/validation prefer Mongoose
