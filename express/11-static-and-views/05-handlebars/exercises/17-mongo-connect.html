// Handlebars — Exercise 17
// Focus: mongo connect | Scenario: realtime chat

const { engine } = require("express-handlebars");

app.engine("hbs", engine({ extname: ".hbs", defaultLayout: "main" }));
app.set("view engine", "hbs");
app.set("views", "./views");

app.get("/", (req, res) => {
  res.render("home", { title: "Home", items: ["a", "b"] });
});
// {{title}} and {{#each items}} in .hbs templates
