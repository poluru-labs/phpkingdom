Handlebars

Section: Static and Views

Scenario: REST API

Study the Express route/middleware pattern and adapt it to your API.

Express

// Handlebars — Example 11
// Focus: controller split | Scenario: REST API

const { engine } = require("express-handlebars");

app.engine("hbs", engine({ extname: ".hbs", defaultLayout: "main" }));
app.set("view engine", "hbs");
app.set("views", "./views");

app.get("/", (req, res) => {
  res.render("home", { title: "Home", items: ["a", "b"] });
});
// {{title}} and {{#each items}} in .hbs templates
