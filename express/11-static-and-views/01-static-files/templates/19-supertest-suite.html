Express

// Static Files — Template 19
// Focus: supertest suite | Scenario: microservice

const path = require("path");
const express = require("express");
const app = express();

app.use(express.static(path.join(__dirname, "public"), {
  maxAge: "1d",
  index: "index.html",
}));

app.get("*", (req, res) => {
  res.sendFile(path.join(__dirname, "public", "index.html"));
});
// Put CSS/JS/images in public/; API routes should be registered first
