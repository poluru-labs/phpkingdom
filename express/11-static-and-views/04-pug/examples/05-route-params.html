// Pug — Example 05

app.set("view engine", "pug");

app.get("/about", (req, res) => {
  res.render("about", { title: "About", site: "phpkingdom.com" });
});
// views/about.pug:
// html
//   head
//     title= title
//   body
//     h1= site

Checklist for Pug
