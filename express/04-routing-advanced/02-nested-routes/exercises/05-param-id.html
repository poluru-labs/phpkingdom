Express

// Nested Routes — Exercise 05
// Focus: param id | Scenario: blog API

const express = require("express");
const users = express.Router();
const posts = express.Router({ mergeParams: true });

posts.get("/", (req, res) => {
  res.json({ userId: req.params.userId, posts: [] });
});

users.use("/:userId/posts", posts);
users.get("/:userId", (req, res) => res.json({ id: req.params.userId }));

app.use("/users", users);
// mergeParams keeps parent :userId visible in child routers

Practice Notes

Checklist for Nested Routes
- Practice: param id
- Scenario: blog API
- Keep route handlers thin; push logic into controllers/services
- Always attach final 4-arg error middleware
- Validate input; load secrets from environment variables
