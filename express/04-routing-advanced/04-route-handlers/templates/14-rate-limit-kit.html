// Route Handlers — Template 14
// Focus: rate limit kit | Scenario: file service

function loadUser(req, res, next) {
  req.user = { id: req.params.id, name: "Ada" };
  next();
}

function authorize(req, res, next) {
  if (!req.user) return res.sendStatus(401);
  next();
}

function sendUser(req, res) {
  res.json(req.user);
}

// Multiple handlers in a chain
app.get("/users/:id", loadUser, authorize, sendUser);
