Starter Server

const express = require("express");
const app = express();

app.get("/", (req, res) => {
  res.type("html").send("<h1>Hello phpkingdom.com</h1>");
});

app.listen(3000, () => console.log("http://localhost:3000"));

Run

npm install express
node server.js
curl http://localhost:3000/health
