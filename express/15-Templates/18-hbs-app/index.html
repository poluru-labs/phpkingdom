express-handlebars

Handlebars layout + home view

Starter Server

const express = require("express");
const { engine } = require("express-handlebars");

const app = express();
app.engine("hbs", engine({ extname: ".hbs", defaultLayout: "main" }));
app.set("view engine", "hbs");
app.set("views", "./views");

app.get("/", (req, res) => {
  res.render("home", { title: "Home", items: ["a", "b", "c"] });
});
app.listen(3000);

Run

npm install express
node server.js
curl http://localhost:3000/health
