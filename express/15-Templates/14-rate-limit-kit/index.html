express-rate-limit

API + auth rate limiters

Starter Server

const express = require("express");
const rateLimit = require("express-rate-limit");

const app = express();
app.use("/api", rateLimit({ windowMs: 15 * 60 * 1000, max: 100 }));
app.post("/auth/login", rateLimit({ windowMs: 60 * 1000, max: 5 }), (req, res) => {
  res.json({ token: "demo" });
});
app.get("/api/ping", (req, res) => res.json({ pong: true }));
app.listen(3000);

Run

npm install express
node server.js
curl http://localhost:3000/health
