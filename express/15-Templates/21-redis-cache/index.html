Simple Redis GET/SET cache sketch

Starter Server

const express = require("express");
const Redis = require("ioredis");

const redis = new Redis(process.env.REDIS_URL || "redis://127.0.0.1:6379");
const app = express();

app.get("/stats", async (req, res, next) => {
  try {
    const cached = await redis.get("stats:v1");
    if (cached) return res.json({ source: "cache", ...JSON.parse(cached) });
    const data = { users: 42, at: Date.now() };
    await redis.set("stats:v1", JSON.stringify(data), "EX", 60);
    res.json({ source: "live", ...data });
  } catch (e) { next(e); }
});
app.listen(3000);

Run

npm install express
node server.js
curl http://localhost:3000/health
