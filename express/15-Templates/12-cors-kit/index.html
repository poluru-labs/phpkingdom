Restricted origin CORS setup

Starter Server

const express = require("express");
const cors = require("cors");

const app = express();
app.use(cors({
  origin: ["https://phpkingdom.com", "http://localhost:5173"],
  credentials: true,
  methods: ["GET", "POST", "PUT", "PATCH", "DELETE"],
}));

app.get("/api/public", (req, res) => res.json({ ok: true }));
app.listen(3000);

Run

npm install express
node server.js
curl http://localhost:3000/health
