express.Router

Mounted users router module

Starter Server

const express = require("express");
const users = express.Router();

users.get("/", (req, res) => res.json([{ id: 1, name: "Ada" }]));
users.get("/:id", (req, res) => res.json({ id: req.params.id }));

const app = express();
app.use("/users", users);
app.listen(3000);

module.exports = { app, users };

Run

npm install express
node server.js
curl http://localhost:3000/health
