Starter Server

const express = require("express");
const app = express();

app.set("view engine", "ejs");
app.set("views", "./views");

app.get("/", (req, res) => {
  res.render("home", { title: "Home", site: "phpkingdom.com" });
});
app.listen(3000);
// views/home.ejs → <h1><%= site %></h1>

Run

npm install express
node server.js
curl http://localhost:3000/health
