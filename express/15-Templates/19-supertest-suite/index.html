// app.js
const express = require("express");
const app = express();
app.get("/health", (req, res) => res.json({ ok: true }));
module.exports = app;

// app.test.js
const request = require("supertest");
const app = require("./app");

test("GET /health", async () => {
  await request(app).get("/health").expect(200, { ok: true });
});
