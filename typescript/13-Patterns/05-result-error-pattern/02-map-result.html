map on Result

Transform success values without unwrapping.

Keep error channel untouched.
type Result<T, E = string> =
  | { ok: true; value: T }
  | { ok: false; error: E };

function map<T, U, E>(r: Result<T, E>, fn: (v: T) => U): Result<U, E> {
  return r.ok ? { ok: true, value: fn(r.value) } : r;
}
