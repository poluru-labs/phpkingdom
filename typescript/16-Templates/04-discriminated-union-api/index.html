type ApiResponse<T> =
  | { status: "ok"; data: T }
  | { status: "error"; code: number; message: string };

function handle<T>(res: ApiResponse<T>) {
  if (res.status === "ok") return res.data;
  throw new Error(`${res.code}: ${res.message}`);
}
