Parse unknown input at runtime
import { z } from "zod";

const UserSchema = z.object({
  id: z.string(),
  email: z.string().email(),
});

type User = z.infer<typeof UserSchema>;

export function parseUser(data: unknown): User {
  return UserSchema.parse(data);
}
