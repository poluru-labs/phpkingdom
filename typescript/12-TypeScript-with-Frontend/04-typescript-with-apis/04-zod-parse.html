Runtime Parse Pattern

Validate unknown JSON.

Types alone do not validate network data.
type User = { id: string; name: string };
function isUser(v: unknown): v is User {
  return (
    typeof v === "object" &&
    v !== null &&
    typeof (v as User).id === "string" &&
    typeof (v as User).name === "string"
  );
}
async function getUser(url: string): Promise<User> {
  const data: unknown = await fetch(url).then((r) => r.json());
  if (!isUser(data)) throw new Error("Invalid user");
  return data;
}
