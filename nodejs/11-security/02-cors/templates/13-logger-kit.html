Node.js

// CORS — Template 13
// Focus: logger kit | Scenario: file processor

import express from "express";
import cors from "cors";

const app = express();
app.use(cors({
  origin: process.env.WEB_ORIGIN || "http://localhost:5173",
  credentials: true,
}));
app.get("/api/me", (_req, res) => res.json({ id: 1 }));
app.listen(3000);
