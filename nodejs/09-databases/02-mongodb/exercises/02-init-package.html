MongoDB with Node.js

Section: Databases

Focus: Init Package

Scenario: CLI tool

Run the code locally, then verify with curl, logs, or tests.

Node.js

// MongoDB with Node.js — Exercise 02
// Focus: init package | Scenario: CLI tool

import { MongoClient } from "mongodb";

const client = new MongoClient(process.env.MONGODB_URI);
await client.connect();
const db = client.db("app");
const users = db.collection("users");

await users.createIndex({ email: 1 }, { unique: true });
const doc = await users.findOne({ email: "ada@example.com" });
console.log(doc);
// await client.close() on shutdown
