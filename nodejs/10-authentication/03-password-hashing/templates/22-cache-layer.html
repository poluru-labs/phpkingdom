// Password Hashing — Template 22
// Focus: cache layer | Scenario: CLI tool

import bcrypt from "bcrypt";

const rounds = 12;
const hash = await bcrypt.hash("correct-horse", rounds);
const ok = await bcrypt.compare("correct-horse", hash);
console.log({ hash, ok });
// Never store plaintext; prefer argon2/bcrypt; constant-time compare
