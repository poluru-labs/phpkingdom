Sessions and Cookies

Section: Authentication

Focus: Stream Pipe

Scenario: public website API

Study the Node.js pattern and adapt it to your service or script.

Node.js

// Sessions and Cookies — Example 10
// Focus: stream pipe | Scenario: public website API

import express from "express";
import session from "express-session";

const app = express();
app.use(session({
  name: "sid",
  secret: process.env.SESSION_SECRET,
  resave: false,
  saveUninitialized: false,
  cookie: {
    httpOnly: true,
    sameSite: "lax",
    secure: process.env.NODE_ENV === "production",
    maxAge: 1000 * 60 * 60 * 8,
  },
}));

app.post("/login", express.json(), (req, res) => {
  req.session.userId = req.body.userId;
  res.json({ ok: true });
});
app.listen(3000);
