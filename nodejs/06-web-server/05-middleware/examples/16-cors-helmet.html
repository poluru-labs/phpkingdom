// Middleware Basics — Example 16
// Focus: cors helmet | Scenario: webhook receiver

import http from "node:http";

function compose(stack) {
  return (req, res) => {
    let i = 0;
    function next(err) {
      if (err) {
        res.writeHead(500); res.end("error"); return;
      }
      const fn = stack[i++];
      if (!fn) return;
      fn(req, res, next);
    }
    next();
  };
}

const withLog = (req, _res, next) => { console.log(req.method, req.url); next(); };
const ok = (_req, res) => { res.writeHead(200); res.end("ok"); };

http.createServer(compose([withLog, ok])).listen(3000);
