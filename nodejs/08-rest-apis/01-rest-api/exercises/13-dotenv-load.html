Node.js

// Node.js REST API — Exercise 13
// Focus: dotenv load | Scenario: file processor

import express from "express";

const app = express();
app.use(express.json());

app.get("/api/health", (_req, res) => res.json({ status: "ok" }));
app.get("/api/posts", (_req, res) => res.json([{ id: 1, title: "Hello" }]));
app.post("/api/posts", (req, res) => {
  res.status(201).json({ id: 2, ...req.body });
});
// REST: nouns as resources, HTTP verbs for actions, proper status codes
app.listen(3000);
