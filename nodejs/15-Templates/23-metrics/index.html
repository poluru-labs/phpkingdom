Server Scaffold

const metrics = {
  requests: 0,
  errors: 0,
};

export function trackRequest() {
  metrics.requests += 1;
}
export function trackError() {
  metrics.errors += 1;
}

import express from "express";
const app = express();
app.use((_req, _res, next) => { trackRequest(); next(); });
app.get("/metrics", (_req, res) => res.json(metrics));
app.listen(3000);
// Prefer Prometheus client / OpenTelemetry in production

Verify

node server.mjs
# or: npm start
curl http://localhost:3000/health
