import express from "express";
import rateLimit from "express-rate-limit";

const app = express();
const limiter = rateLimit({
  windowMs: 60_000,
  max: 60,
  standardHeaders: true,
  legacyHeaders: false,
});

app.use("/api", limiter);
app.get("/api/data", (_req, res) => res.json({ items: [] }));
app.listen(3000);
