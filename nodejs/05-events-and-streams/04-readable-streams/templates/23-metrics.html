Readable Streams

Node.js

// Readable Streams — Template 23
// Focus: metrics | Scenario: file processor

import { createReadStream } from "node:fs";
import { Readable } from "node:stream";

const file = createReadStream("README.md", { encoding: "utf8" });
file.on("data", (chunk) => process.stdout.write(chunk.slice(0, 20)));
file.on("end", () => console.log("\n-- eof"));
file.on("error", console.error);

// Or from iterable:
Readable.from(["a\n", "b\n"]).pipe(process.stdout);
