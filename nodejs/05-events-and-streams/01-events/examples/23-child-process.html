Node.js

// Node.js Events — Example 23
// Focus: child process | Scenario: file processor

import { EventEmitter } from "node:events";

class JobQueue extends EventEmitter {
  push(job) {
    this.emit("job", job);
  }
}

const q = new JobQueue();
q.on("job", (job) => console.log("process", job.id));
q.push({ id: 1, type: "email" });
