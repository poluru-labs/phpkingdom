Run the code locally, then verify with curl, logs, or tests.

Node.js

// EventEmitter — Exercise 21
// Focus: sql param | Scenario: REST microservice

import { EventEmitter } from "node:events";

const bus = new EventEmitter();
bus.on("job", (id) => console.log("job", id));
bus.once("ready", () => console.log("ready once"));
bus.emit("ready");
bus.emit("job", 42);
bus.emit("job", 43);
// Always handle 'error' events or the process may crash
