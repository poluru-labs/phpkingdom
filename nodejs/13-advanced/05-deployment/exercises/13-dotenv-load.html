# Production Node deploy checklist
# 1. NODE_ENV=production ; secrets via env / secret manager
# 2. node --watch off; use process manager (systemd / pm2) or containers
# 3. Health: GET /health ; readiness vs liveness
# 4. Graceful shutdown on SIGTERM (close server + DB pools)
# 5. Structured logs, metrics, and error tracking
# 6. Run behind reverse proxy / TLS terminator

process.on("SIGTERM", () => {
  console.log("shutting down");
  process.exit(0);
});
