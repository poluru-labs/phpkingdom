// Node.js Performance — Template 15
// Focus: test suite | Scenario: admin backend

import { performance, PerformanceObserver } from "node:perf_hooks";

const obs = new PerformanceObserver((list) => {
  for (const e of list.getEntries()) console.log(e.name, e.duration.toFixed(2), "ms");
});
obs.observe({ entryTypes: ["measure"] });

performance.mark("work-start");
// ... do work ...
for (let i = 0; i < 1e6; i++) Math.sqrt(i);
performance.mark("work-end");
performance.measure("work", "work-start", "work-end");
// Profile with: node --cpu-prof app.mjs ; avoid sync I/O on hot paths

Checklist for Node.js Performance
