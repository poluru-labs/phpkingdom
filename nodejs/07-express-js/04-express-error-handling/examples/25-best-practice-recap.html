// Express Error Handling — Example 25
// Focus: best practice recap | Scenario: admin backend

import express from "express";

const app = express();
app.get("/boom", (_req, _res, next) => {
  next(Object.assign(new Error("explode"), { status: 400 }));
});

// Must be 4-arity
app.use((err, _req, res, _next) => {
  console.error(err);
  res.status(err.status || 500).json({ error: err.message || "Server error" });
});

app.listen(3000);
