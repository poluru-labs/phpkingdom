// Reduce Package Size — Template 16
// Focus: localstack | Scenario: image service

// AWS Lambda (Node.js) topic: Reduce Package Size
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "package-size",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: package-size
  const result = { ok: true, topic: "package-size", label: "Reduce Package Size", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_PACKAGE_SIZE = "package-size";
