// Integration Testing — Template 13
// Focus: sam template | Scenario: queue worker

// AWS Lambda (Node.js) topic: Integration Testing
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "integration-testing",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: integration-testing
  const result = { ok: true, topic: "integration-testing", label: "Integration Testing", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_INTEGRATION_TESTING = "integration-testing";

Practice Notes

Checklist for Integration Testing
- Practice: sam template
- Scenario: queue worker
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
