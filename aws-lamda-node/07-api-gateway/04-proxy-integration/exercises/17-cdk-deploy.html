// Lambda Proxy Integration — Exercise 17
// Focus: cdk deploy | Scenario: notification hub

// AWS Lambda (Node.js) topic: Lambda Proxy Integration
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "proxy-integration",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: proxy-integration
  const result = { ok: true, topic: "proxy-integration", label: "Lambda Proxy Integration", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_PROXY_INTEGRATION = "proxy-integration";

Practice Notes

Checklist for Lambda Proxy Integration
- Practice: cdk deploy
- Scenario: notification hub
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts

Useful commands:
  aws lambda invoke --function-name demo out.json
  sam local invoke -e events/event.json
  npm test
