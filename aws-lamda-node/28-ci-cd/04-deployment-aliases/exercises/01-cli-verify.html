// Lambda Versions and Aliases — Exercise 01
// Focus: cli verify | Scenario: REST microservice

// AWS Lambda (Node.js) topic: Lambda Versions and Aliases
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "deployment-aliases",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: deployment-aliases
  const result = { ok: true, topic: "deployment-aliases", label: "Lambda Versions and Aliases", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_DEPLOYMENT_ALIASES = "deployment-aliases";

Practice Notes

Checklist for Lambda Versions and Aliases
- Practice: cli verify
- Scenario: REST microservice
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
