// CDK API Gateway — Exercise 08
// Focus: s3 trigger | Scenario: order workflow

export const handler = async (event) => {
  const method = event.requestContext?.http?.method || event.httpMethod;
  const path = event.rawPath || event.path;
  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify({ method, path, qs: event.queryStringParameters }),
  };
};

Practice Notes

Checklist for CDK API Gateway
- Practice: s3 trigger
- Scenario: order workflow
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts

Useful commands:
  aws lambda invoke --function-name demo out.json
  sam local invoke -e events/event.json
  npm test
  npx esbuild index.js --bundle --platform=node --target=node20 --outfile=dist/index.js
