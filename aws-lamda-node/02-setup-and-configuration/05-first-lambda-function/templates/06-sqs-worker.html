// First Node.js Lambda — Template 06
// Focus: sqs worker | Scenario: image service

// AWS Lambda (Node.js) topic: First Node.js Lambda
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "first-lambda-function",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: first-lambda-function
  const result = { ok: true, topic: "first-lambda-function", label: "First Node.js Lambda", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_FIRST_LAMBDA_FUNCTION = "first-lambda-function";

Practice Notes

Checklist for First Node.js Lambda
- Practice: sqs worker
- Scenario: image service
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts

Useful commands:
  aws lambda invoke --function-name demo out.json
  sam local invoke -e events/event.json
  npm test
  npx esbuild index.js --bundle --platform=node --target=node20 --outfile=dist/index.js
