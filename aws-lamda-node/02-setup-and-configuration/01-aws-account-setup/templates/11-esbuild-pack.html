// AWS Lambda (Node.js) topic: AWS Account Setup
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "aws-account-setup",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: aws-account-setup
  const result = { ok: true, topic: "aws-account-setup", label: "AWS Account Setup", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_AWS_ACCOUNT_SETUP = "aws-account-setup";
