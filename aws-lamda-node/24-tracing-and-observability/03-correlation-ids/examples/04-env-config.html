// Correlation IDs — Example 04
// Focus: env config | Scenario: scheduled job

// AWS Lambda (Node.js) topic: Correlation IDs
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "correlation-ids",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: correlation-ids
  const result = { ok: true, topic: "correlation-ids", label: "Correlation IDs", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_CORRELATION_IDS = "correlation-ids";
