// Environment Variables — Example 10
// Focus: sns publish | Scenario: data sync

// AWS Lambda (Node.js) topic: Environment Variables
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "environment-variables",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: environment-variables
  const result = { ok: true, topic: "environment-variables", label: "Environment Variables", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_ENVIRONMENT_VARIABLES = "environment-variables";
