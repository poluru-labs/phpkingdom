// AWS Lambda (Node.js) topic: Ephemeral Storage
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "ephemeral-storage",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: ephemeral-storage
  const result = { ok: true, topic: "ephemeral-storage", label: "Ephemeral Storage", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_EPHEMERAL_STORAGE = "ephemeral-storage";
