// LLRT = experimental low-latency JS runtime for Lambda (subset of Node APIs)
// Validate: fetch, JSON, crypto — avoid Node-only native modules

export const handler = async (event) => {
  const body = typeof event.body === "string" ? JSON.parse(event.body || "{}") : event;
  return { statusCode: 200, body: JSON.stringify({ runtime: "llrt?", echo: body }) };
};

export const TOPIC_ID = "llrt";
