// AWS Lambda Powertools — Template 20
// Focus: canary alias | Scenario: data sync

import { Logger } from "@aws-lambda-powertools/logger";
import { Tracer } from "@aws-lambda-powertools/tracer";
import { Metrics, MetricUnits } from "@aws-lambda-powertools/metrics";

const logger = new Logger({ serviceName: "orders" });
const tracer = new Tracer({ serviceName: "orders" });
const metrics = new Metrics({ namespace: "Orders", serviceName: "orders" });

export const handler = async (event, context) => {
  logger.addContext(context);
  logger.info("request", { path: event.rawPath });
  metrics.addMetric("Request", MetricUnits.Count, 1);
  metrics.publishStoredMetrics();
  return { statusCode: 200, body: JSON.stringify({ ok: true }) };
};

export const TOPIC_ID = "powertools";

Practice Notes

Checklist for AWS Lambda Powertools
- Practice: canary alias
- Scenario: data sync
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
