S3 Event Notifications

Study the Node.js Lambda pattern and adapt it to your AWS project.

Node.js Lambda

// S3 Event Notifications — Example 22
// Focus: idempotent write | Scenario: file pipeline

export const handler = async (event) => {
  for (const record of event.Records || []) {
    const bucket = record.s3.bucket.name;
    const key = decodeURIComponent(record.s3.object.key.replace(/\+/g, " "));
    console.log(JSON.stringify({ bucket, key, eventName: record.eventName }));
  }
  return { ok: true };
};

export const TOPIC_ID = "s3-event-notifications";
