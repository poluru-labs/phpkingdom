// AWS Lambda (Node.js) topic: Lambda Container Images
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "container-images",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: container-images
  const result = { ok: true, topic: "container-images", label: "Lambda Container Images", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_CONTAINER_IMAGES = "container-images";
