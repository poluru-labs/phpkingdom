Amazon Bedrock Invoke

Implement the handler locally, then invoke with SAM/CLI and verify logs.

Node.js Lambda

// Amazon Bedrock Invoke — Exercise 13
// Focus: parameter store | Scenario: queue worker

import { BedrockRuntimeClient, InvokeModelCommand } from "@aws-sdk/client-bedrock-runtime";

const bedrock = new BedrockRuntimeClient({});

export const handler = async (event) => {
  const prompt = JSON.parse(event.body || "{}").prompt || "Hello";
  const res = await bedrock.send(new InvokeModelCommand({
    modelId: "amazon.titan-text-express-v1",
    contentType: "application/json",
    accept: "application/json",
    body: JSON.stringify({ inputText: prompt }),
  }));
  const payload = JSON.parse(new TextDecoder().decode(res.body));
  return { statusCode: 200, body: JSON.stringify(payload) };
};

export const TOPIC_ID = "bedrock-invoke";
