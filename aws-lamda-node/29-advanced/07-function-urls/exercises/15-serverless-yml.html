// Lambda Function URL — HTTPS endpoint without API Gateway
export const handler = async (event) => {
  const method = event.requestContext?.http?.method || "GET";
  if (method === "OPTIONS") {
    return { statusCode: 204, headers: { "access-control-allow-origin": "*" } };
  }
  return {
    statusCode: 200,
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ via: "function-url", method }),
  };
};

Practice Notes

Checklist for Lambda Function URLs
- Practice: serverless yml
- Scenario: auth API
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts

Useful commands:
  aws lambda invoke --function-name demo out.json
  sam local invoke -e events/event.json
  npm test
  npx esbuild index.js --bundle --platform=node --target=node20 --outfile=dist/index.js
