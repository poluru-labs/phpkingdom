// Lambda Destinations — Example 22
// Focus: idempotent write | Scenario: file pipeline

// AWS Lambda (Node.js) topic: Lambda Destinations
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "lambda-destinations",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: lambda-destinations
  const result = { ok: true, topic: "lambda-destinations", label: "Lambda Destinations", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_LAMBDA_DESTINATIONS = "lambda-destinations";
