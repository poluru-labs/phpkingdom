// Module Not Found Errors — Exercise 15

// AWS Lambda (Node.js) topic: Module Not Found Errors
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "module-not-found",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: module-not-found
  const result = { ok: true, topic: "module-not-found", label: "Module Not Found Errors", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_MODULE_NOT_FOUND = "module-not-found";
