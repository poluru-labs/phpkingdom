// Amazon CloudWatch — Template 20

// AWS Lambda (Node.js) topic: Amazon CloudWatch
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "cloudwatch",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: cloudwatch
  const result = { ok: true, topic: "cloudwatch", label: "Amazon CloudWatch", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_CLOUDWATCH = "cloudwatch";
