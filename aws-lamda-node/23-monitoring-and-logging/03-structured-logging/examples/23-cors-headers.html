Node.js Lambda

// Structured Logging — Example 23
// Focus: cors headers | Scenario: queue worker

// AWS Lambda (Node.js) topic: Structured Logging
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "structured-logging",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: structured-logging
  const result = { ok: true, topic: "structured-logging", label: "Structured Logging", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_STRUCTURED_LOGGING = "structured-logging";
