// PUT and DELETE Requests — Exercise 06
// Focus: post body | Scenario: image service

// AWS Lambda (Node.js) topic: PUT and DELETE Requests
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "put-delete-request",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: put-delete-request
  const result = { ok: true, topic: "put-delete-request", label: "PUT and DELETE Requests", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_PUT_DELETE_REQUEST = "put-delete-request";
