// Local Event Testing — Example 25

// AWS Lambda (Node.js) topic: Local Event Testing
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "local-events",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: local-events
  const result = { ok: true, topic: "local-events", label: "Local Event Testing", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_LOCAL_EVENTS = "local-events";
