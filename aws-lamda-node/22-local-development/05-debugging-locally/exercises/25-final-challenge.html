// AWS Lambda (Node.js) topic: Debug Lambda Locally
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "debugging-locally",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: debugging-locally
  const result = { ok: true, topic: "debugging-locally", label: "Debug Lambda Locally", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_DEBUGGING_LOCALLY = "debugging-locally";
