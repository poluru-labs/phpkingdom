// Lambda Authorizer — Example 23
// Focus: cors headers | Scenario: queue worker

// AWS Lambda (Node.js) topic: Lambda Authorizer
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "api-authorizer",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: api-authorizer
  const result = { ok: true, topic: "api-authorizer", label: "Lambda Authorizer", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_API_AUTHORIZER = "api-authorizer";
