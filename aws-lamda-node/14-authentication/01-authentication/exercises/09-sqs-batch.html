// Serverless Authentication — Exercise 09
// Focus: sqs batch | Scenario: webhook receiver

// AWS Lambda (Node.js) topic: Serverless Authentication
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "authentication",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: authentication
  const result = { ok: true, topic: "authentication", label: "Serverless Authentication", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_AUTHENTICATION = "authentication";

Practice Notes

Checklist for Serverless Authentication
- Practice: sqs batch
- Scenario: webhook receiver
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
