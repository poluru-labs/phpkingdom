// Download Files from S3 — Exercise 24

// AWS Lambda (Node.js) topic: Download Files from S3
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "download-files",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: download-files
  const result = { ok: true, topic: "download-files", label: "Download Files from S3", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_DOWNLOAD_FILES = "download-files";
