// Upload Files to S3 — Template 11
// Focus: esbuild pack | Scenario: REST microservice

// AWS Lambda (Node.js) topic: Upload Files to S3
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "upload-files",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: upload-files
  const result = { ok: true, topic: "upload-files", label: "Upload Files to S3", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_UPLOAD_FILES = "upload-files";
