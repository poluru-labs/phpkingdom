// AWS Lambda (Node.js) topic: IAM Permission Errors
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "permission-errors",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: permission-errors
  const result = { ok: true, topic: "permission-errors", label: "IAM Permission Errors", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_PERMISSION_ERRORS = "permission-errors";
