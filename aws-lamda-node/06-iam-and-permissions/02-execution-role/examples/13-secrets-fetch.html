// Lambda Execution Role — Example 13
// Focus: secrets fetch | Scenario: queue worker

// AWS Lambda (Node.js) topic: Lambda Execution Role
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "execution-role",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: execution-role
  const result = { ok: true, topic: "execution-role", label: "Lambda Execution Role", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_EXECUTION_ROLE = "execution-role";
