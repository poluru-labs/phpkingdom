// AWS Lambda (Node.js) topic: Resource-Based Policies
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "resource-policies",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: resource-policies
  const result = { ok: true, topic: "resource-policies", label: "Resource-Based Policies", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_RESOURCE_POLICIES = "resource-policies";
