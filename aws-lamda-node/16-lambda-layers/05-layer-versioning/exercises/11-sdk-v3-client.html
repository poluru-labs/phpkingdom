// Layer Versioning — Exercise 11
// Focus: sdk v3 client | Scenario: REST microservice

// AWS Lambda (Node.js) topic: Layer Versioning
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "layer-versioning",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: layer-versioning
  const result = { ok: true, topic: "layer-versioning", label: "Layer Versioning", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_LAYER_VERSIONING = "layer-versioning";

Practice Notes

Checklist for Layer Versioning
- Practice: sdk v3 client
- Scenario: REST microservice
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
