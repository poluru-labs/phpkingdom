Node.js Lambda

// Create a Lambda Layer — Example 22
// Focus: idempotent write | Scenario: file pipeline

// AWS Lambda (Node.js) topic: Create a Lambda Layer
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "create-layer",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: create-layer
  const result = { ok: true, topic: "create-layer", label: "Create a Lambda Layer", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_CREATE_LAYER = "create-layer";
