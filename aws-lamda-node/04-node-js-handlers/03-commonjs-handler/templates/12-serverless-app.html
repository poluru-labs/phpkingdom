// AWS Lambda (Node.js) topic: CommonJS Lambda Handler
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "commonjs-handler",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: commonjs-handler
  const result = { ok: true, topic: "commonjs-handler", label: "CommonJS Lambda Handler", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_COMMONJS_HANDLER = "commonjs-handler";
