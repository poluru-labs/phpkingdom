// Async Lambda Handler — Example 23
// Focus: cors headers | Scenario: queue worker

// AWS Lambda (Node.js) topic: Async Lambda Handler
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "async-handler",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: async-handler
  const result = { ok: true, topic: "async-handler", label: "Async Lambda Handler", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_ASYNC_HANDLER = "async-handler";
