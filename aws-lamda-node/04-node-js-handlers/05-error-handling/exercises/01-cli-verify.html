// Handler Error Handling — Exercise 01
// Focus: cli verify | Scenario: REST microservice

// AWS Lambda (Node.js) topic: Handler Error Handling
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "error-handling",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: error-handling
  const result = { ok: true, topic: "error-handling", label: "Handler Error Handling", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_ERROR_HANDLING = "error-handling";
