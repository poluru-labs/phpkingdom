// AWS Lambda (Node.js) topic: Callback Handler
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "callback-handler",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: callback-handler
  const result = { ok: true, topic: "callback-handler", label: "Callback Handler", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_CALLBACK_HANDLER = "callback-handler";
