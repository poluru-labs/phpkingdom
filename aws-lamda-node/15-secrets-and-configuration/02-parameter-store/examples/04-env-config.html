// SSM Parameter Store — Example 04
// Focus: env config | Scenario: scheduled job

// AWS Lambda (Node.js) topic: SSM Parameter Store
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "parameter-store",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: parameter-store
  const result = { ok: true, topic: "parameter-store", label: "SSM Parameter Store", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_PARAMETER_STORE = "parameter-store";
