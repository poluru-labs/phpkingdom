// Secret Caching — Example 24
// Focus: dlq setup | Scenario: scheduled job

// AWS Lambda (Node.js) topic: Secret Caching
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "secret-caching",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: secret-caching
  const result = { ok: true, topic: "secret-caching", label: "Secret Caching", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SECRET_CACHING = "secret-caching";
