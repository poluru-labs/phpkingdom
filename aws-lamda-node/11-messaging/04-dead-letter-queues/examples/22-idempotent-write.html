// Dead-Letter Queues — Example 22

// AWS Lambda (Node.js) topic: Dead-Letter Queues
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "dead-letter-queues",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: dead-letter-queues
  const result = { ok: true, topic: "dead-letter-queues", label: "Dead-Letter Queues", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_DEAD_LETTER_QUEUES = "dead-letter-queues";
