// Amazon SNS with Lambda — Exercise 21
// Focus: memory tune | Scenario: REST microservice

// AWS Lambda (Node.js) topic: Amazon SNS with Lambda
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "sns",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: sns
  const result = { ok: true, topic: "sns", label: "Amazon SNS with Lambda", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SNS = "sns";
