// Message Retry Strategies — Exercise 03
// Focus: async handler | Scenario: queue worker

// AWS Lambda (Node.js) topic: Message Retry Strategies
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "message-retries",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: message-retries
  const result = { ok: true, topic: "message-retries", label: "Message Retry Strategies", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_MESSAGE_RETRIES = "message-retries";
