// Scheduled Lambda Functions — Example 12
// Focus: jwt verify | Scenario: file pipeline

// AWS Lambda (Node.js) topic: Scheduled Lambda Functions
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "scheduled-events",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: scheduled-events
  const result = { ok: true, topic: "scheduled-events", label: "Scheduled Lambda Functions", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SCHEDULED_EVENTS = "scheduled-events";

Practice Notes

Checklist for Scheduled Lambda Functions
- Practice: jwt verify
- Scenario: file pipeline
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts

Useful commands:

  sam local invoke -e events/event.json
