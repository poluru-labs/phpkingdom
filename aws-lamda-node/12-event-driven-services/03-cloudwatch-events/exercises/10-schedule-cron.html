// AWS Lambda (Node.js) topic: CloudWatch Events
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "cloudwatch-events",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: cloudwatch-events
  const result = { ok: true, topic: "cloudwatch-events", label: "CloudWatch Events", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_CLOUDWATCH_EVENTS = "cloudwatch-events";
