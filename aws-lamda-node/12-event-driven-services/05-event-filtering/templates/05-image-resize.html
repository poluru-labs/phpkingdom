// AWS Lambda (Node.js) topic: Event Source Filtering
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "event-filtering",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: event-filtering
  const result = { ok: true, topic: "event-filtering", label: "Event Source Filtering", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_EVENT_FILTERING = "event-filtering";
