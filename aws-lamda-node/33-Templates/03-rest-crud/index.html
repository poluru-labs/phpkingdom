import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient, GetCommand, PutCommand, DeleteCommand } from "@aws-sdk/lib-dynamodb";
const doc = DynamoDBDocumentClient.from(new DynamoDBClient({}));
const TableName = process.env.TABLE_NAME;
export const handler = async (event) => {
  const id = event.pathParameters?.id;
  const method = event.requestContext?.http?.method;
  if (method === "GET") {
    const { Item } = await doc.send(new GetCommand({ TableName, Key: { id } }));
    return { statusCode: Item ? 200 : 404, body: JSON.stringify(Item || {}) };
  }
  if (method === "PUT") {
    const Item = { id, ...JSON.parse(event.body || "{}") };
    await doc.send(new PutCommand({ TableName, Item }));
    return { statusCode: 200, body: JSON.stringify(Item) };
  }
  if (method === "DELETE") {
    await doc.send(new DeleteCommand({ TableName, Key: { id } }));
    return { statusCode: 204, body: "" };
  }
  return { statusCode: 405, body: "Method Not Allowed" };
};
