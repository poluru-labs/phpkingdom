// Input Validation — Exercise 15

// AWS Lambda (Node.js) topic: Input Validation
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "input-validation",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: input-validation
  const result = { ok: true, topic: "input-validation", label: "Input Validation", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_INPUT_VALIDATION = "input-validation";
