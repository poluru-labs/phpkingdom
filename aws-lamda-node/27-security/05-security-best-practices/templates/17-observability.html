// Serverless Security Practices — Template 17
// Focus: observability | Scenario: notification hub

// AWS Lambda (Node.js) topic: Serverless Security Practices
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "security-best-practices",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: security-best-practices
  const result = { ok: true, topic: "security-best-practices", label: "Serverless Security Practices", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SECURITY_BEST_PRACTICES = "security-best-practices";

Practice Notes

Checklist for Serverless Security Practices
- Practice: observability
- Scenario: notification hub
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
