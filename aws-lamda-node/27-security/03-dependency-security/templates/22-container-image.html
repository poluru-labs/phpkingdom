// Dependency Security — Template 22
// Focus: container image | Scenario: file pipeline

// AWS Lambda (Node.js) topic: Dependency Security
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "dependency-security",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: dependency-security
  const result = { ok: true, topic: "dependency-security", label: "Dependency Security", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_DEPENDENCY_SECURITY = "dependency-security";

Practice Notes

Checklist for Dependency Security
- Practice: container image
- Scenario: file pipeline
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
