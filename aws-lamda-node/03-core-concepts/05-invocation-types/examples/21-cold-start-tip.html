// Lambda Invocation Types — Example 21
// Focus: cold start tip | Scenario: REST microservice

// AWS Lambda (Node.js) topic: Lambda Invocation Types
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "invocation-types",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: invocation-types
  const result = { ok: true, topic: "invocation-types", label: "Lambda Invocation Types", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_INVOCATION_TYPES = "invocation-types";
