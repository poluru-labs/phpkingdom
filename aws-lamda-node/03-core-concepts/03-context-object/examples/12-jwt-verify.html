// Lambda Context Object — Example 12
// Focus: jwt verify | Scenario: file pipeline

// AWS Lambda (Node.js) topic: Lambda Context Object
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "context-object",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: context-object
  const result = { ok: true, topic: "context-object", label: "Lambda Context Object", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_CONTEXT_OBJECT = "context-object";
