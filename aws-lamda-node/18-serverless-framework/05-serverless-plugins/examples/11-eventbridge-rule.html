// Serverless Plugins — Example 11
// Focus: eventbridge rule | Scenario: REST microservice

// AWS Lambda (Node.js) topic: Serverless Plugins
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "serverless-plugins",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: serverless-plugins
  const result = { ok: true, topic: "serverless-plugins", label: "Serverless Plugins", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SERVERLESS_PLUGINS = "serverless-plugins";
