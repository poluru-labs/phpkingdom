// Serverless Framework Setup — Example 08
// Focus: s3 upload | Scenario: order workflow

// AWS Lambda (Node.js) topic: Serverless Framework Setup
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "serverless-setup",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: serverless-setup
  const result = { ok: true, topic: "serverless-setup", label: "Serverless Framework Setup", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SERVERLESS_SETUP = "serverless-setup";

Practice Notes

Checklist for Serverless Framework Setup
- Practice: s3 upload
- Scenario: order workflow
- Keep handlers thin; put business logic in modules
- Reuse SDK clients outside the handler when possible
- Log requestId; avoid logging secrets
- Set memory/timeout for your workload; test cold starts
