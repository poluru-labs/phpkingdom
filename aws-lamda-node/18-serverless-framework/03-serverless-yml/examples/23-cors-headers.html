// serverless.yml — Example 23
// Focus: cors headers | Scenario: queue worker

// AWS Lambda (Node.js) topic: serverless.yml
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "serverless-yml",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: serverless-yml
  const result = { ok: true, topic: "serverless-yml", label: "serverless.yml", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SERVERLESS_YML = "serverless-yml";
