// Serverless File Upload — Template 09
// Focus: cognito jwt | Scenario: webhook receiver

// AWS Lambda (Node.js) topic: Serverless File Upload
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "file-upload-project",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: file-upload-project
  const result = { ok: true, topic: "file-upload-project", label: "Serverless File Upload", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_FILE_UPLOAD_PROJECT = "file-upload-project";
