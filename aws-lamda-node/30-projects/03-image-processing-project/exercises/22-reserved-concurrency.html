// AWS Lambda (Node.js) topic: Image Processing Service
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "image-processing-project",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: image-processing-project
  const result = { ok: true, topic: "image-processing-project", label: "Image Processing Service", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_IMAGE_PROCESSING_PROJECT = "image-processing-project";
