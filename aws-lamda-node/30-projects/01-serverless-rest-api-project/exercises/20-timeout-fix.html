// AWS Lambda (Node.js) topic: Serverless REST API
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "serverless-rest-api-project",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: serverless-rest-api-project
  const result = { ok: true, topic: "serverless-rest-api-project", label: "Serverless REST API", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_SERVERLESS_REST_API_PROJECT = "serverless-rest-api-project";
