// Notification Service — Example 17
// Focus: cdk construct | Scenario: notification hub

// AWS Lambda (Node.js) topic: Notification Service
export const handler = async (event, context) => {
  console.log(JSON.stringify({
    topic: "notification-project",
    requestId: context.awsRequestId,
    remainingMs: context.getRemainingTimeInMillis(),
  }));

  const body = typeof event?.body === "string"
    ? JSON.parse(event.body || "{}")
    : (event?.body || event || {});

  // Focus area: notification-project
  const result = { ok: true, topic: "notification-project", label: "Notification Service", input: body };

  return {
    statusCode: 200,
    headers: {
      "content-type": "application/json",
      "access-control-allow-origin": "*",
    },
    body: JSON.stringify(result),
  };
};

export const TOPIC_NOTIFICATION_PROJECT = "notification-project";
