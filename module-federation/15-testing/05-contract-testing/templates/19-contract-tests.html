const exposes = ["./ProductList", "./ProductDetail"];

test("remote exposes stable modules", async () => {
  // After loading container in jsdom/node harness:
  for (const mod of exposes) {
    const factory = await container.get(mod);
    const m = factory();
    expect(m.default || m).toBeTypeOf("function");
  }
});
