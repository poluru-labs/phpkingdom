Module Federation

// Module Federation - Introduction — Exercise 08
// Focus: requiredVersion | Scenario: legacy migration

// Module Federation (Webpack 5+) loads code from other builds at runtime

const { ModuleFederationPlugin } = require("webpack").container;

// Remote exposes modules:
new ModuleFederationPlugin({
  name: "products",
  filename: "remoteEntry.js",
  exposes: { "./ProductList": "./src/ProductList" },
});

// Host consumes them:
new ModuleFederationPlugin({
  name: "shell",
  remotes: {
    products: "products@http://localhost:3002/remoteEntry.js",
  },
});

// Host app: import("products/ProductList")
// Key ideas: host, remote, exposes, remotes, shared
