// Application Shell — Example 10
// Focus: error boundary | Scenario: B2B portal

// Application shell: chrome + outlets for remotes

// ShellApp.jsx
import React, { Suspense, lazy } from "react";
import { BrowserRouter, Routes, Route, Link } from "react-router-dom";

const Catalog = lazy(() => import("catalog/CatalogApp"));
const Checkout = lazy(() => import("checkout/CheckoutApp"));

export default function ShellApp() {
  return (
    <BrowserRouter>
      <header>
        <Link to="/">Home</Link>
        <Link to="/catalog">Catalog</Link>
        <Link to="/checkout">Checkout</Link>
      </header>
      <Suspense fallback={<div>Loading…</div>}>
        <Routes>
          <Route path="/catalog/*" element={<Catalog />} />
          <Route path="/checkout/*" element={<Checkout />} />
        </Routes>
      </Suspense>
    </BrowserRouter>
  );
}
// Shell owns auth banner, nav, and error boundaries around outlets

- Deploy remotes independently; version remoteEntry URLs carefully
