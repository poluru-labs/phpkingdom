// Fallback Components — Exercise 15

// Local fallback when a remote fails to load

const ProductList = React.lazy(() =>
  import("products/ProductList").catch(() => ({
    default: function ProductsOffline() {
      return <div role="alert">Products unavailable — try again later.</div>;
    },
  }))
);

export function ProductsOutlet() {
  return (
    <React.Suspense fallback={<div>Loading products…</div>}>
      <ProductList />
    </React.Suspense>
  );
}
