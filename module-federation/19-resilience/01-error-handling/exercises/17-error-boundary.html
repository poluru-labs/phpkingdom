// 2) Error boundary → render failures
// 3) Retry / timeout in remote loader
// 4) User-visible toast + telemetry

async function importRemote(path) {
  try {
    return await import(path);
  } catch (e) {
    reportError(e, { remote: path });
    throw e;
  }
}
