// Use the URL as cross-app shared state (deep-link friendly)

// Host router keeps query in sync
// /catalog?sku=SKU-1&sort=price

import { useSearchParams } from "react-router-dom";

export function CatalogHost() {
  const [params, setParams] = useSearchParams();
  const sku = params.get("sku");
  return (
    <RemoteCatalog
      sku={sku}
      onSku={(id) => setParams({ sku: id })}
    />
  );
}

// Remotes read the same query via shared router or window.location
// Good for shareable links; avoid putting secrets in the URL
