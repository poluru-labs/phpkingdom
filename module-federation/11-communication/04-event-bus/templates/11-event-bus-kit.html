// Tiny typed event bus shared as a singleton package

// packages/event-bus/index.js
const listeners = new Map();
export const bus = {
  on(type, fn) {
    if (!listeners.has(type)) listeners.set(type, new Set());
    listeners.get(type).add(fn);
    return () => listeners.get(type).delete(fn);
  },
  emit(type, detail) {
    listeners.get(type)?.forEach((fn) => fn(detail));
  },
};

// shared: { "@acme/bus": { singleton: true, requiredVersion: "1.0.0" } }

// remote: bus.emit("cart:add", { id: "sku-1" });
// host:   bus.on("cart:add", (item) => dispatch(addItem(item)));
