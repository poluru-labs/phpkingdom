// Cross-remote messaging with DOM CustomEvent

// remote cart
export function addToCart(item) {
  window.dispatchEvent(
    new CustomEvent("cart:add", { detail: item, bubbles: true })
  );
}

// host shell
useEffect(() => {
  const handler = (e) => setCart((c) => [...c, e.detail]);
  window.addEventListener("cart:add", handler);
  return () => window.removeEventListener("cart:add", handler);
}, []);
