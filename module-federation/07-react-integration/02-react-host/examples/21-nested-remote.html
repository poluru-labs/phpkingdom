Module Federation

// React Host Application — Example 21
// Focus: nested remote | Scenario: e-commerce shell

// React host webpack + lazy remote

const { ModuleFederationPlugin } = require("webpack").container;

new ModuleFederationPlugin({
  name: "shell",
  remotes: {
    products: "products@http://localhost:3002/remoteEntry.js",
  },
  shared: {
    react: { singleton: true, requiredVersion: "^18.0.0", eager: true },
    "react-dom": { singleton: true, requiredVersion: "^18.0.0", eager: true },
  },
});

// App.jsx
import React, { Suspense, lazy } from "react";
const ProductList = lazy(() => import("products/ProductList"));

export default function App() {
  return (
    <Suspense fallback={<div>Loading products…</div>}>
      <ProductList />
    </Suspense>
  );
}

Practice Notes

Checklist for React Host Application
- Practice: nested remote
- Scenario: e-commerce shell
- Share framework runtimes as singletons with requiredVersion
- Load remotes lazily; wrap with error boundaries / fallbacks
- Deploy remotes independently; version remoteEntry URLs carefully
- Align CORS and CSP for cross-origin remoteEntry.js

Useful commands:
  npm install -D webpack webpack-cli webpack-dev-server
  # run host + remote on different ports
  npx webpack serve --config webpack.host.js --port 3000
  npx webpack serve --config webpack.remote.js --port 3002
