// Remote Preloading — Template 07

// Prefetch remotes during idle time or on link hover

function prefetchRemote(entryUrl) {
  const link = document.createElement("link");
  link.rel = "prefetch";
  link.as = "script";
  link.href = entryUrl;
  document.head.appendChild(link);
}

// On intent:
onMouseEnter={() => prefetchRemote("https://cdn/.../products/remoteEntry.js")}

// Or after shell interactive:
requestIdleCallback(() => prefetchRemote(productsEntry));

// Don't prefetch every remote on first paint — prioritize likely next routes

Practice Notes

Checklist for Remote Preloading
- Practice: shared ui kit
- Scenario: multi-team platform
- Share framework runtimes as singletons with requiredVersion
- Load remotes lazily; wrap with error boundaries / fallbacks
- Deploy remotes independently; version remoteEntry URLs carefully
- Align CORS and CSP for cross-origin remoteEntry.js

Useful commands:
  npm install -D webpack webpack-cli webpack-dev-server
  # run host + remote on different ports
  npx webpack serve --config webpack.host.js --port 3000
  npx webpack serve --config webpack.remote.js --port 3002
