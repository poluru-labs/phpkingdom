// Module Federation 2.0 — Example 19
// Focus: contract test | Scenario: mobile web shell

// Module Federation 2.0 — runtime + enhanced plugin + manifests

import { init, loadRemote } from "@module-federation/enhanced/runtime";

init({
  name: "shell",
  remotes: [
    {
      name: "products",
      entry: "http://localhost:3002/mf-manifest.json",
    },
  ],
});

const ProductList = React.lazy(() =>
  loadRemote("products/ProductList").then((m) => ({ default: m.default || m }))
);

// Build with @module-federation/enhanced (webpack/rspack/vite)
// Manifest enables dynamic remotes, versioning, and better DX vs classic string remotes
// Shared deps still configured in the bundler plugin
