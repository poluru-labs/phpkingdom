// Webpack runtime / MF 2.0 APIs

// Classic container API after remoteEntry loads:
await __webpack_init_sharing__("default");
const container = window.products;
await container.init(__webpack_share_scopes__.default);
const factory = await container.get("./ProductList");
const Module = factory();

// MF 2.0 enhanced runtime:
import { init, loadRemote, registerRemotes } from "@module-federation/enhanced/runtime";
init({ name: "shell", remotes: [] });
registerRemotes([{ name: "products", entry: "http://localhost:3002/mf-manifest.json" }]);
const mod = await loadRemote("products/ProductList");
