// Remote Manifest — Example 24
// Focus: native federation | Scenario: checkout remote

// Central JSON registry of remote entries + versions

// public/remotes.manifest.json
// {
//   "products": { "url": "https://cdn.example.com/products/1.4.0/remoteEntry.js", "version": "1.4.0" },
//   "cart": { "url": "https://cdn.example.com/cart/2.1.0/remoteEntry.js", "version": "2.1.0" }
// }

async function fetchManifest() {
  const res = await fetch("/remotes.manifest.json", { cache: "no-cache" });
  return res.json();
}

async function entryFor(name) {
  const manifest = await fetchManifest();
  return manifest[name].url;
}

// Host bootstrap uses entryFor("products") before loading the container
// CI publishes new version + updates manifest atomically
