// Register remotes after startup (feature flags / user entitlements)

async function loadDynamicRemote(name, entryUrl) {
  await __webpack_init_sharing__("default");
  await new Promise((resolve, reject) => {
    const s = document.createElement("script");
    s.src = entryUrl;
    s.onload = resolve;
    s.onerror = reject;
    document.head.appendChild(s);
  });
  const container = window[name];
  await container.init(__webpack_share_scopes__.default);
  return container;
}

// Usage:
const products = await loadDynamicRemote(
  "products",
  "https://cdn.example.com/products/remoteEntry.js"
);
const factory = await products.get("./ProductList");
const Module = factory();
