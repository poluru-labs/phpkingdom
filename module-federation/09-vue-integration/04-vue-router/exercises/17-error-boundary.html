Module Federation

// Vue Router Integration — Exercise 17
// Focus: error boundary | Scenario: multi-team platform

// Vue host owns createRouter; remotes export route records

// host
import { createRouter, createWebHistory } from "vue-router";
const routes = [
  {
    path: "/products/:pathMatch(.*)*",
    component: () => import("products/ProductsApp"),
  },
];
export const router = createRouter({ history: createWebHistory(), routes });

// remote ProductsApp.vue can use useRoute() against the host router
// Share vue-router as singleton:

shared: {
  vue: { singleton: true, requiredVersion: "^3.0.0" },
  "vue-router": { singleton: true, requiredVersion: "^4.0.0" },
}
