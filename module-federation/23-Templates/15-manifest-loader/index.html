// remotes.manifest.json
// { "products": { "url": "https://cdn.example.com/products/1.4.2/remoteEntry.js", "version": "1.4.2" } }

async function loadFromManifest(name, modulePath) {
  const manifest = await fetch("/remotes.manifest.json", { cache: "no-cache" }).then((r) => r.json());
  const url = manifest[name].url;
  await new Promise((res, rej) => {
    const s = document.createElement("script");
    s.src = url;
    s.onload = res;
    s.onerror = rej;
    document.head.appendChild(s);
  });
  await __webpack_init_sharing__("default");
  await window[name].init(__webpack_share_scopes__.default);
  return (await window[name].get(modulePath))();
}
