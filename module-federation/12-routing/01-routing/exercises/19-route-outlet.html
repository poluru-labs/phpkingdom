// Federated routing: one history, many remotes

import { BrowserRouter, Routes, Route } from "react-router-dom";

const Products = React.lazy(() => import("products/App"));
const Account = React.lazy(() => import("account/App"));

export default function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route path="/products/*" element={<Products />} />
        <Route path="/account/*" element={<Account />} />
      </Routes>
    </BrowserRouter>
  );
}

// Share react-router-dom as singleton across host + remotes

Practice Notes

Checklist for Federated Routing
- Practice: route outlet
- Scenario: mobile web shell
- Share framework runtimes as singletons with requiredVersion
- Load remotes lazily; wrap with error boundaries / fallbacks
- Deploy remotes independently; version remoteEntry URLs carefully
- Align CORS and CSP for cross-origin remoteEntry.js

Useful commands:
  npm install -D webpack webpack-cli webpack-dev-server
  # run host + remote on different ports
  npx webpack serve --config webpack.host.js --port 3000
  npx webpack serve --config webpack.remote.js --port 3002
