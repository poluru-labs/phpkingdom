// Framework-agnostic remotes: mount/unmount + events only

// exposes: { "./mount": "./src/mount.js" }
export function mount(el, props, { onEvent } = {}) {
  el.textContent = `Remote widget: ${props.title}`;
  const btn = document.createElement("button");
  btn.textContent = "Add";
  btn.onclick = () => onEvent?.({ type: "cart:add", detail: { id: props.id } });
  el.appendChild(btn);
  return () => { el.innerHTML = ""; };
}

// Host (any framework):
const { mount } = await import("widget/mount");
const api = mount(slotEl, { title: "Mug", id: "sku-1" }, {
  onEvent: (e) => window.dispatchEvent(new CustomEvent(e.type, { detail: e.detail })),
});
