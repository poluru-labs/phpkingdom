// Cross-framework: wrap remotes as Web Components or iframes

// React remote exposes a mount function
// exposes: { "./mount": "./src/mount" }
export function mount(el, props) {
  const root = ReactDOM.createRoot(el);
  root.render(<ProductList {...props} />);
  return () => root.unmount();
}

// Angular/Vue host:
const { mount } = await import("products/mount");
const el = document.getElementById("slot");
const unmount = mount(el, { sku: "SKU-1" });

Practice Notes

Checklist for Cross-Framework Federation
- Practice: redux share
- Scenario: admin portal
- Share framework runtimes as singletons with requiredVersion
- Load remotes lazily; wrap with error boundaries / fallbacks
- Deploy remotes independently; version remoteEntry URLs carefully
- Align CORS and CSP for cross-origin remoteEntry.js
