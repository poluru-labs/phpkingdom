// RxJS subjects as a cross-app event stream (Angular-friendly)

// packages/streams/cart$.ts
import { Subject } from "rxjs";
export const cartAdd$ = new Subject();

// shared: { rxjs: { singleton: true }, "@acme/streams": { singleton: true } }

// Remote:
import { cartAdd$ } from "@acme/streams";
cartAdd$.next({ id: "sku-1" });

// Host:
cartAdd$.subscribe((item) => this.cartService.add(item));
