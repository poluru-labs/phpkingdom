// NestJS - Basics — Example 06
// Focus: dto validation | Scenario: billing microservice

// Core building blocks: Module → Controller → Provider (Service)
import { Module, Controller, Get, Injectable } from "@nestjs/common";

@Injectable()
class HelloService {
  greet() {
    return "Hello Nest";
  }
}

@Controller("hello")
class HelloController {
  constructor(private readonly hello: HelloService) {}

  @Get()
  get() {
    return { message: this.hello.greet() };
  }
}

@Module({
  controllers: [HelloController],
  providers: [HelloService],
})
export class HelloModule {}
