import { NestFactory } from "@nestjs/core";
import { MicroserviceOptions, Transport, EventPattern, Payload } from "@nestjs/microservices";
import { Controller, Module } from "@nestjs/common";

@Controller()
class KafkaController {
  @EventPattern("order.created")
  handle(@Payload() message: { id: string; total: number }) {
    console.log("order.created", message);
  }
}

@Module({ controllers: [KafkaController] })
class AppModule {}

async function bootstrap() {
  const app = await NestFactory.createMicroservice<MicroserviceOptions>(AppModule, {
    transport: Transport.KAFKA,
    options: {
      client: { brokers: ["localhost:9092"], clientId: "orders" },
      consumer: { groupId: "orders-consumer" },
    },
  });
  await app.listen();
}
bootstrap();
