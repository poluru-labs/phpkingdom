import { Prop, Schema, SchemaFactory, InjectModel, MongooseModule } from "@nestjs/mongoose";
import { HydratedDocument, Model } from "mongoose";
import { Injectable, Module } from "@nestjs/common";

export type UserDocument = HydratedDocument<User>;

@Schema({ timestamps: true })
export class User {
  @Prop({ required: true }) name: string;
  @Prop({ required: true, unique: true }) email: string;
}
export const UserSchema = SchemaFactory.createForClass(User);

@Injectable()
export class UsersService {
  constructor(@InjectModel(User.name) private model: Model<UserDocument>) {}
  findAll() { return this.model.find().exec(); }
  create(data: { name: string; email: string }) { return this.model.create(data); }
}

@Module({
  imports: [
    MongooseModule.forRoot(process.env.MONGO_URL || "mongodb://localhost/app"),
    MongooseModule.forFeature([{ name: User.name, schema: UserSchema }]),
  ],
  providers: [UsersService],
})
export class UsersModule {}
