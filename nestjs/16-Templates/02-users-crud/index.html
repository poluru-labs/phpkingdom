import { Controller, Get, Post, Put, Delete, Body, Param, ParseIntPipe, Injectable, Module } from "@nestjs/common";

@Injectable()
export class UsersService {
  private users = [{ id: 1, name: "Ada", email: "ada@example.com" }];
  findAll() { return this.users; }
  findOne(id: number) { return this.users.find((u) => u.id === id); }
  create(dto: { name: string; email: string }) {
    const user = { id: this.users.length + 1, ...dto };
    this.users.push(user);
    return user;
  }
  update(id: number, dto: Partial<{ name: string; email: string }>) {
    const user = this.findOne(id);
    if (!user) return null;
    Object.assign(user, dto);
    return user;
  }
  remove(id: number) {
    this.users = this.users.filter((u) => u.id !== id);
  }
}

@Controller("users")
export class UsersController {
  constructor(private users: UsersService) {}
  @Get() findAll() { return this.users.findAll(); }
  @Get(":id") findOne(@Param("id", ParseIntPipe) id: number) { return this.users.findOne(id); }
  @Post() create(@Body() dto: { name: string; email: string }) { return this.users.create(dto); }
  @Put(":id") update(@Param("id", ParseIntPipe) id: number, @Body() dto: Partial<{ name: string; email: string }>) {
    return this.users.update(id, dto);
  }
  @Delete(":id") remove(@Param("id", ParseIntPipe) id: number) { this.users.remove(id); return { deleted: true }; }
}

@Module({ controllers: [UsersController], providers: [UsersService] })
export class UsersModule {}
