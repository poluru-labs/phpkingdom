import { NestFactory } from "@nestjs/core";
import { MicroserviceOptions, Transport, MessagePattern, Payload } from "@nestjs/microservices";
import { Controller, Module } from "@nestjs/common";

@Controller()
class RmqController {
  @MessagePattern("notifications.send")
  send(@Payload() data: { to: string; body: string }) {
    return { queued: true, ...data };
  }
}

@Module({ controllers: [RmqController] })
class AppModule {}

async function bootstrap() {
  const app = await NestFactory.createMicroservice<MicroserviceOptions>(AppModule, {
    transport: Transport.RMQ,
    options: {
      urls: [process.env.RABBITMQ_URL || "amqp://localhost:5672"],
      queue: "notifications_queue",
      queueOptions: { durable: true },
    },
  });
  await app.listen();
}
bootstrap();
