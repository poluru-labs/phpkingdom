NestJS

// GraphQL Subscriptions — Exercise 11
// Focus: class validator | Scenario: REST API backend

import { Resolver, Subscription, Mutation, Args } from "@nestjs/graphql";
import { PubSub } from "graphql-subscriptions";

const pubSub = new PubSub();

@Resolver()
export class OrdersResolver {
  @Mutation(() => Boolean)
  async orderShipped(@Args("orderId") orderId: string) {
    await pubSub.publish("orderShipped", { orderShipped: { orderId } });
    return true;
  }

  @Subscription(() => Object, {
    // filter optional per-user
  })
  orderShipped() {
    return pubSub.asyncIterator("orderShipped");
  }
}
// GraphQLModule: installSubscriptionHandlers / subscriptions: { "graphql-ws": true }
