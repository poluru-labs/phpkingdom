// CRUD API — Exercise 13
// Focus: canActivate | Scenario: e-commerce API

import {
  Controller,
  Get,
  Post,
  Put,
  Delete,
  Body,
  Param,
  ParseIntPipe,
} from "@nestjs/common";

@Controller("products")
export class ProductsController {
  @Get() findAll() { return []; }
  @Get(":id") findOne(@Param("id", ParseIntPipe) id: number) { return { id }; }
  @Post() create(@Body() dto: { name: string; price: number }) { return dto; }
  @Put(":id") update(
    @Param("id", ParseIntPipe) id: number,
    @Body() dto: { name?: string; price?: number },
  ) { return { id, ...dto }; }
  @Delete(":id") remove(@Param("id", ParseIntPipe) id: number) {
    return { id, deleted: true };
  }
}
// nest g resource products — generates full CRUD scaffold

Practice Notes

Checklist for CRUD API
- Practice: canActivate
- Scenario: e-commerce API
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory

Useful commands:
  npm i -g @nestjs/cli
  nest new my-api
  nest g resource users
  nest start --watch
  npm run test
  npm run test:e2e
