// DTO — Exercise 20
// Focus: jwt guard | Scenario: B2B API

import { IsEmail, IsOptional, IsString, MinLength } from "class-validator";
import { PartialType } from "@nestjs/mapped-types";

export class CreateUserDto {
  @IsString()
  @MinLength(2)
  name: string;

  @IsEmail()
  email: string;

  @IsOptional()
  @IsString()
  role?: string;
}

export class UpdateUserDto extends PartialType(CreateUserDto) {}
// DTOs define the API contract; keep entities separate from transport shapes

Practice Notes

Checklist for DTO
- Practice: jwt guard
- Scenario: B2B API
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory

Useful commands:
  npm i -g @nestjs/cli
  nest new my-api
  nest g resource users
  nest start --watch
  npm run test
  npm run test:e2e
