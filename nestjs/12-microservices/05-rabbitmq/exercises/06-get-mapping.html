NestJS

// RabbitMQ with NestJS — Exercise 06
// Focus: get mapping | Scenario: billing microservice

import { NestFactory } from "@nestjs/core";
import { MicroserviceOptions, Transport } from "@nestjs/microservices";
import { Controller } from "@nestjs/common";
import { MessagePattern, Payload } from "@nestjs/microservices";
import { AppModule } from "./app.module";

@Controller()
export class RmqController {
  @MessagePattern("notifications.send")
  send(@Payload() data: { to: string; body: string }) {
    return { queued: true, ...data };
  }
}

async function bootstrap() {
  const app = await NestFactory.createMicroservice<MicroserviceOptions>(
    AppModule,
    {
      transport: Transport.RMQ,
      options: {
        urls: [process.env.RABBITMQ_URL || "amqp://localhost:5672"],
        queue: "notifications_queue",
        queueOptions: { durable: true },
      },
    },
  );
  await app.listen();
}

Practice Notes

Checklist for RabbitMQ with NestJS
- Practice: get mapping
- Scenario: billing microservice
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory
