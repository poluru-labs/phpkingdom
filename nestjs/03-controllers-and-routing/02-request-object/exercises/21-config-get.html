Request Object

Section: Controllers and Routing

Focus: Config Get

Scenario: REST API backend

Generate the feature with Nest CLI, implement the focus, and verify with HTTP requests.

NestJS

// Request Object — Exercise 21
// Focus: config get | Scenario: REST API backend

import { Controller, Get, Req, Headers, Ip } from "@nestjs/common";
import type { Request } from "express";

@Controller("debug")
export class DebugController {
  @Get("whoami")
  whoami(
    @Req() req: Request,
    @Headers("user-agent") ua: string,
    @Ip() ip: string,
  ) {
    return {
      method: req.method,
      url: req.url,
      ua,
      ip,
      // Prefer @Param/@Query/@Body over raw req when possible
    };
  }
}
