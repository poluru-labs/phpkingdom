import { Module } from "@nestjs/common";
import { UsersController } from "./users.controller";
import { UsersService } from "./users.service";

@Module({
  imports: [],              // other modules
  controllers: [UsersController],
  providers: [UsersService],
  exports: [UsersService],  // make available to importers
})
export class UsersModule {}

// AppModule imports UsersModule — feature modules keep boundaries clear

Practice Notes

Checklist for NestJS Modules
- Practice: kafka consumer
- Scenario: billing microservice
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory

Useful commands:
  npm i -g @nestjs/cli
  nest new my-api
  nest g resource users
  nest start --watch
  npm run test
  npm run test:e2e
