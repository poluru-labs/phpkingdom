# Dockerfile (multi-stage)
# FROM node:20-alpine AS build
# WORKDIR /app
# COPY package*.json ./
# RUN npm ci
# COPY . .
# RUN npm run build
#
# FROM node:20-alpine
# WORKDIR /app
# ENV NODE_ENV=production
# COPY --from=build /app/dist ./dist
# COPY --from=build /app/node_modules ./node_modules
# COPY package*.json ./
# EXPOSE 3000
# CMD ["node", "dist/main.js"]

import { NestFactory } from "@nestjs/core";
import { AppModule } from "./app.module";
import { ValidationPipe } from "@nestjs/common";

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  app.useGlobalPipes(new ValidationPipe({ whitelist: true }));
  app.enableShutdownHooks();
  await app.listen(process.env.PORT || 3000);
}
bootstrap();

Practice Notes

Checklist for NestJS Deployment
- Practice: starter api
- Scenario: REST API backend
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory

Useful commands:
  npm i -g @nestjs/cli
  nest new my-api
  nest g resource users
  nest start --watch
  npm run test
  npm run test:e2e
