// Lifecycle Events — Template 13

import {
  Injectable,
  OnModuleInit,
  OnModuleDestroy,
  OnApplicationBootstrap,
} from "@nestjs/common";

@Injectable()
export class CacheService
  implements OnModuleInit, OnModuleDestroy, OnApplicationBootstrap
{
  onModuleInit() {
    console.log("CacheService: module init");
  }

  onApplicationBootstrap() {
    console.log("CacheService: app bootstrap complete");
  }

  onModuleDestroy() {
    console.log("CacheService: closing connections");
  }
}
