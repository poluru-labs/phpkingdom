NestJS

// Custom Pipes — Example 07
// Focus: validation pipe | Scenario: notification service

import { PipeTransform, Injectable, BadRequestException } from "@nestjs/common";

@Injectable()
export class ParseSlugPipe implements PipeTransform<string, string> {
  transform(value: string): string {
    if (!/^[a-z0-9-]+$/.test(value)) {
      throw new BadRequestException("Invalid slug");
    }
    return value;
  }
}

Practice Notes

Checklist for Custom Pipes
- Practice: validation pipe
- Scenario: notification service
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory
