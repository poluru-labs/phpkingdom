// Environment Variables — Exercise 11
// Focus: class validator | Scenario: REST API backend

// .env
// PORT=3000
// DATABASE_URL=postgres://user:pass@localhost:5432/app

import { Module } from "@nestjs/common";
import { ConfigModule, ConfigService } from "@nestjs/config";

@Module({
  imports: [ConfigModule.forRoot({ isGlobal: true })],
})
export class AppModule {}

// usage
constructor(private config: ConfigService) {}
const port = this.config.get<number>("PORT", 3000);
const db = this.config.getOrThrow<string>("DATABASE_URL");
// Never commit secrets; use .env.example for docs

Practice Notes

Checklist for Environment Variables
- Practice: class validator
- Scenario: REST API backend
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory
