NestJS

// Custom Providers — Exercise 09
// Focus: query decorator | Scenario: IoT gateway

import { Module, Inject } from "@nestjs/common";

export const HASH_ROUNDS = "HASH_ROUNDS";

@Module({
  providers: [
    { provide: HASH_ROUNDS, useValue: 12 },
    {
      provide: "CONNECTION",
      useFactory: async (rounds: number) => {
        return { rounds, ready: true };
      },
      inject: [HASH_ROUNDS],
    },
  ],
  exports: ["CONNECTION"],
})
export class CryptoModule {}

Practice Notes

Checklist for Custom Providers

- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory
