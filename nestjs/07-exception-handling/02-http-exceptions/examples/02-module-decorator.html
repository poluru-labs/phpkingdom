NestJS

// HTTP Exceptions — Example 02
// Focus: module decorator | Scenario: SaaS platform

import {
  Controller,
  Get,
  Param,
  ParseIntPipe,
  NotFoundException,
  BadRequestException,
  ForbiddenException,
} from "@nestjs/common";

@Controller("users")
export class UsersController {
  private users = [{ id: 1, name: "Ada" }];

  @Get(":id")
  findOne(@Param("id", ParseIntPipe) id: number) {
    if (id < 1) throw new BadRequestException("id must be positive");
    const user = this.users.find((u) => u.id === id);
    if (!user) throw new NotFoundException(`User ${id} not found`);
    if (user.name === "banned") throw new ForbiddenException();
    return user;
  }
}
