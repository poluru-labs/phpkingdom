// Custom Exceptions — Template 13

import { HttpException, HttpStatus } from "@nestjs/common";

export class UserAlreadyExistsException extends HttpException {
  constructor(email: string) {
    super(
      { statusCode: HttpStatus.CONFLICT, error: "Conflict", email },
      HttpStatus.CONFLICT,
    );
  }
}

// throw new UserAlreadyExistsException("ada@example.com");
// → 409 { statusCode: 409, error: "Conflict", email: "..." }
