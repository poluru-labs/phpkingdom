// NestJS Authentication — Example 22
// Focus: microservice pattern | Scenario: SaaS platform

import { Injectable, UnauthorizedException, Module } from "@nestjs/common";
import { JwtModule, JwtService } from "@nestjs/jwt";

@Injectable()
export class AuthService {
  constructor(private jwt: JwtService) {}

  async login(email: string, password: string) {
    // verify credentials against DB...
    if (password !== "secret") throw new UnauthorizedException();
    const accessToken = await this.jwt.signAsync({ sub: 1, email });
    return { accessToken };
  }
}

@Module({
  imports: [
    JwtModule.register({
      global: true,
      secret: process.env.JWT_SECRET || "dev-secret",
      signOptions: { expiresIn: "1h" },
    }),
  ],
  providers: [AuthService],
  exports: [AuthService],
})
export class AuthModule {}

Practice Notes

Checklist for NestJS Authentication
- Practice: microservice pattern
- Scenario: SaaS platform
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory
