Refresh Tokens

Section: Authentication

Focus: Testing Module

Scenario: e-commerce API

Generate the feature with Nest CLI, implement the focus, and verify with HTTP requests.

NestJS

// Refresh Tokens — Exercise 23
// Focus: testing module | Scenario: e-commerce API

import { Injectable } from "@nestjs/common";
import { JwtService } from "@nestjs/jwt";
import { createHash, randomBytes } from "node:crypto";

@Injectable()
export class TokensService {
  constructor(private jwt: JwtService) {}

  issueAccess(userId: number) {
    return this.jwt.sign(
      { sub: userId },
      { secret: process.env.JWT_SECRET, expiresIn: "15m" },
    );
  }

  issueRefresh() {
    const raw = randomBytes(48).toString("hex");
    const hash = createHash("sha256").update(raw).digest("hex");
    // store hash in DB; return raw to client once
    return { raw, hash };
  }
}
// Rotate refresh tokens on each use; revoke on logout
