NestJS

// NestJS Guards — Example 06
// Focus: dto validation | Scenario: billing microservice

import { CanActivate, ExecutionContext, Injectable } from "@nestjs/common";

@Injectable()
export class AuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const req = context.switchToHttp().getRequest();
    const auth = req.headers.authorization as string | undefined;
    return Boolean(auth?.startsWith("Bearer "));
  }
}

// @UseGuards(AuthGuard) on controller or handler
// app.useGlobalGuards(new AuthGuard()) for all routes
