Auth Guards

Section: Middleware and Guards

Focus: Nest New App

Scenario: REST API backend

Generate the feature with Nest CLI, implement the focus, and verify with HTTP requests.

NestJS

// Auth Guards — Exercise 01
// Focus: nest new app | Scenario: REST API backend

import { AuthGuard } from "@nestjs/passport";
import { Injectable, UseGuards, Controller, Get } from "@nestjs/common";

@Injectable()
export class JwtAuthGuard extends AuthGuard("jwt") {}

@Controller("profile")
export class ProfileController {
  @UseGuards(JwtAuthGuard)
  @Get()
  getProfile() {
    return { email: "ada@example.com" };
  }
}
// JwtAuthGuard → Passport JWT strategy → request.user
