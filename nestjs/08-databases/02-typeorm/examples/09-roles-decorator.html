// TypeORM — Example 09
// Focus: roles decorator | Scenario: IoT gateway

import { Entity, Column, PrimaryGeneratedColumn, Repository } from "typeorm";
import { InjectRepository, TypeOrmModule } from "@nestjs/typeorm";
import { Injectable, Module } from "@nestjs/common";

@Entity("users")
export class User {
  @PrimaryGeneratedColumn()
  id: number;

  @Column()
  name: string;

  @Column({ unique: true })
  email: string;
}

@Injectable()
export class UsersService {
  constructor(
    @InjectRepository(User) private readonly users: Repository<User>,
  ) {}

  findAll() {
    return this.users.find();
  }

  findOne(id: number) {
    return this.users.findOneBy({ id });
  }
}

@Module({
  imports: [TypeOrmModule.forFeature([User])],
  providers: [UsersService],
  exports: [UsersService],
})
export class UsersModule {}

Practice Notes

Checklist for TypeORM
- Practice: roles decorator
- Scenario: IoT gateway
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory
