// Mongoose — Exercise 11
// Focus: class validator | Scenario: REST API backend

import { Prop, Schema, SchemaFactory, InjectModel, MongooseModule } from "@nestjs/mongoose";
import { HydratedDocument, Model } from "mongoose";
import { Injectable, Module } from "@nestjs/common";

export type UserDocument = HydratedDocument<User>;

@Schema({ timestamps: true })
export class User {
  @Prop({ required: true })
  name: string;

  @Prop({ required: true, unique: true })
  email: string;
}

export const UserSchema = SchemaFactory.createForClass(User);

@Injectable()
export class UsersService {
  constructor(@InjectModel(User.name) private userModel: Model<UserDocument>) {}

  findAll() {
    return this.userModel.find().exec();
  }
}

@Module({
  imports: [MongooseModule.forFeature([{ name: User.name, schema: UserSchema }])],
  providers: [UsersService],
})
export class UsersModule {}

Checklist for Mongoose
- Practice: class validator
- Scenario: REST API backend
- Structure by feature modules: module + controller + service + DTOs
- Prefer constructor injection; keep providers @Injectable()
- Validate input with class-validator + ValidationPipe (whitelist)
- Use guards for auth/roles, pipes for transform/validate, filters for errors
- Test with TestingModule; e2e with supertest against NestFactory
