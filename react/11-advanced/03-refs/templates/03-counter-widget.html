// React Refs — Template 03
// Focus: counter widget | Scenario: todo app

import { useRef } from "react";

export default function VideoPlayer() {
  const videoRef = useRef(null);
  return (
    <>
      <video ref={videoRef} src="/demo.mp4" />
      <button onClick={() => videoRef.current?.play()}>Play</button>
    </>
  );
}
// Prefer refs for DOM access; avoid reading/writing during render

Practice Notes

Checklist for React Refs
- Practice: counter widget
- Scenario: todo app
- Prefer function components + Hooks over class components
- Keep state local; lift only when siblings need to share
- Keys must be stable and unique among siblings
- Effects: declare dependencies; clean up subscriptions/timers
- For shared UI state: Context / Zustand / Redux; for server state: React Query
