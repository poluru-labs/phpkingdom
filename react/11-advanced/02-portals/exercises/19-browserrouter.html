// React Portals — Exercise 19
// Focus: BrowserRouter | Scenario: learning app

import { useState } from "react";
import { createPortal } from "react-dom";

export default function ModalDemo() {
  const [open, setOpen] = useState(false);
  return (
    <>
      <button onClick={() => setOpen(true)}>Open</button>
      {open &&
        createPortal(
          <div role="dialog" className="modal">
            <p>Modal</p>
            <button onClick={() => setOpen(false)}>Close</button>
          </div>,
          document.body
        )}
    </>
  );
}

Practice Notes

Checklist for React Portals
- Practice: BrowserRouter
- Scenario: learning app
- Prefer function components + Hooks over class components
- Keep state local; lift only when siblings need to share
- Keys must be stable and unique among siblings
- Effects: declare dependencies; clean up subscriptions/timers
- For shared UI state: Context / Zustand / Redux; for server state: React Query
