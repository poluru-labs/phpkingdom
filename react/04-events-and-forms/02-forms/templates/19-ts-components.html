// React Forms — Template 19

import { useState } from "react";

export default function Login() {
  const [form, setForm] = useState({ email: "", password: "" });
  function onChange(e) {
    const { name, value } = e.target;
    setForm((f) => ({ ...f, [name]: value }));
  }
  function onSubmit(e) {
    e.preventDefault();
    console.log(form);
  }
  return (
    <form onSubmit={onSubmit}>
      <input name="email" value={form.email} onChange={onChange} />
      <input name="password" type="password" value={form.password} onChange={onChange} />
      <button type="submit">Sign in</button>
    </form>
  );
}

Practice Notes

Checklist for React Forms
- Practice: ts components
- Scenario: learning app
- Prefer function components + Hooks over class components
- Keep state local; lift only when siblings need to share
- Keys must be stable and unique among siblings
- Effects: declare dependencies; clean up subscriptions/timers
- For shared UI state: Context / Zustand / Redux; for server state: React Query
