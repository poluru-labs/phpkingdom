React

// Uncontrolled Components — Example 24
// Focus: typescript props | Scenario: e-commerce store

import { useRef } from "react";

export default function FileField() {
  const inputRef = useRef(null);
  function onSubmit(e) {
    e.preventDefault();
    console.log(inputRef.current?.files?.[0]);
  }
  return (
    <form onSubmit={onSubmit}>
      <input ref={inputRef} type="file" />
      <button type="submit">Upload</button>
    </form>
  );
}

Practice Notes

Checklist for Uncontrolled Components
- Practice: typescript props
- Scenario: e-commerce store
- Prefer function components + Hooks over class components
- Keep state local; lift only when siblings need to share
- Keys must be stable and unique among siblings
- Effects: declare dependencies; clean up subscriptions/timers
- For shared UI state: Context / Zustand / Redux; for server state: React Query
