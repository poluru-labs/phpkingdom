// Route Parameters — Template 07
// Focus: tabs children | Scenario: portfolio site

import { useParams, Link } from "react-router-dom";

export default function PostPage() {
  const { id } = useParams();
  return (
    <>
      <h1>Post {id}</h1>
      <Link to="/">Home</Link>
    </>
  );
}
// route: { path: "/posts/:id", element: <PostPage /> }
