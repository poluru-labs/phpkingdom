useCallback Hook

Section: Advanced Hooks

Study the React component/hooks pattern and adapt it to your UI.

React

// useCallback Hook — Example 17
// Focus: react router | Scenario: portfolio site

import { memo, useCallback, useState } from "react";

const Item = memo(function Item({ id, onSelect }) {
  return <button onClick={() => onSelect(id)}>Select {id}</button>;
});

export default function List({ ids }) {
  const [selected, setSelected] = useState(null);
  const onSelect = useCallback((id) => setSelected(id), []);
  return (
    <>
      <p>Selected: {selected}</p>
      {ids.map((id) => <Item key={id} id={id} onSelect={onSelect} />)}
    </>
  );
}
