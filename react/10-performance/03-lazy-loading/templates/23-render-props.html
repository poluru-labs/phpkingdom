// Lazy Loading — Template 23
// Focus: render props | Scenario: todo app

import { lazy, Suspense } from "react";

const Dashboard = lazy(() => import("./Dashboard"));

export default function App() {
  return (
    <Suspense fallback={<p>Loading page…</p>}>
      <Dashboard />
    </Suspense>
  );
}

Practice Notes

Checklist for Lazy Loading
- Practice: render props
- Scenario: todo app
- Prefer function components + Hooks over class components
- Keep state local; lift only when siblings need to share
- Keys must be stable and unique among siblings
- Effects: declare dependencies; clean up subscriptions/timers
- For shared UI state: Context / Zustand / Redux; for server state: React Query
