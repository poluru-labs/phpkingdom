// TypeScript State — Exercise 20
// Focus: useParams | Scenario: B2B portal

import { useState } from "react";

type User = { id: string; name: string };

export function UserPicker() {
  const [user, setUser] = useState<User | null>(null);
  return (
    <button onClick={() => setUser({ id: "1", name: "Ada" })}>
      {user?.name ?? "Select user"}
    </button>
  );
}

Practice Notes

Checklist for TypeScript State
- Practice: useParams
- Scenario: B2B portal
- Prefer function components + Hooks over class components
- Keep state local; lift only when siblings need to share
- Keys must be stable and unique among siblings
- Effects: declare dependencies; clean up subscriptions/timers
- For shared UI state: Context / Zustand / Redux; for server state: React Query
