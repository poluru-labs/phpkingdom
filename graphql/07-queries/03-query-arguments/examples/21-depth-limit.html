GraphQL (Node.js)

// Query Arguments — Example 21
// Focus: depth limit | Scenario: blog API

const typeDefs = `#graphql
  type Query { posts(limit: Int = 10, offset: Int = 0, authorId: ID): [Post!]! }
`;
const resolvers = {
  Query: {
    posts: (_p, { limit, offset, authorId }, { db }) =>
      db.posts.findMany({ take: limit, skip: offset, where: authorId ? { authorId } : {} }),
  },
};
