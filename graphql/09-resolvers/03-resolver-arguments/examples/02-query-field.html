// Resolver Arguments — Example 02
// Focus: query field | Scenario: e-commerce catalog

const resolvers = {
  Query: {
    search: (_parent, args) => {
      const { q, limit = 10 } = args;
      return db.filter((row) => row.name.includes(q)).slice(0, limit);
    },
  },
};
