const resolvers = {
  Mutation: {
    createUser: (_p, { input }) => {
      assertEmail(input.email);
      if (!input.name?.trim()) throw new GraphQLError("name required", { extensions: { code: "BAD_USER_INPUT" } });
      return db.create(input);
    },
  },
};
