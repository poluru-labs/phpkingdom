GraphQL (Node.js)

// Custom GraphQL Errors — Example 14
// Focus: yoga setup | Scenario: chat service

class NotFoundError extends GraphQLError {
  constructor(entity, id) {
    super(`${entity} ${id} not found`, { extensions: { code: "NOT_FOUND", entity, id } });
  }
}
