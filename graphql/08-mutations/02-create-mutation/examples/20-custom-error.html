GraphQL (Node.js)

// Create Mutation — Example 20
// Focus: custom error | Scenario: analytics API

const typeDefs = `#graphql
  input CreateUserInput { email: String! name: String }
  type Mutation { createUser(input: CreateUserInput!): User! }
`;
const resolvers = {
  Mutation: {
    createUser: (_p, { input }, ctx) => {
      const id = String(ctx.db.size + 1);
      const user = { id, ...input };
      ctx.db.set(id, user);
      return user;
    },
  },
};
