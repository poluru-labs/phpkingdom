// OAuth Authentication — Exercise 19

# Exchange OAuth code → access token → load profile → issue app JWT
const resolvers = {
  Mutation: {
    loginWithGoogle: async (_p, { code }, ctx) => {
      const profile = await exchangeGoogleCode(code);
      const user = await upsertUser(profile);
      return { token: signJwt(user), user };
    },
  },
};
