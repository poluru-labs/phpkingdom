// Dynamic Filters — Example 12
// Focus: interface type | Scenario: e-commerce catalog

function buildWhere(filter = {}) {
  const where = {};
  if (filter.titleContains) where.title = { contains: filter.titleContains, mode: "insensitive" };
  if (filter.authorId) where.authorId = filter.authorId;
  if (filter.published != null) where.published = filter.published;
  return where;
}
