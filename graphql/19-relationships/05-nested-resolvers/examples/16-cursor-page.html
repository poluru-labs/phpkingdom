  Query: { user: (_p, { id }, { db }) => db.users.findById(id) },
  User: {
    posts: (user, _a, { db }) => db.posts.findByAuthor(user.id),
  },
  Post: {
    comments: (post, _a, { db }) => db.comments.findByPost(post.id),
  },
};
