function withLogging(resolve) {
  return async (parent, args, ctx, info) => {
    const start = Date.now();
    try {
      return await resolve(parent, args, ctx, info);
    } finally {
      console.log(info.parentType.name, info.fieldName, Date.now() - start);
    }
  };
}
