// DataLoader Batching — Exercise 04

const postsByUser = new DataLoader(async (userIds) => {
  const posts = await db.posts.findMany({ where: { authorId: { in: [...userIds] } } });
  return userIds.map((id) => posts.filter((p) => p.authorId === id));
});
