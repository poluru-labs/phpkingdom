// Cursor Pagination — Template 23
// Focus: typescript codegen | Scenario: social feed

const resolvers = {
  Query: {
    posts: async (_p, { first = 10, after }, { prisma }) => {
      const items = await prisma.post.findMany({
        take: first + 1,
        ...(after ? { cursor: { id: after }, skip: 1 } : {}),
        orderBy: { id: "asc" },
      });
      const hasNext = items.length > first;
      const nodes = hasNext ? items.slice(0, -1) : items;
      return { nodes, nextCursor: hasNext ? nodes.at(-1)?.id : null };
    },
  },
};
