async function getUser(id) {
  const res = await fetch(`/users/${id}`);
  if (res.status === 404) throw new GraphQLError("Not found", { extensions: { code: "NOT_FOUND" } });
  if (!res.ok) throw new GraphQLError("Upstream error", { extensions: { code: "BAD_GATEWAY" } });
  return res.json();
}
