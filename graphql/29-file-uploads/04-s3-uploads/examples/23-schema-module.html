GraphQL (Node.js)

// Upload Files to Amazon S3 — Example 23
// Focus: schema module | Scenario: social feed

import { S3Client, PutObjectCommand } from "@aws-sdk/client-s3";
const s3 = new S3Client({});
async function putAvatar(key, body, contentType) {
  await s3.send(new PutObjectCommand({ Bucket: process.env.S3_BUCKET, Key: key, Body: body, ContentType: contentType }));
  return `https://${process.env.S3_BUCKET}.s3.amazonaws.com/${key}`;
}
