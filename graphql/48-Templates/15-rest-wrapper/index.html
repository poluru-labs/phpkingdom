Wrap a REST users API in GraphQL

Server Scaffold

const typeDefs = `#graphql
  type User { id: ID! email: String! name: String }
  type Query { user(id: ID!): User users: [User!]! }
`;
async function restUser(id) {
  const res = await fetch(`https://api.example.com/users/${id}`);
  if (res.status === 404) return null;
  const data = await res.json();
  return { id: data.id, email: data.email, name: data.full_name };
}
const resolvers = {
  Query: {
    user: (_p, { id }) => restUser(id),
    users: async () => {
      const res = await fetch("https://api.example.com/users");
      const rows = await res.json();
      return rows.map((d) => ({ id: d.id, email: d.email, name: d.full_name }));
    },
  },
};
