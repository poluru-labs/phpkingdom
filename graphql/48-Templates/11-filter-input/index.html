const typeDefs = `#graphql
  input PostFilter {
    titleContains: String
    authorId: ID
    published: Boolean
  }
  enum PostSort { CREATED_DESC CREATED_ASC TITLE_ASC }
  type Query { posts(filter: PostFilter, sort: PostSort = CREATED_DESC): [Post!]! }
`;
function whereFrom(filter = {}) {
  const where = {};
  if (filter.titleContains) where.title = { contains: filter.titleContains, mode: "insensitive" };
  if (filter.authorId) where.authorId = filter.authorId;
  if (filter.published != null) where.published = filter.published;
  return where;
}
const orderBy = {
  CREATED_DESC: { createdAt: "desc" },
  CREATED_ASC: { createdAt: "asc" },
  TITLE_ASC: { title: "asc" },
};
const resolvers = {
  Query: {
    posts: (_p, { filter, sort }, { prisma }) =>
      prisma.post.findMany({ where: whereFrom(filter), orderBy: orderBy[sort] }),
  },
};
