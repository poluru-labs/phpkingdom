PubSub

Real-time chat with pubsub

Server Scaffold

import { PubSub } from "graphql-subscriptions";
const pubsub = new PubSub();
const typeDefs = `#graphql
  type Message { id: ID! roomId: ID! body: String! }
  type Mutation { sendMessage(roomId: ID!, body: String!): Message! }
  type Subscription { messageAdded(roomId: ID!): Message! }
  type Query { noop: Boolean }
`;
const resolvers = {
  Query: { noop: () => true },
  Mutation: {
    sendMessage: async (_p, { roomId, body }) => {
      const message = { id: crypto.randomUUID(), roomId, body };
      await pubsub.publish(`MSG_${roomId}`, { messageAdded: message });
      return message;
    },
  },
  Subscription: {
    messageAdded: {
      subscribe: (_p, { roomId }) => pubsub.asyncIterator([`MSG_${roomId}`]),
    },
  },
};
