import jwt from "jsonwebtoken";
import { GraphQLError } from "graphql";
const SECRET = process.env.JWT_SECRET || "dev-secret";
const typeDefs = `#graphql
  type User { id: ID! email: String! }
  type AuthPayload { token: String! user: User! }
  type Query { me: User }
  type Mutation { login(email: String!, password: String!): AuthPayload! }
`;
const resolvers = {
  Query: {
    me: (_p, _a, ctx) => {
      if (!ctx.user) return null;
      return { id: ctx.user.sub, email: ctx.user.email };
    },
  },
  Mutation: {
    login: (_p, { email, password }) => {
      if (password !== "secret") throw new GraphQLError("Bad credentials");
      const user = { id: "1", email };
      const token = jwt.sign({ sub: user.id, email }, SECRET, { expiresIn: "1h" });
      return { token, user };
    },
  },
};
