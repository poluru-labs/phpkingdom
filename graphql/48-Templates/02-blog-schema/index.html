Post/Author

Blog API schema with posts and authors

Server Scaffold

const typeDefs = `#graphql
  type Author { id: ID! name: String! posts: [Post!]! }
  type Post { id: ID! title: String! body: String! author: Author! }
  type Query {
    posts: [Post!]!
    post(id: ID!): Post
  }
  input CreatePostInput { title: String! body: String! authorId: ID! }
  type Mutation { createPost(input: CreatePostInput!): Post! }
`;
const authors = new Map([["1", { id: "1", name: "Ada" }]]);
const posts = new Map();
const resolvers = {
  Query: {
    posts: () => [...posts.values()],
    post: (_p, { id }) => posts.get(id) || null,
  },
  Mutation: {
    createPost: (_p, { input }) => {
      const id = String(posts.size + 1);
      const post = { id, ...input };
      posts.set(id, post);
      return post;
    },
  },
  Post: { author: (p) => authors.get(p.authorId) },
  Author: { posts: (a) => [...posts.values()].filter((p) => p.authorId === a.id) },
};
