const typeDefs = `#graphql
  type PageInfo { hasNextPage: Boolean! endCursor: String }
  type PostEdge { cursor: String! node: Post! }
  type PostConnection { edges: [PostEdge!]! pageInfo: PageInfo! }
  type Post { id: ID! title: String! }
  type Query { posts(first: Int = 10, after: String): PostConnection! }
`;
function connection(nodes, hasNextPage) {
  return {
    edges: nodes.map((node) => ({ cursor: node.id, node })),
    pageInfo: { hasNextPage, endCursor: nodes.at(-1)?.id ?? null },
  };
}
const resolvers = {
  Query: {
    posts: async (_p, { first, after }, { prisma }) => {
      const rows = await prisma.post.findMany({
        take: first + 1,
        ...(after ? { cursor: { id: after }, skip: 1 } : {}),
        orderBy: { id: "asc" },
      });
      const hasNext = rows.length > first;
      return connection(hasNext ? rows.slice(0, -1) : rows, hasNext);
    },
  },
};
