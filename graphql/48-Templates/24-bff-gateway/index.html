const typeDefs = `#graphql
  type Hero { title: String! imageUrl: String! }
  type Card { id: ID! title: String! }
  type MobileHome { hero: Hero! feed: [Card!]! unreadCount: Int! }
  type Query { mobileHome: MobileHome! }
`;
const resolvers = {
  Query: {
    mobileHome: async (_p, _a, { user, services }) => {
      const [hero, feed, unreadCount] = await Promise.all([
        services.cms.hero(),
        services.feed.forUser(user.id),
        services.notify.unread(user.id),
      ]);
      return { hero, feed, unreadCount };
    },
  },
};
