Product/Order

Shop catalog and place-order mutation

Server Scaffold

const typeDefs = `#graphql
  type Product { id: ID! sku: String! name: String! price: Float! stock: Int! }
  type OrderItem { product: Product! quantity: Int! }
  type Order { id: ID! items: [OrderItem!]! total: Float! }
  type Query { products: [Product!]! product(id: ID!): Product }
  input OrderItemInput { productId: ID! quantity: Int! }
  input PlaceOrderInput { items: [OrderItemInput!]! }
  type Mutation { placeOrder(input: PlaceOrderInput!): Order! }
`;
const catalog = [
  { id: "1", sku: "TEE-01", name: "Tee", price: 24, stock: 50 },
  { id: "2", sku: "HAT-01", name: "Hat", price: 18, stock: 20 },
];
const resolvers = {
  Query: {
    products: () => catalog,
    product: (_p, { id }) => catalog.find((p) => p.id === id) || null,
  },
  Mutation: {
    placeOrder: (_p, { input }) => {
      const items = input.items.map((i) => ({
        product: catalog.find((p) => p.id === i.productId),
        quantity: i.quantity,
      }));
      const total = items.reduce((s, i) => s + i.product.price * i.quantity, 0);
      return { id: crypto.randomUUID(), items, total };
    },
  },
};
