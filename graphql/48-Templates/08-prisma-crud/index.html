import { PrismaClient } from "@prisma/client";
const prisma = new PrismaClient();
const typeDefs = `#graphql
  type User { id: ID! email: String! name: String }
  type Query { users: [User!]! user(id: ID!): User }
  input CreateUserInput { email: String! name: String }
  input UpdateUserInput { email: String name: String }
  type Mutation {
    createUser(input: CreateUserInput!): User!
    updateUser(id: ID!, input: UpdateUserInput!): User!
    deleteUser(id: ID!): Boolean!
  }
`;
const resolvers = {
  Query: {
    users: () => prisma.user.findMany(),
    user: (_p, { id }) => prisma.user.findUnique({ where: { id } }),
  },
  Mutation: {
    createUser: (_p, { input }) => prisma.user.create({ data: input }),
    updateUser: (_p, { id, input }) => prisma.user.update({ where: { id }, data: input }),
    deleteUser: async (_p, { id }) => { await prisma.user.delete({ where: { id } }); return true; },
  },
};
# schema.prisma: model User { id String @id @default(cuid()) email String @unique name String? }
