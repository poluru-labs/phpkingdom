User/Follow

Follow graph with followers/following

Server Scaffold

const typeDefs = `#graphql
  type User {
    id: ID!
    handle: String!
    followers: [User!]!
    following: [User!]!
  }
  type Query { user(id: ID!): User users: [User!]! }
  type Mutation { follow(userId: ID!): User! }
`;
const users = new Map([
  ["1", { id: "1", handle: "ada" }],
  ["2", { id: "2", handle: "grace" }],
]);
const follows = new Set(); // "followerId>followeeId"
const resolvers = {
  Query: {
    users: () => [...users.values()],
    user: (_p, { id }) => users.get(id) || null,
  },
  Mutation: {
    follow: (_p, { userId }, { user }) => {
      follows.add(`${user.id}>${userId}`);
      return users.get(userId);
    },
  },
  User: {
    followers: (u) =>
      [...follows].filter((e) => e.endsWith(`>${u.id}`)).map((e) => users.get(e.split(">")[0])),
    following: (u) =>
      [...follows].filter((e) => e.startsWith(`${u.id}>`)).map((e) => users.get(e.split(">")[1])),
  },
};
