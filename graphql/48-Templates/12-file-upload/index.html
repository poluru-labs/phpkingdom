import { GraphQLUpload } from "graphql-upload";
import { createWriteStream } from "node:fs";
import { pipeline } from "node:stream/promises";

const typeDefs = `#graphql
  scalar Upload
  type User { id: ID! avatarUrl: String }
  type Mutation { uploadAvatar(file: Upload!): User! }
`;
const resolvers = {
  Upload: GraphQLUpload,
  Mutation: {
    uploadAvatar: async (_p, { file }, { user }) => {
      const { createReadStream, filename } = await file;
      const path = `/uploads/${user.id}-${filename}`;
      await pipeline(createReadStream(), createWriteStream(`.${path}`));
      return { id: user.id, avatarUrl: path };
    },
  },
};
