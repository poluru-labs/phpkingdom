// Resolver Caching — Exercise 22
// Focus: schema check | Scenario: e-commerce catalog

const cache = new Map();
const resolvers = {
  Query: {
    expensive: async (_p, { id }) => {
      if (cache.has(id)) return cache.get(id);
      const value = await compute(id);
      cache.set(id, value);
      return value;
    },
  },
};
