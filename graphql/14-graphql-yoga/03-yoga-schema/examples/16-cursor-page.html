Study the GraphQL schema/resolver pattern and adapt it to your API.

GraphQL (Node.js)

// Yoga Schema — Example 16
// Focus: cursor page | Scenario: BFF for mobile

import { createSchema, createYoga } from "graphql-yoga";
const schema = createSchema({
  typeDefs: `#graphql
    type Query { time: String! }
  `,
  resolvers: { Query: { time: () => new Date().toISOString() } },
});
export const yoga = createYoga({ schema });
