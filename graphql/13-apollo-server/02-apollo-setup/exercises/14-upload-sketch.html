// Apollo Server Setup — Exercise 14
// Focus: upload sketch | Scenario: chat service

import { ApolloServer } from "@apollo/server";
import { expressMiddleware } from "@as-integrations/express4";
import express from "express";
const app = express();
const server = new ApolloServer({ typeDefs, resolvers });
await server.start();
app.use("/graphql", express.json(), expressMiddleware(server));
app.listen(4000);

Practice Notes

Checklist for Apollo Server Setup
- Practice: upload sketch

- Keep resolvers thin; push logic into services
- Use DataLoader to avoid N+1 nested queries
- Put authz checks in context / field resolvers / directives
- Prefer non-null carefully; return typed GraphQLError codes
- Limit depth/complexity; deprecate before breaking changes
