Standalone Component Scaffold

@Component({
  standalone: true,
  template: `<p>{{ count() }} / {{ doubled() }}</p><button (click)="inc()">+1</button>`,
})
export class CounterComponent {
  count = signal(0);
  doubled = computed(() => this.count() * 2);
  inc() { this.count.update((n) => n + 1); }
}
