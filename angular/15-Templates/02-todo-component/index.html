Standalone Component Scaffold

@Component({
  standalone: true,
  template: `
    <input #t (keyup.enter)="add(t.value); t.value=''" />
    <ul>@for (item of todos(); track item) { <li>{{ item }}</li> }</ul>
  `,
})
export class TodoComponent {
  todos = signal<string[]>([]);
  add(v: string) { if (v.trim()) this.todos.update((xs) => [...xs, v.trim()]); }
}
