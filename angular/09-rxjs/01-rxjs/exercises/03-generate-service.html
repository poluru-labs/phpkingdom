// RxJS in Angular — Exercise 03
// Focus: generate service | Scenario: e-commerce store

import { Component, DestroyRef, inject } from "@angular/core";
import { FormControl, ReactiveFormsModule } from "@angular/forms";
import { debounceTime, distinctUntilChanged, switchMap } from "rxjs/operators";
import { takeUntilDestroyed } from "@angular/core/rxjs-interop";

@Component({
  standalone: true,
  imports: [ReactiveFormsModule],
  template: `<input [formControl]="q" />`,
})
export class SearchBox {
  q = new FormControl("", { nonNullable: true });
  private api = inject(SearchApi);
  constructor() {
    this.q.valueChanges.pipe(
      debounceTime(300),
      distinctUntilChanged(),
      switchMap((term) => this.api.search(term)),
      takeUntilDestroyed(),
    ).subscribe();
  }
}

export const TOPIC_ID = "rxjs";

Practice Notes

Checklist for RxJS in Angular
- Practice: generate service
- Scenario: e-commerce store
- Prefer standalone components + inject() for new Angular apps
- Use signals for local UI state; RxJS for async streams and HttpClient
- Reactive forms for complex validation; template-driven for simple forms
- Route with provideRouter; guard auth routes; lazy-load feature areas
- Prefer OnPush + async pipe / signals to reduce change detection work

Useful commands:
