Solution 08 — @Input / @Output

Accept a title input and emit update.

Solution

import { Component, EventEmitter, Input, Output } from '@angular/core';

@Component({
  selector: 'app-card',
  standalone: true,
  template: `<button (click)="update.emit(1)">{{ title }}</button>`,
})
export class CardComponent {
  @Input({ required: true }) title!: string;
  @Output() update = new EventEmitter<number>();
}
