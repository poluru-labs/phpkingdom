import { TestBed } from "@angular/core/testing";
import { CounterComponent } from "./counter.component";

describe("CounterComponent", () => {
  beforeEach(() => TestBed.configureTestingModule({ imports: [CounterComponent] }));
  it("renders", () => {
    const fixture = TestBed.createComponent(CounterComponent);
    fixture.detectChanges();
    expect(fixture.nativeElement.textContent).toContain("0");
  });
});

Practice Notes

Checklist for Angular Testing
- Practice: shared pipe
- Scenario: B2B portal
- Prefer standalone components + inject() for new Angular apps
- Use signals for local UI state; RxJS for async streams and HttpClient
- Reactive forms for complex validation; template-driven for simple forms
- Route with provideRouter; guard auth routes; lazy-load feature areas
- Prefer OnPush + async pipe / signals to reduce change detection work
