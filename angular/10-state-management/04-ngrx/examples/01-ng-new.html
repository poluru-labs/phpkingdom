// NgRx — Example 01
// Focus: ng new | Scenario: enterprise admin

import { createAction, createReducer, on, props, provideStore } from "@ngrx/store";

export const increment = createAction("[Counter] Increment");
export const set = createAction("[Counter] Set", props<{ value: number }>());

export const counterReducer = createReducer(
  0,
  on(increment, (s) => s + 1),
  on(set, (_s, { value }) => value),
);
// provideStore({ counter: counterReducer })

export const TOPIC_ID = "ngrx";

Practice Notes

Checklist for NgRx
- Practice: ng new
- Scenario: enterprise admin
- Prefer standalone components + inject() for new Angular apps
- Use signals for local UI state; RxJS for async streams and HttpClient
- Reactive forms for complex validation; template-driven for simple forms
- Route with provideRouter; guard auth routes; lazy-load feature areas
- Prefer OnPush + async pipe / signals to reduce change detection work
