// Linked Signal — Example 06
// Focus: ngIf ngFor | Scenario: banking UI

import { Component, linkedSignal, signal } from "@angular/core";

@Component({
  standalone: true,
  template: `
    <select [value]="selected()" (change)="selected.set($any($event.target).value)">
      @for (o of options(); track o) { <option [value]="o">{{ o }}</option> }
    </select>
  `,
})
export class LinkedSignalDemo {
  options = signal(["a", "b", "c"]);
  // Resets when options identity changes
  selected = linkedSignal(() => this.options()[0]);
}

export const TOPIC_ID = "linked-signal";
