// Defer Blocks — Template 03
// Focus: counter signals | Scenario: e-commerce store

import { Component } from "@angular/core";

@Component({
  standalone: true,
  template: `
    @defer (on viewport) {
      <app-heavy-chart />
    } @placeholder {
      <p>Loading chart…</p>
    } @loading (minimum 300ms) {
      <p>Fetching…</p>
    } @error {
      <p>Could not load chart.</p>
    }
  `,
})
export class DeferDemoComponent {}

export const TOPIC_ID = "defer-blocks";
