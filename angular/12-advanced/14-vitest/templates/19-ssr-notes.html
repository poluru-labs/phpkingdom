// vitest + @angular/build unit-test sketch
import { describe, it, expect } from "vitest";
import { CounterComponent } from "./counter.component";

describe("CounterComponent", () => {
  it("increments", () => {
    const c = new CounterComponent();
    c.increment();
    expect(c.count()).toBe(1);
  });
});
