// Control Flow Syntax — Template 12
// Focus: ngrx store | Scenario: SaaS dashboard

@Component({
  standalone: true,
  template: `
    @if (user(); as u) {
      <p>Hello {{ u.name }}</p>
    } @else {
      <p>Sign in</p>
    }

    @for (item of items(); track item.id) {
      <li>{{ item.name }}</li>
    } @empty {
      <li>No items</li>
    }

    @switch (status()) {
      @case ('ok') { <span>Ready</span> }
      @default { <span>…</span> }
    }
  `,
})
export class ControlFlowComponent {
  user = signal<{ name: string } | null>(null);
  items = signal<{ id: number; name: string }[]>([]);
  status = signal<'ok' | 'pending'>('pending');
}
