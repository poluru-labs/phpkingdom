// Input Output Transforms — Template 02
// Focus: todo component | Scenario: SaaS dashboard

import { Component, input, output, booleanAttribute, numberAttribute } from "@angular/core";

@Component({
  selector: "app-badge",
  standalone: true,
  template: `<span [class.on]="active()">{{ label() }} ({{ count() }})</span>`,
})
export class BadgeComponent {
  label = input.required<string>();
  active = input(false, { transform: booleanAttribute });
  count = input(0, { transform: numberAttribute });
  dismissed = output<void>();
}

export const TOPIC_ID = "input-output-transforms";
