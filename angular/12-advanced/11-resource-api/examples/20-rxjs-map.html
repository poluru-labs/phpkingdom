// Resource API — Example 20
// Focus: rxjs map | Scenario: B2B portal

import { Component, resource, signal } from "@angular/core";

@Component({
  standalone: true,
  template: `
    @if (userResource.isLoading()) { <p>Loading…</p> }
    @else if (userResource.error()) { <p>Error</p> }
    @else { <pre>{{ userResource.value() | json }}</pre> }
  `,
})
export class UserResourceComponent {
  userId = signal(1);
  userResource = resource({
    request: () => ({ id: this.userId() }),
    loader: ({ request }) => fetch(`/api/users/${request.id}`).then((r) => r.json()),
  });
}

export const TOPIC_ID = "resource-api";
