// HTTP Resource — Template 23

import { Component, inject } from "@angular/core";
import { httpResource } from "@angular/common/http";

@Component({
  standalone: true,
  template: `
    @if (todos.isLoading()) { Loading… }
    @else {
      <ul>
        @for (t of todos.value() ?? []; track t.id) {
          <li>{{ t.title }}</li>
        }
      </ul>
    }
  `,
})
export class TodosHttpResource {
  todos = httpResource<{ id: number; title: string }[]>(() => "/api/todos");
}

export const TOPIC_ID = "http-resource";
