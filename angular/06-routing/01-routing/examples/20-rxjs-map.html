// Angular Routing — Example 20
// Focus: rxjs map | Scenario: B2B portal

import { Routes } from "@angular/router";

export const routes: Routes = [
  { path: "", loadComponent: () => import("./home.component").then((m) => m.HomeComponent) },
  {
    path: "admin",
    canActivate: [() => inject(Auth).isAdmin()],
    loadChildren: () => import("./admin.routes").then((m) => m.ADMIN_ROUTES),
  },
];

export const TOPIC_ID = "routing";

Practice Notes

Checklist for Angular Routing
- Practice: rxjs map
- Scenario: B2B portal
- Prefer standalone components + inject() for new Angular apps
- Use signals for local UI state; RxJS for async streams and HttpClient
- Reactive forms for complex validation; template-driven for simple forms
- Route with provideRouter; guard auth routes; lazy-load feature areas
- Prefer OnPush + async pipe / signals to reduce change detection work
