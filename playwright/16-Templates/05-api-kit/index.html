Starter Spec

import { test, expect } from '@playwright/test';

test('CRUD via request fixture', async ({ request }) => {
  const created = await request.post('/api/items', { data: { name: 'Book' } });
  expect(created.status()).toBe(201);
  const { id } = await created.json();
  const got = await request.get(`/api/items/${id}`);
  expect(await got.json()).toMatchObject({ name: 'Book' });
  const del = await request.delete(`/api/items/${id}`);
  expect(del.status()).toBe(204);
});

Run

npx playwright test
