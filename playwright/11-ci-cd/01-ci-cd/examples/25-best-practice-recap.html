Spec Sample

// Playwright CI/CD — Example 25
// Focus: best practice recap | Scenario: auth portal

# CI checklist for Playwright
# 1) npm ci
# 2) npx playwright install --with-deps
# 3) npx playwright test
# 4) upload playwright-report/ and test-results/ on failure
# 5) retries + trace on-first-retry in config

import { defineConfig } from '@playwright/test';

export default defineConfig({
  forbidOnly: !!process.env.CI,
  retries: process.env.CI ? 2 : 0,
  workers: process.env.CI ? 2 : undefined,
  use: { trace: 'on-first-retry' },
});

Config Notes

// playwright.config.ts — notes for Playwright CI/CD
import { defineConfig, devices } from '@playwright/test';

export default defineConfig({
  testDir: './tests',
  fullyParallel: true,
  forbidOnly: !!process.env.CI,
  retries: process.env.CI ? 2 : 0,
  workers: process.env.CI ? 2 : undefined,
  reporter: [['html'], ['list']],
  use: {
    baseURL: 'http://localhost:3000',
    trace: 'on-first-retry',
    screenshot: 'only-on-failure',
    video: 'retain-on-failure',
  },
  projects: [
    { name: 'chromium', use: { ...devices['Desktop Chrome'] } },
    { name: 'firefox', use: { ...devices['Desktop Firefox'] } },
    { name: 'webkit', use: { ...devices['Desktop Safari'] } },
  ],
  // Focus: best practice recap for auth portal (ci-cd)
});

Notes

This example shows one way to apply Playwright CI/CD in a auth portal context.

Prefer role/label/test-id locators and rely on Playwright auto-waiting over fixed sleeps.
