Spec Sample

// Playwright Fixtures — Template 14
// Focus: docker compose | Scenario: admin panel

import { test as base, expect } from '@playwright/test';

type Fixtures = {
  todoPage: import('@playwright/test').Page;
};

export const test = base.extend<Fixtures>({
  todoPage: async ({ page }, use) => {
    await page.goto('/todos');
    await page.getByPlaceholder('What needs doing?').fill('Ship fixtures');
    await page.getByRole('button', { name: 'Add' }).click();
    await use(page);
  },
});

test('custom fixture prepares todos', async ({ todoPage }) => {
  await expect(todoPage.getByText('Ship fixtures')).toBeVisible();
});
