Spec Sample

// JWT Auth Testing — Example 05
// Focus: getByTestId | Scenario: auth portal

import { test, expect } from '@playwright/test';

test('API call with JWT bearer', async ({ request }) => {
  const res = await request.get('/api/me', {
    headers: { Authorization: `Bearer ${process.env.TOKEN}` },
  });
  expect(res.ok()).toBeTruthy();
  expect(await res.json()).toHaveProperty('email');
});

test('inject token into browser context', async ({ browser }) => {
  const context = await browser.newContext({
    extraHTTPHeaders: { Authorization: `Bearer ${process.env.TOKEN}` },
  });
  const page = await context.newPage();
  await page.goto('/account');
  await expect(page.getByTestId('email')).toBeVisible();
  await context.close();
});
