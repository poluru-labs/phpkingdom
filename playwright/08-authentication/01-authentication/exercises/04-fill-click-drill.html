Spec Sample

// Authentication Testing — Exercise 04
// Focus: fill click drill | Scenario: admin panel

import { test, expect } from '@playwright/test';

test('login sets session cookie', async ({ page, context }) => {
  await page.goto('/login');
  await page.getByLabel('Email').fill('user@phpkingdom.com');
  await page.getByLabel('Password').fill('secret');
  await page.getByRole('button', { name: 'Sign in' }).click();
  await expect(page).toHaveURL(/dashboard/);
  const cookies = await context.cookies();
  expect(cookies.some((c) => c.name === 'session')).toBeTruthy();
});
