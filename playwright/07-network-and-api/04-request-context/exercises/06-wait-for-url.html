Exercise 06

Request Context

Section: Network and API

Focus: Wait For Url

Scenario: component library

Complete the task locally and verify with npx playwright test / --ui.

Spec Sample

// Request Context — Exercise 06
// Focus: wait for url | Scenario: component library

import { test, expect } from '@playwright/test';

test('APIRequestContext CRUD', async ({ request }) => {
  const created = await request.post('/api/items', {
    data: { name: 'Notebook' },
  });
  expect(created.status()).toBe(201);
  const body = await created.json();
  expect(body).toHaveProperty('id');

  const got = await request.get(`/api/items/${body.id}`);
  expect(got.ok()).toBeTruthy();
  expect(await got.json()).toMatchObject({ name: 'Notebook' });
});
