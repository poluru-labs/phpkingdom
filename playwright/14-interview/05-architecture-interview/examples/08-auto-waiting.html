Spec Sample

// Architecture Interview — Example 08
// Focus: auto waiting | Scenario: mobile web app

// Architecture interview sketch — layered Playwright suite
// 1) API setup  2) storageState auth  3) POM UI  4) route mocks for edge cases

import { test, expect } from '@playwright/test';
import { DashboardPage } from '../pages/DashboardPage';

test.describe('suite architecture', () => {
  test.use({ storageState: 'playwright/.auth/user.json' });

  test('UI layer uses POM; data seeded via API', async ({ page, request }) => {
    await request.post('/api/seed', { data: { scenario: 'dashboard' } });
    const dash = new DashboardPage(page);
    await dash.goto();
    await expect(dash.heading).toBeVisible();
  });
});
// Discuss: fixtures vs POM, project dependencies, and isolating flaky network.
