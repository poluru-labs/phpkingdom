Spec Sample

// Playwright Interview Questions — Exercise 17
// Focus: debug headed | Scenario: API-backed SPA

// Interview warm-up: answer with working Playwright patterns
import { test, expect } from '@playwright/test';

test.describe('Common interview demos', () => {
  test('Q: How do you wait for UI?', async ({ page }) => {
    await page.goto('/');
    // A: web-first assertions auto-retry — no waitForTimeout
    await expect(page.getByRole('heading', { name: 'Welcome' })).toBeVisible();
  });

  test('Q: How do you mock network?', async ({ page }) => {
    await page.route('**/api/user', (route) =>
      route.fulfill({ json: { name: 'Ada' } }),
    );
    await page.goto('/profile');
    await expect(page.getByText('Ada')).toBeVisible();
  });
});
