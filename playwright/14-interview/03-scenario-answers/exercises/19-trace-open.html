// Scenario Answers — Exercise 19
// Focus: trace open | Scenario: docs site

// Scenario-style interview answers as runnable sketches
import { test, expect } from '@playwright/test';

test('Scenario: login once, reuse session', async ({ browser }) => {
  const context = await browser.newContext();
  const page = await context.newPage();
  await page.goto('/login');
  await page.getByLabel('Email').fill('user@phpkingdom.com');
  await page.getByLabel('Password').fill('secret');
  await page.getByRole('button', { name: 'Sign in' }).click();
  await context.storageState({ path: 'playwright/.auth/user.json' });
  await context.close();
});

test('Scenario: assert API + UI together', async ({ page, request }) => {
  const res = await request.get('/api/health');
  expect(res.ok()).toBeTruthy();
  await page.goto('/');
  await expect(page.getByText('Healthy')).toBeVisible();
});
