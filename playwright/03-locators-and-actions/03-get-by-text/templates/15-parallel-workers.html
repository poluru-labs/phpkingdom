// getByText — Template 15
// Focus: parallel workers | Scenario: auth portal

import { test, expect } from '@playwright/test';

test('getByText matching', async ({ page }) => {
  await page.goto('/pricing');
  await expect(page.getByText('Free forever')).toBeVisible();
  await expect(page.getByText(/pro plan/i)).toBeVisible();
  await page.getByText('Start trial', { exact: true }).click();
  await expect(page).toHaveURL(/trial/);
});
