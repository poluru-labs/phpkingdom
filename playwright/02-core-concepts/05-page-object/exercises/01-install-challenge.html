Spec Sample

// Page Object — Exercise 01
// Focus: install challenge | Scenario: SaaS dashboard

import { Page, expect } from '@playwright/test';

export class HomePage {
  constructor(private page: Page) {}
  readonly hero = () => this.page.getByTestId('hero');
  async open() {
    await this.page.goto('/');
  }
  async expectReady() {
    await expect(this.hero()).toBeVisible();
  }
}

import { test } from '@playwright/test';

test('page object home', async ({ page }) => {
  const home = new HomePage(page);
  await home.open();
  await home.expectReady();
});
