Starter Element

import { LitElement, html, css } from "lit";

export class PkCard extends LitElement {
  static styles = css`
    :host {
      display: block;
      border: 1px solid #e2e8f0;
      border-radius: 12px;
      padding: 1rem;
      background: #fff;
    }
    header { font-weight: 700; margin-bottom: 0.5rem; }
  `;
  render() {
    return html`
      <header><slot name="header">Card</slot></header>
      <div><slot></slot></div>
      <footer><slot name="footer"></slot></footer>
    `;
  }
}
customElements.define("pk-card", PkCard);
