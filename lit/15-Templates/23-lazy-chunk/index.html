Starter Element

const mount = document.querySelector("#lazy-host");
const trigger = document.querySelector("#load-chart");

trigger?.addEventListener("click", async () => {
  const { PkChart } = await import("./pk-chart.js");
  if (!customElements.get("pk-chart")) {
    customElements.define("pk-chart", PkChart);
  }
  mount?.replaceChildren(document.createElement("pk-chart"));
});
