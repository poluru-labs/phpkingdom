Starter Element

import { LitElement, html, css } from "lit";

export class PkFormField extends LitElement {
  static properties = {
    label: { type: String },
    name: { type: String },
    value: { type: String },
    type: { type: String },
  };
  static styles = css`
    :host { display: block; margin-bottom: 0.75rem; }
    label { display: grid; gap: 0.35rem; font-size: 0.9rem; }
    input {
      padding: 0.5rem 0.65rem; border: 1px solid #cbd5e1; border-radius: 8px;
    }
  `;
  constructor() {
    super();
    this.label = "Field";
    this.name = "field";
    this.value = "";
    this.type = "text";
  }
  render() {
    return html`
      <label>
        ${this.label}
        <input
          type=${this.type}
          name=${this.name}
          .value=${this.value}
          @input=${(e) => {
            this.value = e.target.value;
            this.dispatchEvent(new CustomEvent("pk-input", {
              detail: { value: this.value }, bubbles: true, composed: true,
            }));
          }}
        />
      </label>
    `;
  }
}
customElements.define("pk-form-field", PkFormField);
