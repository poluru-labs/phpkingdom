Starter Element

import { LitElement, html, css } from "lit";
import { classMap } from "lit/directives/class-map.js";

export class PkButtonSet extends LitElement {
  static properties = { variant: { type: String, reflect: true } };
  static styles = css`
    .row { display: flex; gap: 0.5rem; }
    button {
      border-radius: 8px; padding: 0.45rem 0.85rem; border: 1px solid #0f766e;
      background: #0f766e; color: #fff; cursor: pointer;
    }
    button.ghost { background: transparent; color: #0f766e; }
  `;
  constructor() {
    super();
    this.variant = "primary";
  }
  render() {
    return html`
      <div class="row">
        <button class=${classMap({ ghost: this.variant === "ghost" })}>
          <slot>Action</slot>
        </button>
      </div>
    `;
  }
}
customElements.define("pk-button-set", PkButtonSet);
