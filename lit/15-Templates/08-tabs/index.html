Starter Element

import { LitElement, html, css } from "lit";

export class PkTabs extends LitElement {
  static properties = {
    selected: { type: Number },
    tabs: { type: Array },
  };
  static styles = css`
    [role="tablist"] { display: flex; gap: 0.25rem; }
    [role="tab"][aria-selected="true"] { font-weight: 700; border-bottom: 2px solid #0f766e; }
  `;
  constructor() {
    super();
    this.selected = 0;
    this.tabs = ["One", "Two", "Three"];
  }
  render() {
    return html`
      <div role="tablist">
        ${this.tabs.map(
          (label, i) => html`
            <button
              role="tab"
              aria-selected=${i === this.selected}
              @click=${() => (this.selected = i)}
            >${label}</button>
          `
        )}
      </div>
      <div role="tabpanel"><slot name=${`panel-${this.selected}`}></slot></div>
    `;
  }
}
customElements.define("pk-tabs", PkTabs);
