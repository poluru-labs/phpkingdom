Starter Element

import { fixture, html, expect, oneEvent } from "@open-wc/testing";
import "../pk-counter.js";

describe("pk-counter", () => {
  it("renders label", async () => {
    const el = await fixture(html`<pk-counter label="Hits"></pk-counter>`);
    expect(el.shadowRoot.textContent).to.include("Hits");
  });

  it("emits on increment", async () => {
    const el = await fixture(html`<pk-counter></pk-counter>`);
    setTimeout(() => el.shadowRoot.querySelector("button").click());
    const ev = await oneEvent(el, "count-changed");
    expect(ev.detail.count).to.be.a("number");
  });
});
