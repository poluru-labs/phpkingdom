Starter Element

import { LitElement, html, css } from "lit";

export class PkThemeToken extends LitElement {
  static styles = css`
    :host {
      --pk-accent: #0f766e;
      --pk-surface: #f8fafc;
      --pk-radius: 10px;
      display: block;
      background: var(--pk-surface);
      border-radius: var(--pk-radius);
      padding: 1rem;
    }
    .swatch {
      width: 3rem; height: 3rem; border-radius: var(--pk-radius);
      background: var(--pk-accent);
    }
  `;
  render() {
    return html`<div class="swatch" part="swatch" title="--pk-accent"></div>`;
  }
}
customElements.define("pk-theme-token", PkThemeToken);
