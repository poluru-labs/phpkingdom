Starter Element

import { LitElement, html, css } from "lit";

export class PkCounter extends LitElement {
  static properties = {
    label: { type: String },
    count: { type: Number, state: true },
  };
  static styles = css`
    button {
      background: #0f766e;
      color: #fff;
      border: 0;
      border-radius: 8px;
      padding: 0.5rem 0.9rem;
    }
  `;
  constructor() {
    super();
    this.label = "Count";
    this.count = 0;
  }
  render() {
    return html`
      <button @click=${() => (this.count += 1)}>
        ${this.label}: ${this.count}
      </button>
    `;
  }
}
customElements.define("pk-counter", PkCounter);
