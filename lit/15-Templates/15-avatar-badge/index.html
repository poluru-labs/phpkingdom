Starter Element

import { LitElement, html, css } from "lit";

export class PkAvatarBadge extends LitElement {
  static properties = {
    name: { type: String },
    status: { type: String, reflect: true },
  };
  static styles = css`
    :host { display: inline-grid; place-items: center; position: relative; }
    .avatar {
      width: 2.5rem; height: 2.5rem; border-radius: 999px;
      background: #ccfbf1; display: grid; place-items: center; font-weight: 700;
    }
    .badge {
      position: absolute; right: 0; bottom: 0;
      width: 0.65rem; height: 0.65rem; border-radius: 999px; background: #22c55e;
    }
    :host([status="busy"]) .badge { background: #ef4444; }
  `;
  constructor() {
    super();
    this.name = "Ada";
    this.status = "online";
  }
  render() {
    const initial = (this.name || "?").slice(0, 1).toUpperCase();
    return html`
      <div class="avatar" part="avatar">${initial}</div>
      <span class="badge" part="badge" aria-label=${this.status}></span>
    `;
  }
}
customElements.define("pk-avatar-badge", PkAvatarBadge);
