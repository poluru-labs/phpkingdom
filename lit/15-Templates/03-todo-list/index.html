Starter Element

import { LitElement, html } from "lit";

export class PkTodoList extends LitElement {
  static properties = {
    text: { type: String, state: true },
    items: { type: Array, state: true },
  };
  constructor() {
    super();
    this.text = "";
    this.items = [];
  }
  #add(e) {
    e.preventDefault();
    const t = this.text.trim();
    if (!t) return;
    this.items = [...this.items, { id: Date.now(), text: t }];
    this.text = "";
  }
  #remove(id) {
    this.items = this.items.filter((i) => i.id !== id);
  }
  render() {
    return html`
      <form @submit=${this.#add}>
        <input .value=${this.text} @input=${(e) => (this.text = e.target.value)} />
        <button>Add</button>
      </form>
      <ul>
        ${this.items.map(
          (i) => html`<li>${i.text}
            <button @click=${() => this.#remove(i.id)}>×</button></li>`
        )}
      </ul>
    `;
  }
}
customElements.define("pk-todo-list", PkTodoList);
