Simple table element

Starter Element

import { LitElement, html, css } from "lit";
import { repeat } from "lit/directives/repeat.js";

export class PkDataTable extends LitElement {
  static properties = { rows: { type: Array } };
  static styles = css`
    table { width: 100%; border-collapse: collapse; }
    th, td { border-bottom: 1px solid #e2e8f0; padding: 0.5rem; text-align: left; }
  `;
  constructor() {
    super();
    this.rows = [
      { id: 1, name: "Ada", role: "Admin" },
      { id: 2, name: "Grace", role: "Editor" },
    ];
  }
  render() {
    return html`
      <table>
        <thead><tr><th>Name</th><th>Role</th></tr></thead>
        <tbody>
          ${repeat(
            this.rows,
            (r) => r.id,
            (r) => html`<tr><td>${r.name}</td><td>${r.role}</td></tr>`
          )}
        </tbody>
      </table>
    `;
  }
}
customElements.define("pk-data-table", PkDataTable);
