Starter Element

import { LitElement, html, css } from "lit";

export class PkModal extends LitElement {
  static properties = { open: { type: Boolean, reflect: true } };
  static styles = css`
    .backdrop {
      position: fixed; inset: 0; background: rgba(15, 23, 42, 0.45);
      display: grid; place-items: center;
    }
    .panel {
      background: #fff; border-radius: 12px; padding: 1.25rem; min-width: 280px;
    }
  `;
  #close() { this.open = false; }
  render() {
    if (!this.open) return html``;
    return html`
      <div class="backdrop" @click=${this.#close}>
        <div class="panel" role="dialog" aria-modal="true"
             @click=${(e) => e.stopPropagation()}>
          <header><slot name="title">Dialog</slot></header>
          <slot></slot>
          <button @click=${this.#close}>Close</button>
        </div>
      </div>
    `;
  }
}
customElements.define("pk-modal", PkModal);
