Starter Element

import { LitElement, html, css } from "lit";

export class PkSearchInput extends LitElement {
  static properties = {
    value: { type: String },
    placeholder: { type: String },
  };
  static styles = css`
    input {
      width: 100%; padding: 0.55rem 0.75rem;
      border: 1px solid #cbd5e1; border-radius: 999px;
    }
  `;
  #timer;
  constructor() {
    super();
    this.value = "";
    this.placeholder = "Search…";
  }
  #onInput(e) {
    this.value = e.target.value;
    clearTimeout(this.#timer);
    this.#timer = setTimeout(() => {
      this.dispatchEvent(new CustomEvent("pk-search", {
        detail: { q: this.value }, bubbles: true, composed: true,
      }));
    }, 250);
  }
  disconnectedCallback() {
    clearTimeout(this.#timer);
    super.disconnectedCallback();
  }
  render() {
    return html`
      <input
        type="search"
        placeholder=${this.placeholder}
        .value=${this.value}
        @input=${this.#onInput}
      />
    `;
  }
}
customElements.define("pk-search-input", PkSearchInput);
