import { LitElement, html, css } from "lit";

export class PkSelectMenu extends LitElement {
  static properties = {
    value: { type: String },
    options: { type: Array },
    open: { type: Boolean, state: true },
  };
  static styles = css`
    :host { position: relative; display: inline-block; }
    ul {
      position: absolute; margin: 0; padding: 0.25rem; list-style: none;
      background: #fff; border: 1px solid #e2e8f0; border-radius: 8px;
    }
  `;
  constructor() {
    super();
    this.value = "";
    this.open = false;
    this.options = [
      { value: "lit", label: "Lit" },
      { value: "wc", label: "Web Components" },
    ];
  }
  #pick(opt) {
    this.value = opt.value;
    this.open = false;
    this.dispatchEvent(new CustomEvent("pk-change", {
      detail: { value: this.value }, bubbles: true, composed: true,
    }));
  }
  render() {
    const current = this.options.find((o) => o.value === this.value);
    return html`
      <button @click=${() => (this.open = !this.open)}>
        ${current?.label ?? "Select…"}
      </button>
      ${this.open
        ? html`<ul>
            ${this.options.map(
              (o) => html`<li><button @click=${() => this.#pick(o)}>${o.label}</button></li>`
            )}
          </ul>`
        : null}
    `;
  }
}
customElements.define("pk-select-menu", PkSelectMenu);
