Starter Element

import { LitElement, html, css } from "lit";

export class PkAccordion extends LitElement {
  static properties = {
    openId: { type: String, state: true },
    panels: { type: Array },
  };
  static styles = css`
    details { border-bottom: 1px solid #e2e8f0; padding: 0.5rem 0; }
  `;
  constructor() {
    super();
    this.openId = "a";
    this.panels = [
      { id: "a", title: "Section A", body: "Content A" },
      { id: "b", title: "Section B", body: "Content B" },
    ];
  }
  render() {
    return html`
      ${this.panels.map(
        (p) => html`
          <details
            ?open=${this.openId === p.id}
            @toggle=${(e) => {
              if (e.target.open) this.openId = p.id;
            }}
          >
            <summary>${p.title}</summary>
            <p>${p.body}</p>
          </details>
        `
      )}
    `;
  }
}
customElements.define("pk-accordion", PkAccordion);
