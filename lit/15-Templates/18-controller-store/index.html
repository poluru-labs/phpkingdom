Starter Element

import { LitElement, html } from "lit";

class ClockController {
  constructor(host) {
    this.host = host;
    this.now = new Date();
    host.addController(this);
  }
  hostConnected() {
    this._t = setInterval(() => {
      this.now = new Date();
      this.host.requestUpdate();
    }, 1000);
  }
  hostDisconnected() {
    clearInterval(this._t);
  }
}

export class PkControllerStore extends LitElement {
  constructor() {
    super();
    this.clock = new ClockController(this);
  }
  render() {
    return html`<time>${this.clock.now.toLocaleTimeString()}</time>`;
  }
}
customElements.define("pk-controller-store", PkControllerStore);
