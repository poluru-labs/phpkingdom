Starter Element

import { LitElement, html, css } from "lit";

export class PkPagination extends LitElement {
  static properties = {
    page: { type: Number },
    total: { type: Number },
  };
  static styles = css`
    .row { display: flex; gap: 0.5rem; align-items: center; }
  `;
  constructor() {
    super();
    this.page = 1;
    this.total = 10;
  }
  #go(delta) {
    const next = Math.min(this.total, Math.max(1, this.page + delta));
    if (next === this.page) return;
    this.page = next;
    this.dispatchEvent(new CustomEvent("pk-page", {
      detail: { page: this.page }, bubbles: true, composed: true,
    }));
  }
  render() {
    return html`
      <div class="row">
        <button ?disabled=${this.page <= 1} @click=${() => this.#go(-1)}>Prev</button>
        <span>${this.page} / ${this.total}</span>
        <button ?disabled=${this.page >= this.total} @click=${() => this.#go(1)}>Next</button>
      </div>
    `;
  }
}
customElements.define("pk-pagination", PkPagination);
