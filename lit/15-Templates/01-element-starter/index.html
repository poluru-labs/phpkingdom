Starter Element

import { LitElement, html, css } from "lit";

export class PkStarter extends LitElement {
  static styles = css`:host { display: block; }`;
  render() {
    return html`<p>Lit starter · phpkingdom.com</p>`;
  }
}
customElements.define("pk-starter", PkStarter);

npm install lit
