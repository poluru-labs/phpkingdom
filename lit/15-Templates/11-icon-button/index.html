Accessible icon control

Starter Element

import { LitElement, html, css } from "lit";

export class PkIconButton extends LitElement {
  static properties = {
    label: { type: String },
    disabled: { type: Boolean, reflect: true },
  };
  static styles = css`
    button {
      width: 2.25rem; height: 2.25rem; border-radius: 999px;
      border: 1px solid #cbd5e1; background: #fff; cursor: pointer;
    }
    button:disabled { opacity: 0.5; cursor: not-allowed; }
  `;
  constructor() {
    super();
    this.label = "Close";
    this.disabled = false;
  }
  render() {
    return html`
      <button
        type="button"
        aria-label=${this.label}
        ?disabled=${this.disabled}
        @click=${() => this.dispatchEvent(new CustomEvent("pk-click", {
          bubbles: true, composed: true,
        }))}
      >
        <slot>✕</slot>
      </button>
    `;
  }
}
customElements.define("pk-icon-button", PkIconButton);
