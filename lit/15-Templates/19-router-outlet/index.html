Outlet composition

Starter Element

import { LitElement, html } from "lit";

export class PkRouterOutlet extends LitElement {
  static properties = { route: { type: String } };
  constructor() {
    super();
    this.route = "home";
  }
  render() {
    return html`
      <nav>
        <button @click=${() => (this.route = "home")}>Home</button>
        <button @click=${() => (this.route = "about")}>About</button>
      </nav>
      <main>
        <slot name=${this.route}></slot>
      </main>
    `;
  }
}
customElements.define("pk-router-outlet", PkRouterOutlet);
