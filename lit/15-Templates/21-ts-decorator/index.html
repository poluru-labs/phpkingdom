Starter Element

import { LitElement, html, css } from "lit";
import { customElement, property, state } from "lit/decorators.js";

@customElement("pk-ts-counter")
export class PkTsCounter extends LitElement {
  static styles = css`:host { display: inline-block; }`;

  @property({ type: String }) label = "Count";
  @state() private value = 0;

  render() {
    return html`
      <button @click=${() => (this.value += 1)}>
        ${this.label}: ${this.value}
      </button>
    `;
  }
}

declare global {
  interface HTMLElementTagNameMap {
    "pk-ts-counter": PkTsCounter;
  }
}
