// Event Options — Exercise 06
// Focus: state toggle | Scenario: form wizard

import { LitElement, html } from "lit";

export class PkOnce extends LitElement {
  // Declarative listener options (Lit 2+)
  static properties = { armed: { type: Boolean, state: true } };

  constructor() {
    super();
    this.armed = true;
  }

  render() {
    return html`
      <button
        @click=${{
          handleEvent: () => {
            this.armed = false;
            console.log("fired once via options object pattern");
          },
          once: true,
        }}
      >
        ${this.armed ? "Arm" : "Done"}
      </button>
    `;
  }
}
customElements.define("pk-once", PkOnce);
// Also: addEventListener('scroll', handler, { passive: true })
