// Event Handling — Exercise 06
// Focus: state toggle | Scenario: form wizard

import { LitElement, html } from "lit";

export class PkClicker extends LitElement {
  static properties = { count: { type: Number, state: true } };

  constructor() {
    super();
    this.count = 0;
  }

  #onClick(e) {
    e.preventDefault();
    this.count += 1;
  }

  render() {
    return html`
      <button @click=${this.#onClick}>Clicked ${this.count}</button>
      <input @keydown=${(e) => e.key === "Enter" && this.#onClick(e)} />
    `;
  }
}
customElements.define("pk-clicker", PkClicker);
// Prefer @event in templates; use .bind / arrows carefully with `this`
