// Lit Events — Template 17
// Focus: pagination | Scenario: media player

import { LitElement, html } from "lit";

export class PkEvents extends LitElement {
  static properties = { n: { type: Number, state: true } };

  constructor() {
    super();
    this.n = 0;
  }

  #bump() {
    this.n += 1;
    this.dispatchEvent(
      new CustomEvent("pk-bump", {
        detail: { n: this.n },
        bubbles: true,
        composed: true,
      })
    );
  }

  render() {
    return html`<button @click=${this.#bump}>n=${this.n}</button>`;
  }
}
customElements.define("pk-events", PkEvents);
