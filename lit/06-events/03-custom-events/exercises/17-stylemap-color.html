// Custom Events — Exercise 17
// Focus: styleMap color | Scenario: media player

import { LitElement, html } from "lit";

export class PkSaveButton extends LitElement {
  static properties = { id: { type: String } };

  #save() {
    this.dispatchEvent(
      new CustomEvent("pk-save", {
        detail: { id: this.id, at: Date.now() },
        bubbles: true,
        composed: true,
      })
    );
  }

  render() {
    return html`<button @click=${this.#save}>Save</button>`;
  }
}
customElements.define("pk-save-button", PkSaveButton);
// Parent: el.addEventListener('pk-save', (e) => console.log(e.detail));
