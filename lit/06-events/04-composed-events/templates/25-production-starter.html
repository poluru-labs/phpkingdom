// Composed Events — Template 25
// Focus: production starter | Scenario: docs site

import { LitElement, html } from "lit";

export class PkInner extends LitElement {
  #ping() {
    // composed:true crosses shadow boundaries
    this.dispatchEvent(
      new CustomEvent("pk-ping", {
        detail: { from: "inner" },
        bubbles: true,
        composed: true,
      })
    );
  }

  render() {
    return html`<button @click=${this.#ping}>Ping</button>`;
  }
}
customElements.define("pk-inner", PkInner);

export class PkOuter extends LitElement {
  render() {
    return html`<pk-inner @pk-ping=${(e) => console.log(e.detail)}></pk-inner>`;
  }
}
customElements.define("pk-outer", PkOuter);
