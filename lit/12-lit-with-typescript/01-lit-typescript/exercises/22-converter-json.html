// Lit with TypeScript — Exercise 22
// Focus: converter json | Scenario: dashboard widget

import { LitElement, html, css } from "lit";
import { customElement, property, state } from "lit/decorators.js";

@customElement("pk-ts-hello")
export class PkTsHello extends LitElement {
  static styles = css`:host { display: block; }`;

  @property({ type: String }) name = "Lit";
  @state() private ready = false;

  firstUpdated() {
    this.ready = true;
  }

  render() {
    return html`<p>Hello ${this.name}${this.ready ? " ✓" : ""}</p>`;
  }
}

declare global {
  interface HTMLElementTagNameMap {
    "pk-ts-hello": PkTsHello;
  }
}
