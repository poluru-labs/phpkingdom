// TypeScript Properties — Exercise 18
// Focus: repeat keyed | Scenario: chat widget

import { LitElement, html } from "lit";
import { customElement, property } from "lit/decorators.js";

@customElement("pk-ts-props")
export class PkTsProps extends LitElement {
  @property({ type: String }) label = "Count";
  @property({ type: Number, reflect: true }) value = 0;
  @property({ type: Boolean, attribute: "disabled" }) disabled = false;

  render() {
    return html`
      <button ?disabled=${this.disabled}>
        ${this.label}: ${this.value}
      </button>
    `;
  }
}
