// Lit Decorators — Example 25
// Focus: best practice recap | Scenario: docs site

import { LitElement, html } from "lit";
import { customElement, property, state, query } from "lit/decorators.js";

@customElement("pk-decorated")
export class PkDecorated extends LitElement {
  @property({ type: String }) label = "Go";
  @state() private pressed = false;
  @query("button") private button!: HTMLButtonElement;

  firstUpdated() {
    this.button?.focus();
  }

  render() {
    return html`
      <button @click=${() => (this.pressed = !this.pressed)}>
        ${this.label}${this.pressed ? "!" : ""}
      </button>
    `;
  }
}
