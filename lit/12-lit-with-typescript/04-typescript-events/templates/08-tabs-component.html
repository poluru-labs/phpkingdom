// TypeScript Events — Template 08
// Focus: tabs component | Scenario: chat widget

import { LitElement, html } from "lit";
import { customElement, property } from "lit/decorators.js";

export type PkChangeDetail = { value: number };

@customElement("pk-ts-events")
export class PkTsEvents extends LitElement {
  @property({ type: Number }) value = 0;

  #emit() {
    this.dispatchEvent(
      new CustomEvent<PkChangeDetail>("pk-change", {
        detail: { value: this.value },
        bubbles: true,
        composed: true,
      })
    );
  }

  render() {
    return html`
      <button
        @click=${() => {
          this.value += 1;
          this.#emit();
        }}
      >
        ${this.value}
      </button>
    `;
  }
}

declare global {
  interface HTMLElementEventMap {
    "pk-change": CustomEvent<PkChangeDetail>;
  }
}
