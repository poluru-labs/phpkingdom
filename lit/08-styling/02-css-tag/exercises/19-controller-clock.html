// css Template Tag — Exercise 19
// Focus: controller clock | Scenario: settings page

import { LitElement, html, css } from "lit";

const base = css`
  :host {
    display: block;
  }
`;

export class PkCssTag extends LitElement {
  static styles = [
    base,
    css`
      p {
        margin: 0;
        font-weight: 600;
      }
    `,
  ];

  render() {
    return html`<p><slot></slot></p>`;
  }
}
customElements.define("pk-css-tag", PkCssTag);
// css`` returns a CSSResult; arrays merge multiple sheets
