// CSS Shadow Parts — Template 18
// Focus: controller store | Scenario: chat widget

import { LitElement, html, css } from "lit";

export class PkPartButton extends LitElement {
  static styles = css`
    button {
      background: #0f766e;
      color: #fff;
      border: 0;
      padding: 0.5rem 0.85rem;
    }
  `;

  render() {
    return html`<button part="button"><slot></slot></button>`;
  }
}
customElements.define("pk-part-button", PkPartButton);
// Outside shadow: pk-part-button::part(button) { background: crimson; }
