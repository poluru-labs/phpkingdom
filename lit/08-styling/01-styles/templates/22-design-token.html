// Lit Styles — Template 22
// Focus: design token | Scenario: dashboard widget

import { LitElement, html, css } from "lit";

export class PkStyled extends LitElement {
  static styles = css`
    :host {
      display: block;
      color: #0f172a;
    }
    .box {
      padding: 0.75rem 1rem;
      background: #ecfdf5;
      border-radius: 8px;
    }
  `;

  render() {
    return html`<div class="box"><slot></slot></div>`;
  }
}
customElements.define("pk-styled", PkStyled);
// Shadow DOM styles are scoped to the component
