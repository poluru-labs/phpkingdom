// Adopted Stylesheets — Template 25
// Focus: production starter | Scenario: docs site

import { LitElement, html, css, unsafeCSS } from "lit";

// Shared sheet (conceptually Constructable Stylesheet / css result)
export const sharedTokens = css`
  :host {
    --pk-fg: #0f172a;
    --pk-bg: #f8fafc;
  }
`;

export class PkSheetHost extends LitElement {
  static styles = [
    sharedTokens,
    css`
      :host {
        display: block;
        color: var(--pk-fg);
        background: var(--pk-bg);
        padding: 1rem;
      }
    `,
  ];

  render() {
    return html`<slot></slot>`;
  }
}
customElements.define("pk-sheet-host", PkSheetHost);
// Lit adopts CSSResult arrays into the shadow root efficiently
