// CSS Custom Properties — Example 05
// Focus: state field | Scenario: docs site

import { LitElement, html, css } from "lit";

export class PkThemedButton extends LitElement {
  static styles = css`
    :host {
      --pk-accent: #0f766e;
      --pk-radius: 8px;
      display: inline-block;
    }
    button {
      background: var(--pk-accent);
      color: white;
      border: 0;
      border-radius: var(--pk-radius);
      padding: 0.5rem 0.9rem;
    }
  `;

  render() {
    return html`<button part="button"><slot>Action</slot></button>`;
  }
}
customElements.define("pk-themed-button", PkThemedButton);
// Consumers theme: pk-themed-button { --pk-accent: #b45309; }
