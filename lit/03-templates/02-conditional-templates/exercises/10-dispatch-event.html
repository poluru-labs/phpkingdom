// Conditional Templates — Exercise 10
// Focus: dispatch event | Scenario: ecommerce card

import { LitElement, html } from "lit";

export class PkAlert extends LitElement {
  static properties = {
    open: { type: Boolean },
    tone: { type: String },
  };

  constructor() {
    super();
    this.open = true;
    this.tone = "info";
  }

  render() {
    return html`
      <button @click=${() => (this.open = !this.open)}>Toggle</button>
      ${this.open
        ? html`<div class="alert ${this.tone}">Visible alert</div>`
        : html`<p class="muted">Hidden</p>`}
    `;
  }
}
customElements.define("pk-alert", PkAlert);
