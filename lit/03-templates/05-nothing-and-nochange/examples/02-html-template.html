// nothing and noChange — Example 02
// Focus: html template | Scenario: dashboard widget

import { LitElement, html, nothing, noChange } from "lit";

export class PkOptional extends LitElement {
  static properties = {
    showHint: { type: Boolean },
    locked: { type: Boolean },
    label: { type: String },
  };

  constructor() {
    super();
    this.showHint = false;
    this.locked = false;
    this.label = "Save";
  }

  render() {
    return html`
      <button>${this.label}</button>
      ${this.showHint ? html`<small>Tip</small>` : nothing}
      <span>${this.locked ? noChange : this.label}</span>
    `;
  }
}
customElements.define("pk-optional", PkOptional);
// nothing removes the node; noChange keeps the previous rendered value
