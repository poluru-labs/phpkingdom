// Template Expressions — Exercise 21
// Focus: decorator prop | Scenario: design system

import { LitElement, html } from "lit";

export class PkGreeting extends LitElement {
  static properties = {
    name: { type: String },
    score: { type: Number },
  };

  constructor() {
    super();
    this.name = "Ada";
    this.score = 42;
  }

  render() {
    return html`
      <h2>Hello ${this.name}</h2>
      <p>Score: ${this.score * 2}</p>
      <p>${this.score > 40 ? "Great" : "Keep going"}</p>
    `;
  }
}
customElements.define("pk-greeting", PkGreeting);
// Expressions are escaped by default; use unsafeHTML only when trusted
