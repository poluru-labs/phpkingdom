// List Rendering — Exercise 22
// Focus: converter json | Scenario: dashboard widget

import { LitElement, html } from "lit";

export class PkList extends LitElement {
  static properties = {
    items: { type: Array },
  };

  constructor() {
    super();
    this.items = [
      { id: 1, label: "Lit" },
      { id: 2, label: "Web Components" },
    ];
  }

  render() {
    return html`
      <ul>
        ${this.items.map(
          (item) => html`<li data-id=${item.id}>${item.label}</li>`
        )}
      </ul>
    `;
  }
}
customElements.define("pk-list", PkList);
// Prefer repeat() when reordering/inserting keyed items
