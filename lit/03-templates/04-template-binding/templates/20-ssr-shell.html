// Template Binding — Template 20
// Focus: ssr shell | Scenario: ecommerce card

import { LitElement, html } from "lit";

export class PkField extends LitElement {
  static properties = {
    name: { type: String },
    disabled: { type: Boolean },
    value: { type: String },
  };

  constructor() {
    super();
    this.name = "email";
    this.disabled = false;
    this.value = "";
  }

  render() {
    return html`
      <label>
        ${this.name}
        <input
          name=${this.name}
          ?disabled=${this.disabled}
          .value=${this.value}
          @input=${(e) => (this.value = e.target.value)}
        />
      </label>
    `;
  }
}
customElements.define("pk-field", PkField);
// ?attr = boolean attribute, .prop = property binding, @event = listener
