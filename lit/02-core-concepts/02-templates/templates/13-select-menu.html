// Lit Templates — Template 13
// Focus: select menu | Scenario: marketing site

import { LitElement, html } from "lit";

export class PkTemplate extends LitElement {
  static properties = { user: { type: String } };

  constructor() {
    super();
    this.user = "guest";
  }

  render() {
    // html`` returns a TemplateResult — efficient DOM diffing
    return html`
      <section>
        <h1>Hi ${this.user}</h1>
        <slot name="actions"></slot>
      </section>
    `;
  }
}
customElements.define("pk-template", PkTemplate);
