// render() Method — Template 02
// Focus: counter widget | Scenario: dashboard widget

import { LitElement, html } from "lit";

export class PkRenderDemo extends LitElement {
  static properties = {
    title: { type: String },
    items: { type: Array },
  };

  constructor() {
    super();
    this.title = "Render";
    this.items = ["a", "b"];
  }

  // render() is pure-ish: read reactive state, return a TemplateResult
  render() {
    return html`
      <h2>${this.title}</h2>
      <ul>
        ${this.items.map((item) => html`<li>${item}</li>`)}
      </ul>
    `;
  }
}
customElements.define("pk-render-demo", PkRenderDemo);
// Avoid side effects in render(); put them in lifecycle or event handlers
