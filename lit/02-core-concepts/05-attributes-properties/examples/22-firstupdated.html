// Attributes and Properties — Example 22
// Focus: firstUpdated | Scenario: dashboard widget

import { LitElement, html } from "lit";

export class PkBadge extends LitElement {
  static properties = {
    // Attribute "label" syncs to property label
    label: { type: String },
    // Attribute "count" converted to Number
    count: { type: Number, attribute: "count" },
    // Boolean attribute presence => true
    busy: { type: Boolean, reflect: true },
  };

  constructor() {
    super();
    this.label = "Items";
    this.count = 0;
    this.busy = false;
  }

  render() {
    return html`<span>${this.label}: ${this.count}${this.busy ? "…" : ""}</span>`;
  }
}
customElements.define("pk-badge", PkBadge);
// HTML: <pk-badge label="Cart" count="3" busy></pk-badge>
