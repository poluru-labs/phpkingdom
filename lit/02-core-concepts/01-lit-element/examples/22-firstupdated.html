// LitElement — Example 22
// Focus: firstUpdated | Scenario: dashboard widget

import { LitElement, html, css } from "lit";

export class PkElement extends LitElement {
  static properties = {
    label: { type: String },
  };

  static styles = css`
    :host {
      display: inline-flex;
      gap: 0.5rem;
      align-items: center;
    }
  `;

  constructor() {
    super();
    this.label = "LitElement";
  }

  render() {
    return html`<span>${this.label}</span><slot></slot>`;
  }
}
customElements.define("pk-element", PkElement);
// LitElement = ReactiveElement + html rendering into shadow DOM
