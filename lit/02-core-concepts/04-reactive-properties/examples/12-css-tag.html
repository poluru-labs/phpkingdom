// Reactive Properties — Example 12
// Focus: css tag | Scenario: dashboard widget

import { LitElement, html } from "lit";

export class PkCounter extends LitElement {
  static properties = {
    count: { type: Number },
  };

  constructor() {
    super();
    this.count = 0;
  }

  #inc() {
    // Mutating a reactive property schedules an update
    this.count += 1;
  }

  render() {
    return html`<button @click=${this.#inc}>${this.count}</button>`;
  }
}
customElements.define("pk-counter", PkCounter);
