// Component Composition — Template 15
// Focus: avatar badge | Scenario: docs site

import { LitElement, html } from "lit";
import "./pk-avatar.js";
import "./pk-badge.js";

export class PkUserRow extends LitElement {
  static properties = {
    name: { type: String },
    role: { type: String },
  };

  render() {
    return html`
      <div class="row">
        <pk-avatar name=${this.name}></pk-avatar>
        <span>${this.name}</span>
        <pk-badge label=${this.role}></pk-badge>
      </div>
    `;
  }
}
customElements.define("pk-user-row", PkUserRow);
