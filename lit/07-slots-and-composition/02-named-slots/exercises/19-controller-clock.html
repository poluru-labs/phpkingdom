// Named Slots — Exercise 19
// Focus: controller clock | Scenario: settings page

import { LitElement, html, css } from "lit";

export class PkCard extends LitElement {
  static styles = css`
    :host { display: block; border: 1px solid #e2e8f0; padding: 1rem; }
    header { font-weight: 700; }
  `;

  render() {
    return html`
      <header><slot name="header">Untitled</slot></header>
      <div class="body"><slot></slot></div>
      <footer><slot name="footer"></slot></footer>
    `;
  }
}
customElements.define("pk-card", PkCard);
// <pk-card>
//   <span slot="header">Title</span>
//   Body copy
//   <button slot="footer">OK</button>
// </pk-card>
