// Lit Slots — Exercise 21
// Focus: decorator prop | Scenario: design system

import { LitElement, html, css } from "lit";

export class PkSlotHost extends LitElement {
  static styles = css`
    :host { display: block; padding: 1rem; border: 1px dashed #94a3b8; }
  `;

  render() {
    return html`<div class="wrap"><slot>Default content</slot></div>`;
  }
}
customElements.define("pk-slot-host", PkSlotHost);
// Light DOM children project into <slot>
