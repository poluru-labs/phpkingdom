// slotchange Event — Exercise 25

import { LitElement, html } from "lit";

export class PkSlotWatcher extends LitElement {
  static properties = { count: { type: Number, state: true } };

  constructor() {
    super();
    this.count = 0;
  }

  #onSlotChange(e) {
    const slot = e.target;
    this.count = slot.assignedNodes({ flatten: true }).filter(
      (n) => n.nodeType === Node.ELEMENT_NODE || n.textContent?.trim()
    ).length;
  }

  render() {
    return html`
      <p>Projected nodes: ${this.count}</p>
      <slot @slotchange=${this.#onSlotChange}></slot>
    `;
  }
}
customElements.define("pk-slot-watcher", PkSlotWatcher);
