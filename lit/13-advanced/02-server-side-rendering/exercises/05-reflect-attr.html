// Lit Server-Side Rendering — Exercise 05
// Focus: reflect attr | Scenario: docs site

// Node / SSR sketch with @lit-labs/ssr (conceptual)
import { render } from "@lit-labs/ssr";
import { html } from "lit";
import { collectResult } from "@lit-labs/ssr/lib/render-result.js";
import "./pk-hello.js";

export async function renderPage(name: string) {
  const result = render(html`
    <!doctype html>
    <html>
      <body>
        <pk-hello name=${name}></pk-hello>
        <script type="module" src="/pk-hello.js"></script>
      </body>
    </html>
  `);
  return collectResult(result);
}
// Hydrate on the client so the custom element upgrades smoothly
