// Lit Performance — Exercise 23

import { LitElement, html } from "lit";
import { repeat } from "lit/directives/repeat.js";
import { guard } from "lit/directives/guard.js";

export class PkPerfList extends LitElement {
  static properties = {
    rows: { type: Array },
    filter: { type: String },
  };

  constructor() {
    super();
    this.rows = Array.from({ length: 100 }, (_, i) => ({ id: i, label: `Row ${i}` }));
    this.filter = "";
  }

  render() {
    return html`
      ${guard([this.rows, this.filter], () => {
        const q = this.filter.toLowerCase();
        const list = this.rows.filter((r) => r.label.toLowerCase().includes(q));
        return html`
          <ul>
            ${repeat(list, (r) => r.id, (r) => html`<li>${r.label}</li>`)}
          </ul>
        `;
      })}
    `;
  }
}
customElements.define("pk-perf-list", PkPerfList);
// Tips: keyed repeat, guard expensive subtrees, avoid work in render
