// Lit Testing — Template 24
// Focus: test harness | Scenario: admin panel

import { fixture, html, expect, oneEvent } from "@open-wc/testing";
import "../pk-counter.js";

describe("pk-counter", () => {
  it("increments on click", async () => {
    const el = await fixture(html`<pk-counter></pk-counter>`);
    const btn = el.shadowRoot.querySelector("button");
    btn.click();
    await el.updateComplete;
    expect(btn.textContent).to.include("1");
  });

  it("fires count-changed", async () => {
    const el = await fixture(html`<pk-counter></pk-counter>`);
    setTimeout(() => el.shadowRoot.querySelector("button").click());
    const ev = await oneEvent(el, "count-changed");
    expect(ev.detail.count).to.equal(1);
  });
});
