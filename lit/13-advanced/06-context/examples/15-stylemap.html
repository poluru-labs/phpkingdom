// Lit Context — Example 15
// Focus: styleMap | Scenario: docs site

import { LitElement, html } from "lit";
import { createContext, ContextProvider, ContextConsumer } from "@lit/context";

export const themeContext = createContext("pk-theme");

export class PkThemeProvider extends LitElement {
  static properties = { theme: { type: String } };

  constructor() {
    super();
    this.theme = "light";
    this._provider = new ContextProvider(this, {
      context: themeContext,
      initialValue: this.theme,
    });
  }

  willUpdate(changed) {
    if (changed.has("theme")) this._provider.setValue(this.theme);
  }

  render() {
    return html`<slot></slot>`;
  }
}
customElements.define("pk-theme-provider", PkThemeProvider);

export class PkThemeConsumer extends LitElement {
  constructor() {
    super();
    this._theme = new ContextConsumer(this, {
      context: themeContext,
      subscribe: true,
      callback: () => this.requestUpdate(),
    });
  }

  render() {
    return html`<p>Theme: ${this._theme.value}</p>`;
  }
}
customElements.define("pk-theme-consumer", PkThemeConsumer);
