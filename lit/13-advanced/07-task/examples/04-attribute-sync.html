// Lit Task — Example 04
// Focus: attribute sync | Scenario: admin panel

import { LitElement, html } from "lit";
import { Task } from "@lit/task";

export class PkUserTask extends LitElement {
  static properties = { userId: { type: Number } };

  constructor() {
    super();
    this.userId = 1;
    this._task = new Task(this, {
      task: async ([id], { signal }) => {
        const res = await fetch(`/api/users/${id}`, { signal });
        if (!res.ok) throw new Error("Failed");
        return res.json();
      },
      args: () => [this.userId],
    });
  }

  render() {
    return this._task.render({
      pending: () => html`Loading…`,
      complete: (user) => html`<p>${user.name}</p>`,
      error: (e) => html`<p role="alert">${e.message}</p>`,
    });
  }
}
customElements.define("pk-user-task", PkUserTask);
