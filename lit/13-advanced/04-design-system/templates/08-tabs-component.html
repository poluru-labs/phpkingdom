// Design System with Lit — Template 08
// Focus: tabs component | Scenario: chat widget

import { LitElement, html, css } from "lit";

// Design-system primitive: tokens + part + slot + event
export class PkDsButton extends LitElement {
  static properties = {
    variant: { type: String, reflect: true },
    disabled: { type: Boolean, reflect: true },
  };

  static styles = css`
    :host {
      --pk-btn-bg: var(--pk-color-accent, #0f766e);
      --pk-btn-fg: #fff;
      display: inline-block;
    }
    :host([variant="ghost"]) {
      --pk-btn-bg: transparent;
      --pk-btn-fg: var(--pk-color-accent, #0f766e);
    }
    button {
      background: var(--pk-btn-bg);
      color: var(--pk-btn-fg);
      border: 1px solid var(--pk-color-accent, #0f766e);
      border-radius: var(--pk-radius, 8px);
      padding: 0.5rem 0.9rem;
    }
    button:disabled { opacity: 0.5; }
  `;

  render() {
    return html`
      <button part="button" ?disabled=${this.disabled}>
        <slot></slot>
      </button>
    `;
  }
}
customElements.define("pk-ds-button", PkDsButton);
