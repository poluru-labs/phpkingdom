// Lit Localization — Exercise 11
// Focus: default slot | Scenario: design system

import { LitElement, html } from "lit";
import { msg, str, updateWhenLocaleChanges } from "@lit/localize";

export class PkLocalized extends LitElement {
  static properties = { count: { type: Number } };

  constructor() {
    super();
    this.count = 3;
    updateWhenLocaleChanges(this);
  }

  render() {
    return html`
      <h1>${msg("Welcome")}</h1>
      <p>${msg(str`You have ${this.count} messages`)}</p>
      <button>${msg("Continue", { desc: "Primary CTA" })}</button>
    `;
  }
}
customElements.define("pk-localized", PkLocalized);
