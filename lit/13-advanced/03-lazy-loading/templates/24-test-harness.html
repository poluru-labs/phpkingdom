// Lazy Loading — Template 24
// Focus: test harness | Scenario: admin panel

// Eager shell + lazy heavy widget
const btn = document.querySelector("#open-chart");
const host = document.querySelector("#chart-host");

btn?.addEventListener("click", async () => {
  await import("./pk-chart.js"); // code-split chunk
  if (!host.querySelector("pk-chart")) {
    host.append(document.createElement("pk-chart"));
  }
});

// Or define lazily:
// customElements.define('pk-chart', (await import('./pk-chart.js')).PkChart);
