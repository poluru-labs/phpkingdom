// Property Options — Exercise 21
// Focus: decorator prop | Scenario: design system

import { LitElement, html } from "lit";

export class PkOptionDemo extends LitElement {
  static properties = {
    title: { type: String, attribute: "heading" },
    open: { type: Boolean, reflect: true },
    items: { type: Array, state: true },
    size: {
      type: Number,
      hasChanged(newVal, oldVal) {
        return newVal !== oldVal && newVal > 0;
      },
    },
  };

  constructor() {
    super();
    this.title = "Panel";
    this.open = false;
    this.items = [];
    this.size = 1;
  }

  render() {
    return html`<h3>${this.title} (${this.size})</h3>`;
  }
}
customElements.define("pk-option-demo", PkOptionDemo);
