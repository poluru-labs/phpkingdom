// Property Converters — Exercise 13
// Focus: static styles | Scenario: marketing site

import { LitElement, html } from "lit";

const jsonConverter = {
  fromAttribute(value) {
    try {
      return value ? JSON.parse(value) : {};
    } catch {
      return {};
    }
  },
  toAttribute(value) {
    return JSON.stringify(value ?? {});
  },
};

export class PkConfigHost extends LitElement {
  static properties = {
    config: { attribute: "config", converter: jsonConverter, reflect: true },
  };

  constructor() {
    super();
    this.config = { theme: "teal", density: "comfortable" };
  }

  render() {
    return html`<pre>${JSON.stringify(this.config, null, 2)}</pre>`;
  }
}
customElements.define("pk-config-host", PkConfigHost);
// HTML: <pk-config-host config='{"theme":"dark"}'></pk-config-host>
