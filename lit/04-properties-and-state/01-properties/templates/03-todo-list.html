// Lit Properties — Template 03
// Focus: todo list | Scenario: marketing site

import { LitElement, html } from "lit";

export class PkProps extends LitElement {
  static properties = {
    title: { type: String },
    count: { type: Number },
    flags: { type: Object },
  };

  constructor() {
    super();
    this.title = "Props";
    this.count = 0;
    this.flags = { featured: true };
  }

  render() {
    return html`
      <h3>${this.title}</h3>
      <p>${this.count}</p>
      <p>${this.flags.featured ? "Featured" : "Normal"}</p>
    `;
  }
}
customElements.define("pk-props", PkProps);
