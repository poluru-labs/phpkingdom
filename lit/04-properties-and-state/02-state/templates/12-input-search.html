// Lit State — Template 12
// Focus: input search | Scenario: dashboard widget

import { LitElement, html } from "lit";

export class PkPanel extends LitElement {
  static properties = {
    // Public API property
    title: { type: String },
    // Internal reactive state (not an attribute by default)
    open: { type: Boolean, state: true },
  };

  constructor() {
    super();
    this.title = "Panel";
    this.open = false;
  }

  render() {
    return html`
      <button @click=${() => (this.open = !this.open)}>${this.title}</button>
      ${this.open ? html`<div><slot></slot></div>` : null}
    `;
  }
}
customElements.define("pk-panel", PkPanel);
