// classMap Directive — Template 08
// Focus: tabs component | Scenario: chat widget

import { LitElement, html, css } from "lit";
import { classMap } from "lit/directives/class-map.js";

export class PkTile extends LitElement {
  static properties = {
    active: { type: Boolean },
    dense: { type: Boolean },
  };

  static styles = css`
    .tile { padding: 1rem; border: 1px solid #cbd5e1; }
    .active { outline: 2px solid #0f766e; }
    .dense { padding: 0.4rem; }
  `;

  render() {
    return html`
      <div
        class=${classMap({
          tile: true,
          active: this.active,
          dense: this.dense,
        })}
      >
        <slot></slot>
      </div>
    `;
  }
}
customElements.define("pk-tile", PkTile);
