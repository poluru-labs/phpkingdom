// live Directive — Template 24
// Focus: test harness | Scenario: admin panel

import { LitElement, html } from "lit";
import { live } from "lit/directives/live.js";

export class PkLiveInput extends LitElement {
  static properties = { value: { type: String } };

  constructor() {
    super();
    this.value = "phpkingdom.com";
  }

  #reset() {
    // Force DOM property to match even if user typed ahead
    this.value = "phpkingdom.com";
    this.requestUpdate();
  }

  render() {
    return html`
      <input .value=${live(this.value)} @input=${(e) => (this.value = e.target.value)} />
      <button @click=${this.#reset}>Reset</button>
    `;
  }
}
customElements.define("pk-live-input", PkLiveInput);
// live() checks the live DOM value before binding
