// Lit Directives — Template 19
// Focus: router outlet | Scenario: settings page

import { LitElement, html } from "lit";
import { classMap } from "lit/directives/class-map.js";
import { when } from "lit/directives/when.js";

export class PkDirectivesOverview extends LitElement {
  static properties = { on: { type: Boolean } };

  constructor() {
    super();
    this.on = false;
  }

  render() {
    return html`
      <div class=${classMap({ on: this.on })}>Directives demo</div>
      ${when(this.on, () => html`<em>Active</em>`)}
      <button @click=${() => (this.on = !this.on)}>Toggle</button>
    `;
  }
}
customElements.define("pk-directives-overview", PkDirectivesOverview);
