// styleMap Directive — Example 24
// Focus: ssr note | Scenario: admin panel

import { LitElement, html } from "lit";
import { styleMap } from "lit/directives/style-map.js";

export class PkMeter extends LitElement {
  static properties = {
    value: { type: Number },
    color: { type: String },
  };

  constructor() {
    super();
    this.value = 40;
    this.color = "#0f766e";
  }

  render() {
    return html`
      <div
        style=${styleMap({
          width: `${this.value}%`,
          background: this.color,
          height: "8px",
          borderRadius: "4px",
        })}
      ></div>
    `;
  }
}
customElements.define("pk-meter", PkMeter);
