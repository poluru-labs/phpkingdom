// Focus: pagination | Scenario: media player

import { LitElement, html } from "lit";
import { choose } from "lit/directives/choose.js";

export class PkStatus extends LitElement {
  static properties = { status: { type: String } };

  constructor() {
    super();
    this.status = "idle";
  }

  render() {
    return html`
      ${choose(this.status, [
        ["idle", () => html`<p>Ready</p>`],
        ["loading", () => html`<p>Working…</p>`],
        ["error", () => html`<p role="alert">Failed</p>`],
      ], () => html`<p>Unknown</p>`)}
    `;
  }
}
customElements.define("pk-status", PkStatus);
