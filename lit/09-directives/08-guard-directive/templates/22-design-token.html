// guard Directive — Template 22

import { LitElement, html } from "lit";
import { guard } from "lit/directives/guard.js";

export class PkExpensiveList extends LitElement {
  static properties = {
    items: { type: Array },
    filter: { type: String },
  };

  constructor() {
    super();
    this.items = ["Lit", "Vue", "React"];
    this.filter = "";
  }

  render() {
    return html`
      <input @input=${(e) => (this.filter = e.target.value)} />
      ${guard([this.items, this.filter], () => {
        const q = this.filter.toLowerCase();
        const filtered = this.items.filter((i) => i.toLowerCase().includes(q));
        return html`<ul>${filtered.map((i) => html`<li>${i}</li>`)}</ul>`;
      })}
    `;
  }
}
customElements.define("pk-expensive-list", PkExpensiveList);
// guard only re-renders the template when dependency values change
