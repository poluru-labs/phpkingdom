// repeat Directive — Example 03
// Focus: reactive property | Scenario: marketing site

import { LitElement, html } from "lit";
import { repeat } from "lit/directives/repeat.js";

export class PkUsers extends LitElement {
  static properties = { users: { type: Array } };

  constructor() {
    super();
    this.users = [
      { id: "a1", name: "Ada" },
      { id: "b2", name: "Grace" },
    ];
  }

  render() {
    return html`
      <ul>
        ${repeat(
          this.users,
          (u) => u.id,
          (u) => html`<li>${u.name}</li>`
        )}
      </ul>
    `;
  }
}
customElements.define("pk-users", PkUsers);
// Keyed repeat preserves DOM identity across reorder
