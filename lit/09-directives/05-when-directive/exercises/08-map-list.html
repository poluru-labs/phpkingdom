// when Directive — Exercise 08
// Focus: map list | Scenario: chat widget

import { LitElement, html } from "lit";
import { when } from "lit/directives/when.js";

export class PkLoader extends LitElement {
  static properties = {
    loading: { type: Boolean },
    data: { type: String },
  };

  constructor() {
    super();
    this.loading = true;
    this.data = "";
  }

  render() {
    return html`
      ${when(
        this.loading,
        () => html`<p>Loading…</p>`,
        () => html`<p>${this.data}</p>`
      )}
    `;
  }
}
customElements.define("pk-loader", PkLoader);
