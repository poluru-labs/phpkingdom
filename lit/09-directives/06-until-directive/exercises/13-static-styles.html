// until Directive — Exercise 13
// Focus: static styles | Scenario: marketing site

import { LitElement, html } from "lit";
import { until } from "lit/directives/until.js";

async function fetchTitle() {
  const res = await fetch("/api/title");
  const data = await res.json();
  return data.title;
}

export class PkUntilDemo extends LitElement {
  #promise = fetchTitle();

  render() {
    return html`
      <h2>
        ${until(this.#promise, html`<span>Loading title…</span>`)}
      </h2>
    `;
  }
}
customElements.define("pk-until-demo", PkUntilDemo);
// until renders placeholder until the Promise resolves
