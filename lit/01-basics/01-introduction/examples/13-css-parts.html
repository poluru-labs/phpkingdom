// Lit builds fast, lightweight web components
// Install: npm install lit

import { LitElement, html } from "lit";

export class PkIntro extends LitElement {
  render() {
    return html`<p>Welcome to Lit · phpkingdom.com</p>`;
  }
}
customElements.define("pk-intro", PkIntro);
