// Lit - Basics — Exercise 09
// Focus: click handler | Scenario: settings page

import { LitElement, html, css } from "lit";

export class PkBasics extends LitElement {
  static properties = { msg: { type: String } };
  static styles = css`:host { display: block; font-family: system-ui; }`;

  constructor() {
    super();
    this.msg = "Lit basics";
  }

  render() {
    return html`<strong>${this.msg}</strong>`;
  }
}
customElements.define("pk-basics", PkBasics);
// Class + properties + styles + render + customElements.define
