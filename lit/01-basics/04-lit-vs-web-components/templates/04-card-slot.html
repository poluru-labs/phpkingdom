// Lit vs Web Components — Template 04
// Focus: card slot | Scenario: admin panel

// Lit = thin reactive layer on top of native Custom Elements
// Vanilla WC = manual attributeChangedCallback + shadow DOM plumbing

// Vanilla sketch:
// class XButton extends HTMLElement {
//   connectedCallback() { this.attachShadow({mode:'open'}).innerHTML = '...'; }
// }

// Lit sketch:
import { LitElement, html, css } from "lit";

export class PkButton extends LitElement {
  static properties = { label: { type: String } };
  static styles = css`:host { display: inline-block; }`;
  render() {
    return html`<button part="button">${this.label}</button>`;
  }
}
customElements.define("pk-button", PkButton);
// Lit gives reactive properties, declarative templates, and batched updates
