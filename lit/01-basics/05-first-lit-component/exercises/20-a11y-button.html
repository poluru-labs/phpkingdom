// First Lit Component — Exercise 20
// Focus: a11y button | Scenario: ecommerce card

import { LitElement, html } from "lit";

export class HelloLit extends LitElement {
  render() {
    return html`<p>Hello phpkingdom.com</p>`;
  }
}

customElements.define("hello-lit", HelloLit);

// Usage:
// <script type="module" src="./hello-lit.js"></script>
// <hello-lit></hello-lit>
