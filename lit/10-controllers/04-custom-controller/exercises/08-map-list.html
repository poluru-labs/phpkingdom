// Custom Controller — Exercise 08
// Focus: map list | Scenario: chat widget

import { LitElement, html } from "lit";

class MouseController {
  x = 0;
  y = 0;
  constructor(host) {
    this.host = host;
    host.addController(this);
  }
  #onMove = (e) => {
    this.x = e.clientX;
    this.y = e.clientY;
    this.host.requestUpdate();
  };
  hostConnected() {
    window.addEventListener("pointermove", this.#onMove);
  }
  hostDisconnected() {
    window.removeEventListener("pointermove", this.#onMove);
  }
}

export class PkMouse extends LitElement {
  constructor() {
    super();
    this.mouse = new MouseController(this);
  }
  render() {
    return html`<p>${this.mouse.x}, ${this.mouse.y}</p>`;
  }
}
customElements.define("pk-mouse", PkMouse);
