// Reactive Controllers — Example 08
// Focus: event listener | Scenario: chat widget

import { LitElement, html } from "lit";

class ClockController {
  constructor(host) {
    this.host = host;
    host.addController(this);
    this.value = new Date();
  }

  hostConnected() {
    this._t = setInterval(() => {
      this.value = new Date();
      this.host.requestUpdate();
    }, 1000);
  }

  hostDisconnected() {
    clearInterval(this._t);
  }
}

export class PkControlledClock extends LitElement {
  constructor() {
    super();
    this.clock = new ClockController(this);
  }

  render() {
    return html`<time>${this.clock.value.toLocaleTimeString()}</time>`;
  }
}
customElements.define("pk-controlled-clock", PkControlledClock);
