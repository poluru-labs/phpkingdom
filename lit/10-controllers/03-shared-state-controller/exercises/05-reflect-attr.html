// Shared State Controller — Exercise 05
// Focus: reflect attr | Scenario: docs site

import { LitElement, html } from "lit";

const listeners = new Set();
let count = 0;

export const store = {
  get count() {
    return count;
  },
  inc() {
    count += 1;
    listeners.forEach((l) => l());
  },
  subscribe(fn) {
    listeners.add(fn);
    return () => listeners.delete(fn);
  },
};

class StoreController {
  constructor(host) {
    this.host = host;
    host.addController(this);
  }
  hostConnected() {
    this._unsub = store.subscribe(() => this.host.requestUpdate());
  }
  hostDisconnected() {
    this._unsub?.();
  }
}

export class PkStoreView extends LitElement {
  constructor() {
    super();
    this._store = new StoreController(this);
  }
  render() {
    return html`
      <span>${store.count}</span>
      <button @click=${() => store.inc()}>+</button>
    `;
  }
}
customElements.define("pk-store-view", PkStoreView);
