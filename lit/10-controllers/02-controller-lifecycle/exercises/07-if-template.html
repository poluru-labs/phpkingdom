// Controller Lifecycle — Exercise 07
// Focus: if template | Scenario: media player

import { ReactiveController } from "lit";

export class FetchController {
  constructor(host, url) {
    this.host = host;
    this.url = url;
    this.data = null;
    this.error = null;
    host.addController(this);
  }

  hostConnected() {
    this._abort = new AbortController();
    fetch(this.url, { signal: this._abort.signal })
      .then((r) => r.json())
      .then((data) => {
        this.data = data;
        this.host.requestUpdate();
      })
      .catch((err) => {
        if (err.name !== "AbortError") {
          this.error = err;
          this.host.requestUpdate();
        }
      });
  }

  hostDisconnected() {
    this._abort?.abort();
  }

  hostUpdate() {}
  hostUpdated() {}
}
// Controllers hook hostConnected / hostDisconnected / hostUpdate / hostUpdated
