// Lit Forms — Template 25
// Focus: production starter | Scenario: docs site

import { LitElement, html } from "lit";

export class PkLoginForm extends LitElement {
  static properties = {
    email: { type: String },
    password: { type: String },
  };

  constructor() {
    super();
    this.email = "";
    this.password = "";
  }

  #submit(e) {
    e.preventDefault();
    this.dispatchEvent(
      new CustomEvent("pk-login", {
        detail: { email: this.email },
        bubbles: true,
        composed: true,
      })
    );
  }

  render() {
    return html`
      <form @submit=${this.#submit}>
        <input
          type="email"
          .value=${this.email}
          @input=${(e) => (this.email = e.target.value)}
          required
        />
        <input
          type="password"
          .value=${this.password}
          @input=${(e) => (this.password = e.target.value)}
          required
        />
        <button type="submit">Sign in</button>
      </form>
    `;
  }
}
customElements.define("pk-login-form", PkLoginForm);
