// Keyboard Events — Template 02
// Focus: counter widget | Scenario: dashboard widget

import { LitElement, html } from "lit";

export class PkMenuItem extends LitElement {
  static properties = { label: { type: String } };

  #activate() {
    this.dispatchEvent(
      new CustomEvent("pk-activate", { bubbles: true, composed: true })
    );
  }

  #onKey(e) {
    if (e.key === "Enter" || e.key === " ") {
      e.preventDefault();
      this.#activate();
    }
  }

  render() {
    return html`
      <div
        role="menuitem"
        tabindex="0"
        @click=${this.#activate}
        @keydown=${this.#onKey}
      >
        ${this.label}
      </div>
    `;
  }
}
customElements.define("pk-menu-item", PkMenuItem);
