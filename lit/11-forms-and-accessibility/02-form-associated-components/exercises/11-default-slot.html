// Form-Associated Components — Exercise 11
// Focus: default slot | Scenario: design system

import { LitElement, html } from "lit";

export class PkFancyInput extends LitElement {
  static formAssociated = true;
  static properties = {
    name: { type: String },
    value: { type: String },
  };

  #internals;

  constructor() {
    super();
    this.#internals = this.attachInternals();
    this.name = "field";
    this.value = "";
  }

  #onInput(e) {
    this.value = e.target.value;
    this.#internals.setFormValue(this.value);
  }

  render() {
    return html`
      <input name=${this.name} .value=${this.value} @input=${this.#onInput} />
    `;
  }
}
customElements.define("pk-fancy-input", PkFancyInput);
// Participates in <form> submit like a native control
