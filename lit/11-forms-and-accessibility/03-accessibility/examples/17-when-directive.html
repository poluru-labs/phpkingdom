// Lit Accessibility — Example 17
// Focus: when directive | Scenario: media player

import { LitElement, html } from "lit";

export class PkA11yDialog extends LitElement {
  static properties = { open: { type: Boolean, reflect: true } };

  #close() {
    this.open = false;
  }

  render() {
    if (!this.open) return html``;
    return html`
      <div
        role="dialog"
        aria-modal="true"
        aria-labelledby="dlg-title"
        @keydown=${(e) => e.key === "Escape" && this.#close()}
      >
        <h2 id="dlg-title"><slot name="title">Dialog</slot></h2>
        <slot></slot>
        <button @click=${this.#close}>Close</button>
      </div>
    `;
  }
}
customElements.define("pk-a11y-dialog", PkA11yDialog);
// Roles, labels, keyboard, focus management matter for custom elements
