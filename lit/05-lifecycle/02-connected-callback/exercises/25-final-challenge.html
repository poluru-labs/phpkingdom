// connectedCallback — Exercise 25

import { LitElement, html } from "lit";

export class PkLiveClock extends LitElement {
  static properties = {
    now: { type: String, state: true },
  };

  #timer;

  connectedCallback() {
    super.connectedCallback();
    this.#tick();
    this.#timer = setInterval(() => this.#tick(), 1000);
  }

  disconnectedCallback() {
    clearInterval(this.#timer);
    super.disconnectedCallback();
  }

  #tick() {
    this.now = new Date().toLocaleTimeString();
  }

  render() {
    return html`<time>${this.now ?? "…"}</time>`;
  }
}
customElements.define("pk-live-clock", PkLiveClock);
