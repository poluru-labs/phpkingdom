// Lit Lifecycle — Exercise 12
// Focus: named slot | Scenario: dashboard widget

import { LitElement, html } from "lit";

export class PkLifecycle extends LitElement {
  static properties = { value: { type: Number } };

  constructor() {
    super();
    this.value = 0;
    // constructor → connectedCallback → render → firstUpdated → updated…
  }

  connectedCallback() {
    super.connectedCallback();
    console.log("connected");
  }

  willUpdate(changed) {
    if (changed.has("value")) console.log("willUpdate", this.value);
  }

  updated(changed) {
    if (changed.has("value")) console.log("updated", this.value);
  }

  disconnectedCallback() {
    console.log("disconnected");
    super.disconnectedCallback();
  }

  render() {
    return html`<p>${this.value}</p>`;
  }
}
customElements.define("pk-lifecycle", PkLifecycle);
