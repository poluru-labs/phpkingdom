import { LitElement, html } from "lit";

export class PkResizeAware extends LitElement {
  #ro;

  connectedCallback() {
    super.connectedCallback();
    this.#ro = new ResizeObserver(() => {
      this.setAttribute("data-width", String(this.clientWidth));
    });
    this.#ro.observe(this);
  }

  disconnectedCallback() {
    this.#ro?.disconnect();
    this.#ro = undefined;
    super.disconnectedCallback();
  }

  render() {
    return html`<slot></slot>`;
  }
}
customElements.define("pk-resize-aware", PkResizeAware);
// Always tear down listeners/observers in disconnectedCallback
