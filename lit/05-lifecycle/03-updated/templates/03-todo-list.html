// updated() Method — Template 03
// Focus: todo list | Scenario: marketing site

import { LitElement, html } from "lit";

export class PkUpdated extends LitElement {
  static properties = {
    query: { type: String },
  };

  constructor() {
    super();
    this.query = "";
  }

  updated(changedProperties) {
    if (changedProperties.has("query")) {
      // Side effects after DOM update
      this.dispatchEvent(
        new CustomEvent("pk-query-change", {
          detail: { query: this.query },
          bubbles: true,
          composed: true,
        })
      );
    }
  }

  render() {
    return html`
      <input
        .value=${this.query}
        @input=${(e) => (this.query = e.target.value)}
      />
    `;
  }
}
customElements.define("pk-updated", PkUpdated);
