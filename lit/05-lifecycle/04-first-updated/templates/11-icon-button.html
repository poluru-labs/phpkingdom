// firstUpdated() Method — Template 11
// Focus: icon button | Scenario: design system

import { LitElement, html } from "lit";

export class PkAutofocus extends LitElement {
  static properties = { placeholder: { type: String } };

  constructor() {
    super();
    this.placeholder = "Search…";
  }

  firstUpdated() {
    // DOM is ready once; safe for one-time focus / measurement
    this.renderRoot.querySelector("input")?.focus();
  }

  render() {
    return html`<input type="search" placeholder=${this.placeholder} />`;
  }
}
customElements.define("pk-autofocus", PkAutofocus);
