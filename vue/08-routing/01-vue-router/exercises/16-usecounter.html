// Vue Router — Exercise 16
// Focus: useCounter | Scenario: admin console

import { createRouter, createWebHistory } from "vue-router";
import Home from "./views/Home.vue";
import About from "./views/About.vue";

export default createRouter({
  history: createWebHistory(),
  routes: [
    { path: "/", component: Home },
    { path: "/about", component: About },
  ],
});

Practice Notes

Checklist for Vue Router
- Practice: useCounter
- Scenario: admin console
- Prefer <script setup> + Composition API for Vue 3
- Use ref for primitives, reactive for objects; computed for derived state
- Key lists with stable ids; v-if for costly trees, v-show for toggles
- Props down / emits up; Pinia or provide/inject for shared state
- Route with vue-router; guard auth; fetch with composables + loading/error

Useful commands:
  npm create vue@latest
  npm install
  npm run dev
  npm run build
  npm run type-check   # vue-tsc when TypeScript enabled
