Modal to body

Scaffold

<script setup>
import { ref } from "vue";
const open = ref(false);
</script>

<template>
  <button @click="open = true">Open</button>
  <Teleport to="body">
    <div v-if="open" class="modal" role="dialog">
      <p>Modal</p>
      <button @click="open = false">Close</button>
    </div>
  </Teleport>
</template>
