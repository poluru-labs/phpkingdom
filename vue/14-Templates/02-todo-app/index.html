Classic todo list

Scaffold

<script setup>
import { ref } from "vue";
const text = ref("");
const todos = ref([]);
function add() {
  if (!text.value.trim()) return;
  todos.value.push({ id: Date.now(), text: text.value.trim() });
  text.value = "";
}
</script>

<template>
  <form @submit.prevent="add">
    <input v-model="text" />
    <button>Add</button>
  </form>
  <ul>
    <li v-for="t in todos" :key="t.id">{{ t.text }}</li>
  </ul>
</template>
