Fetch + loading state

Scaffold

<script setup>
import { ref, onMounted } from "vue";
const users = ref([]);
const loading = ref(true);
onMounted(async () => {
  users.value = await fetch("/api/users").then((r) => r.json());
  loading.value = false;
});
</script>
