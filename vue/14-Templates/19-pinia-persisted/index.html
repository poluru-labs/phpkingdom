Scaffold

pinia.use(({ store }) => {
  const key = `pinia:${store.$id}`;
  const raw = localStorage.getItem(key);
  if (raw) store.$patch(JSON.parse(raw));
  store.$subscribe((_, state) => localStorage.setItem(key, JSON.stringify(state)));
});
