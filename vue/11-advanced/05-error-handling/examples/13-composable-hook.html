Error Handling

Section: Advanced

Vue.js

// Error Handling — Example 13
// Focus: composable hook | Scenario: todo app

<script setup>
import { ref, computed, onMounted } from "vue";

const count = ref(0);
const title = ref("Vue — {token}");
const doubled = computed(() => count.value * 2);

function increment() {
  count.value += 1;
}

onMounted(() => {
  // Focus: {focus} | Scenario: {scenario}
  console.log("mounted", title.value);
});
</script>

<template>
  <main>
    <h1>{{ title }}</h1>
    <p>Count: {{ count }} · Doubled: {{ doubled }}</p>
    <button @click="increment">+1</button>
  </main>
</template>

// main.js
import { createApp } from "vue";
import App from "./App.vue";
createApp(App).mount("#app");

Practice Notes

Checklist for Error Handling
- Practice: composable hook
- Scenario: todo app
- Prefer <script setup> + Composition API for Vue 3
- Use ref for primitives, reactive for objects; computed for derived state
- Key lists with stable ids; v-if for costly trees, v-show for toggles
- Props down / emits up; Pinia or provide/inject for shared state
- Route with vue-router; guard auth; fetch with composables + loading/error
