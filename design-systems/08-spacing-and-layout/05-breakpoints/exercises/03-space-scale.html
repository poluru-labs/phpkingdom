Design System

// Responsive Breakpoints — Exercise 03
// Focus: space scale | Scenario: enterprise portal

export const breakpoints = {
  sm: 640,
  md: 768,
  lg: 1024,
  xl: 1280,
} as const;

export function mq(bp: keyof typeof breakpoints) {
  return `@media (min-width: ${breakpoints[bp]}px)`;
}

/* Mobile-first: base styles, then mq("md") { ... } */
