// Keyboard Accessibility — Exercise 13
// Focus: prefers reduced | Scenario: enterprise portal

// Modal sketch: Tab cycles; Esc closes; return focus to trigger
export function onKeydown(e: KeyboardEvent, { close, first, last }: {
  close: () => void; first: HTMLElement; last: HTMLElement;
}) {
  if (e.key === "Escape") close();
  if (e.key === "Tab") {
    if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
    else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
  }
}
