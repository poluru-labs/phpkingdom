// Modal Component — Exercise 19
// Focus: visual snapshot | Scenario: checkout flow

<dialog class="ds-modal" id="confirm">
  <form method="dialog">
    <h2>Delete project?</h2>
    <p>This cannot be undone.</p>
    <button value="cancel">Cancel</button>
    <button value="confirm" class="ds-button" data-variant="danger">Delete</button>
  </form>
</dialog>
<script>
  document.getElementById("confirm").showModal();
  // Native <dialog> handles focus trap + Esc in modern browsers
</script>
