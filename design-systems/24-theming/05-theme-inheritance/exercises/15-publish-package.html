Design System

// Theme Inheritance — Exercise 15
// Focus: publish package | Scenario: mobile web

:root, [data-theme="light"] {
  --color-bg: #fff;
  --color-fg: #0f172a;
  --color-action: #2563eb;
}
[data-theme="dark"] {
  --color-bg: #0b1220;
  --color-fg: #e2e8f0;
  --color-action: #60a5fa;
}
/* Nested surfaces inherit; override locally with data-theme on a subtree */
.ds-surface[data-theme="dark"] { /* dark island in light app */ }
