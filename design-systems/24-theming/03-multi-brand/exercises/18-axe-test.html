Design System

// Multi-Brand Design Systems — Exercise 18
// Focus: axe test | Scenario: admin console

export const brandThemes = {
  acme: { "--color-action": "#2563eb", "--radius-md": "8px", "--font-sans": "Inter, sans-serif" },
  contoso: { "--color-action": "#059669", "--radius-md": "2px", "--font-sans": "Georgia, serif" },
};

export function setBrand(brand: keyof typeof brandThemes) {
  document.documentElement.dataset.brand = brand;
  Object.entries(brandThemes[brand]).forEach(([k, v]) =>
    document.documentElement.style.setProperty(k, v),
  );
}
