export function ThemeSwitch() {
  return (
    <button
      type="button"
      aria-label="Toggle color theme"
      onClick={() => {
        const root = document.documentElement;
        root.dataset.theme = root.dataset.theme === "dark" ? "light" : "dark";
      }}
    >
      Theme
    </button>
  );
}
