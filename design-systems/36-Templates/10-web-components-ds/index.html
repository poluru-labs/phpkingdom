import { LitElement, html, css } from "lit";
import { customElement, property } from "lit/decorators.js";

@customElement("pk-button")
export class PkButton extends LitElement {
  static styles = css`
    :host { display: inline-block; }
    button {
      padding: var(--space-2, 0.5rem) var(--space-4, 1rem);
      border-radius: var(--radius-md, 0.5rem);
      border: 0;
      background: var(--color-action, #2563eb);
      color: var(--color-action-fg, #fff);
    }
    button:focus-visible { box-shadow: var(--focus-ring); }
  `;
  @property({ type: String }) variant = "primary";
  render() {
    return html`<button part="button" data-variant=${this.variant}><slot></slot></button>`;
  }
}
