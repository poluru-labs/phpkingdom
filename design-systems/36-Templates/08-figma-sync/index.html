export const figmaToCode = {
  "color/action": "--color-action",
  "color/fg": "--color-fg",
  "color/bg": "--color-bg",
  "space/2": "--space-2",
  "space/4": "--space-4",
  "radius/md": "--radius-md",
};

export function assertParity(figmaNames: string[], cssVars: string[]) {
  const missing = figmaNames.filter((n) => !cssVars.includes(figmaToCode[n as keyof typeof figmaToCode]));
  if (missing.length) throw new Error(`Token drift: ${missing.join(", ")}`);
}
