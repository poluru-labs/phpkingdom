export const brands = {
  acme: { "--color-action": "#2563eb", "--radius-md": "8px" },
  contoso: { "--color-action": "#059669", "--radius-md": "2px" },
} as const;

export function applyBrand(brand: keyof typeof brands) {
  const root = document.documentElement;
  root.dataset.brand = brand;
  Object.entries(brands[brand]).forEach(([k, v]) => root.style.setProperty(k, v));
}
