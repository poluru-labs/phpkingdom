export function Icon({ name, size = 20 }: { name: string; size?: number }) {
  return (
    <svg width={size} height={size} aria-hidden="true" focusable="false">
      <use href={`/icons.svg#${name}`} />
    </svg>
  );
}

export function IconButton({ label, name, ...rest }: React.ButtonHTMLAttributes<HTMLButtonElement> & { label: string; name: string }) {
  return (
    <button type="button" className="ds-icon-button" aria-label={label} {...rest}>
      <Icon name={name} />
    </button>
  );
}
