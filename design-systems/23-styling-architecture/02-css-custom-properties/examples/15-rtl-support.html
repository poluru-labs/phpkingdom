Design System

// CSS Custom Properties — Example 15
// Focus: rtl support | Scenario: mobile web

:root {
  --color-action: #2563eb;
  --space-4: 1rem;
}

.ds-button {
  background: var(--color-action);
  padding-inline: var(--space-4);
}

.ds-button {
  /* fallback example */
  background: var(--color-action, #2563eb);
}
