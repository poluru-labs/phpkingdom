Design System

// Tailwind + Design Tokens — Exercise 15
// Focus: publish package | Scenario: mobile web

// tailwind.config.ts — map DS tokens into theme
import type { Config } from "tailwindcss";

export default {
  content: ["./src/**/*.{ts,tsx}"],
  theme: {
    extend: {
      colors: {
        action: "var(--color-action)",
        fg: "var(--color-fg)",
        bg: "var(--color-bg)",
      },
      spacing: {
        1: "var(--space-1)",
        2: "var(--space-2)",
        4: "var(--space-4)",
      },
      borderRadius: { md: "var(--radius-md)" },
    },
  },
} satisfies Config;
