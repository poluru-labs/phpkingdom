// components/ui/button.tsx (sketch)
import { cva, type VariantProps } from "class-variance-authority";

const button = cva("inline-flex items-center rounded-md px-4 py-2 font-semibold", {
  variants: {
    variant: {
      primary: "bg-[var(--color-action)] text-[var(--color-action-fg)]",
      secondary: "border border-[var(--color-action)] text-[var(--color-action)]",
    },
  },
  defaultVariants: { variant: "primary" },
});

export type ButtonProps = React.ButtonHTMLAttributes<HTMLButtonElement> & VariantProps<typeof button>;
export function Button({ className, variant, ...props }: ButtonProps) {
  return <button className={button({ variant, className })} {...props} />;
}
