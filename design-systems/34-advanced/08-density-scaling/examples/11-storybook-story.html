Design System

// Density Scaling — Example 11
// Focus: storybook story | Scenario: product SaaS

:root, [data-density="comfortable"] {
  --control-height: 2.5rem;
  --control-pad-x: var(--space-4);
  --row-gap: var(--space-3);
}
[data-density="compact"] {
  --control-height: 2rem;
  --control-pad-x: var(--space-3);
  --row-gap: var(--space-2);
}

.ds-input, .ds-button {
  min-height: var(--control-height);
  padding-inline: var(--control-pad-x);
}
