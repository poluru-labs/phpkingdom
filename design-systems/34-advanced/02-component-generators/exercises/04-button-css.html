// Component Generators — Exercise 04
// Focus: button css | Scenario: marketing site

// plopfile.js sketch
export default function (plop) {
  plop.setGenerator("component", {
    description: "React DS component",
    prompts: [{ type: "input", name: "name", message: "Component name?" }],
    actions: [
      { type: "add", path: "src/{{pascalCase name}}/{{pascalCase name}}.tsx", templateFile: "plop/Component.tsx.hbs" },
      { type: "add", path: "src/{{pascalCase name}}/{{pascalCase name}}.stories.tsx", templateFile: "plop/Story.tsx.hbs" },
    ],
  });
}
