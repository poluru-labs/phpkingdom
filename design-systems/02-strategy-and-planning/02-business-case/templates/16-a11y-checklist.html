// Design System Business Case — Template 16
// Focus: a11y checklist | Scenario: component library

export function businessCase({ redesignsAvoided, engHours, designHours, hourly }) {
  const savings = (engHours + designHours) * hourly + redesignsAvoided * 40000;
  return {
    investment: "core team + tooling",
    annualSavingsEstimate: savings,
    risksIfSkip: ["inconsistent UX", "a11y debt", "duplicate components"],
  };
}

console.log(businessCase({ redesignsAvoided: 2, engHours: 1200, designHours: 600, hourly: 85 }));
