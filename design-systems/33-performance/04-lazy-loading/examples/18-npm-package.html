Design System

// Lazy Loading Components — Example 18
// Focus: npm package | Scenario: admin console

import { lazy, Suspense } from "react";

const RichTextEditor = lazy(() => import("@phpkingdom/react/rich-text"));

export function EditorRoute() {
  return (
    <Suspense fallback={<div role="status">Loading editor…</div>}>
      <RichTextEditor />
    </Suspense>
  );
}
