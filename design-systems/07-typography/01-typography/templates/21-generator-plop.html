Design System

// Design System Typography — Template 21
// Focus: generator plop | Scenario: product SaaS

:root {
  --font-sans: "Source Sans 3", system-ui, sans-serif;
  --font-mono: "IBM Plex Mono", ui-monospace, monospace;
  --font-size-body: 1rem;
  --font-size-h1: 2rem;
  --line-height-body: 1.5;
  --line-height-heading: 1.2;
  --font-weight-regular: 400;
  --font-weight-semibold: 600;
}

.ds-body { font: var(--font-weight-regular) var(--font-size-body)/var(--line-height-body) var(--font-sans); }
.ds-h1 { font: var(--font-weight-semibold) var(--font-size-h1)/var(--line-height-heading) var(--font-sans); }
