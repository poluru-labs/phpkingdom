Design System

// Responsive Typography — Exercise 04
// Focus: button css | Scenario: marketing site

:root { --fs-body: 1rem; --fs-display: 2rem; }

@media (min-width: 768px) {
  :root { --fs-body: 1.0625rem; --fs-display: 2.5rem; }
}

@media (min-width: 1024px) {
  :root { --fs-display: clamp(2.5rem, 2rem + 1vw, 3rem); }
}

.ds-display { font-size: var(--fs-display); line-height: 1.15; }
