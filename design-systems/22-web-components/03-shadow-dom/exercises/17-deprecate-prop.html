Design System

// Shadow DOM Styling — Exercise 17
// Focus: deprecate prop | Scenario: docs portal

/* Inside shadow root — :host + CSS variables pierce for theming */
:host {
  display: inline-block;
  color: var(--pk-button-fg, var(--color-action-fg, #fff));
}
:host([disabled]) { opacity: 0.5; pointer-events: none; }
button {
  background: var(--pk-button-bg, var(--color-action, #2563eb));
  border: 0;
  border-radius: var(--radius-md, 0.5rem);
}
