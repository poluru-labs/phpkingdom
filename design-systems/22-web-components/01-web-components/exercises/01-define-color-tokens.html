Design System

// Design System with Web Components — Exercise 01
// Focus: define color tokens | Scenario: product SaaS

class PkButton extends HTMLElement {
  static get observedAttributes() { return ["variant", "disabled"]; }
  connectedCallback() {
    this.attachShadow({ mode: "open" });
    this.render();
  }
  attributeChangedCallback() { this.render(); }
  render() {
    if (!this.shadowRoot) return;
    const variant = this.getAttribute("variant") || "primary";
    this.shadowRoot.innerHTML = `
      <style>:host{display:inline-block} button{padding:.5rem 1rem}</style>
      <button part="button" data-variant="${variant}" ${this.hasAttribute("disabled") ? "disabled" : ""}>
        <slot></slot>
      </button>`;
  }
}
customElements.define("pk-button", PkButton);
