Design System

// Storybook Documentation — Example 23
// Focus: token naming | Scenario: enterprise portal

import type { Meta } from "@storybook/react";
import { Button } from "./Button";

const meta: Meta<typeof Button> = {
  title: "Components/Button",
  component: Button,
  tags: ["autodocs"],
  parameters: {
    docs: {
      description: {
        component: "Use Button for actions. Prefer primary sparingly.",
      },
    },
  },
};
export default meta;
