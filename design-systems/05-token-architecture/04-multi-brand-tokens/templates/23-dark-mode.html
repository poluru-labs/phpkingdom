Design System

// Multi-Brand Tokens — Template 23
// Focus: dark mode | Scenario: enterprise portal

export const brands = {
  acme: { "color.action": "#2563eb", "radius.md": "8px" },
  contoso: { "color.action": "#059669", "radius.md": "2px" },
};

export function applyBrand(brand: keyof typeof brands) {
  const tokens = brands[brand];
  for (const [k, v] of Object.entries(tokens)) {
    document.documentElement.style.setProperty(`--${k.replace(/\./g, "-")}`, v);
  }
}
