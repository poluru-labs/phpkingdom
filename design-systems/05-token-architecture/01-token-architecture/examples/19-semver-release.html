// Layers: primitive → semantic → component
export const tokenLayers = {
  primitive: "color.blue.600",
  semantic: "color.action",
  component: "button.bg",
};

export function resolve(path: string, store: Record<string, string>) {
  let value = store[path];
  const seen = new Set<string>();
  while (value?.startsWith("{") && value.endsWith("}")) {
    if (seen.has(value)) throw new Error("cycle");
    seen.add(value);
    value = store[value.slice(1, -1)];
  }
  return value;
}
