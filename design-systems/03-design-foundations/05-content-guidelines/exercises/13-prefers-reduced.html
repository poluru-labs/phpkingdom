export const contentRules = {
  buttons: { prefer: "Save changes", avoid: "Submit", case: "sentence" },
  errors: { pattern: "What happened + how to fix", example: "Email is required. Enter a valid address." },
  empty: { pattern: "Explain + next action", example: "No projects yet. Create your first project." },
  tone: { do: ["active voice", "specific"], dont: ["jargon", "blame the user"] },
};
