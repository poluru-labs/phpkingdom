// Design Principles — Example 20
// Focus: deprecation note | Scenario: design ops team

export const principles = [
  { name: "Clarity", means: "prefer obvious over clever UI" },
  { name: "Accessibility", means: "ship keyboard + contrast by default" },
  { name: "Consistency", means: "reuse patterns before inventing" },
  { name: "Efficiency", means: "tokens and composition over one-offs" },
];

export function principleCheck(decision: string, principle: string) {
  return `Does "${decision}" uphold ${principle}?`;
}
