/* Design system basics: one source of truth for visual decisions */
:root {
  --ds-space-unit: 4px;
  --ds-font-sans: "Source Sans 3", system-ui, sans-serif;
  --ds-color-fg: #0f172a;
  --ds-color-bg: #ffffff;
}

.ds-surface {
  font-family: var(--ds-font-sans);
  color: var(--ds-color-fg);
  background: var(--ds-color-bg);
  padding: calc(var(--ds-space-unit) * 4);
}
