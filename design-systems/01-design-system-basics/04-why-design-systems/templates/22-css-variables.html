// Why invest? Speed, consistency, a11y, and lower redesign cost
export function roiSketch(products: number, hoursSavedPerSprint: number) {
  const sprintsPerYear = 26;
  return {
    products,
    hoursSavedYear: products * hoursSavedPerSprint * sprintsPerYear,
    qualityWins: ["consistent UX", "shared a11y patterns", "faster onboarding"],
  };
}

console.table(roiSketch(8, 6));
