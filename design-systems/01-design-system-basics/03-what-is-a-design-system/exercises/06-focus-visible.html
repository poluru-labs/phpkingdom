Design System

// What Is a Design System? — Exercise 06
// Focus: focus visible | Scenario: component library

/**
 * A design system is the shared set of standards, tokens, components,
 * and processes that help teams build consistent products faster.
 */
export type DesignSystem = {
  tokens: Record<string, string | number>;
  components: string[];
  guidelines: string[];
  processes: ("contribute" | "review" | "release" | "deprecate")[];
};

export const phpKingdomDS: DesignSystem = {
  tokens: { "color.action": "#2563eb", "space.4": "1rem" },
  components: ["Button", "Input", "Modal"],
  guidelines: ["Do use semantic tokens", "Don't hardcode hex in components"],
  processes: ["contribute", "review", "release", "deprecate"],
};
