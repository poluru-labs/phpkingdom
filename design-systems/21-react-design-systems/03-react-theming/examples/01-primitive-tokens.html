// React Theming — Example 01
// Focus: primitive tokens | Scenario: product SaaS

import { createContext, useMemo, useState } from "react";

type Tokens = { action: string; bg: string; fg: string };
const light: Tokens = { action: "#2563eb", bg: "#fff", fg: "#0f172a" };
const dark: Tokens = { action: "#60a5fa", bg: "#0b1220", fg: "#e2e8f0" };

export const TokenContext = createContext(light);

export function ThemeProvider({ children }: { children: React.ReactNode }) {
  const [mode, setMode] = useState<"light" | "dark">("light");
  const value = useMemo(() => (mode === "light" ? light : dark), [mode]);
  return (
    <TokenContext.Provider value={value}>
      <button type="button" onClick={() => setMode((m) => (m === "light" ? "dark" : "light"))}>
        Toggle
      </button>
      <div style={{ background: value.bg, color: value.fg }}>{children}</div>
    </TokenContext.Provider>
  );
}
