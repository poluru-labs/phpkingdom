Design System

// React with TypeScript — Exercise 02
// Focus: semantic alias | Scenario: multi-brand retail

export type Variant = "primary" | "secondary" | "ghost" | "danger";
export type Size = "sm" | "md" | "lg";

export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: Variant;
  size?: Size;
  isLoading?: boolean;
}

export function Button({ variant = "primary", size = "md", isLoading, children, disabled, ...rest }: ButtonProps) {
  return (
    <button data-variant={variant} data-size={size} disabled={disabled || isLoading} {...rest}>
      {isLoading ? "Loading…" : children}
    </button>
  );
}
