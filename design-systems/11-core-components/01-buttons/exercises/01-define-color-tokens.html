// Button Component — Exercise 01
// Focus: define color tokens | Scenario: product SaaS

export function Button({
  variant = "primary",
  children,
  ...rest
}: React.ButtonHTMLAttributes<HTMLButtonElement> & { variant?: "primary" | "secondary" }) {
  return (
    <button type="button" className="ds-button" data-variant={variant} {...rest}>
      {children}
    </button>
  );
}
