// Icon Component — Template 03
// Focus: form kit | Scenario: enterprise portal

export function IconButton({ label, children, ...rest }: React.ButtonHTMLAttributes<HTMLButtonElement> & { label: string }) {
  return (
    <button type="button" className="ds-icon-button" aria-label={label} {...rest}>
      {children}
    </button>
  );
}

// Decorative icons inside labeled buttons: aria-hidden="true"
