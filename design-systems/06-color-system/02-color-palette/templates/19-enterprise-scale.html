  700: "#1d4ed8",
  800: "#1e40af",
  900: "#1e3a8a",
};

export function cssVars(name: string, scale: Record<number, string>) {
  return Object.entries(scale)
    .map(([step, hex]) => `--${name}-${step}: ${hex};`)
    .join("\n");
}
