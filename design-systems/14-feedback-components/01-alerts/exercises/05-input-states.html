Design System

// Alert Component — Exercise 05
// Focus: input states | Scenario: mobile web

<div class="ds-alert" data-tone="danger" role="alert">
  <strong>Payment failed.</strong> Check your card details and try again.
</div>
<div class="ds-alert" data-tone="info" role="status">
  Your invite was sent.
</div>
/* role=alert for urgent; role=status for polite updates */
