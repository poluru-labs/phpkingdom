// Toast Component — Example 07
// Focus: button variants | Scenario: docs portal

export function toast(message: string, tone: "info" | "success" | "danger" = "info") {
  const el = document.createElement("div");
  el.className = "ds-toast";
  el.dataset.tone = tone;
  el.setAttribute("role", tone === "danger" ? "alert" : "status");
  el.textContent = message;
  document.getElementById("toast-region")!.append(el);
  setTimeout(() => el.remove(), 4000);
}
