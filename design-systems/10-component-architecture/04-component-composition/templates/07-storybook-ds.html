// Component Composition — Template 07
// Focus: storybook ds | Scenario: docs portal

export function Card({ children }: { children: React.ReactNode }) {
  return <section className="ds-card">{children}</section>;
}
export function CardHeader({ children }: { children: React.ReactNode }) {
  return <header className="ds-card__header">{children}</header>;
}
export function CardBody({ children }: { children: React.ReactNode }) {
  return <div className="ds-card__body">{children}</div>;
}

// Compose: <Card><CardHeader/><CardBody/></Card> instead of mega-props

Checklist for Component Composition

- Scenario: docs portal
- Separate primitive → semantic → component tokens
- Document usage, do/don't, and accessibility in Storybook
- Version with semver; deprecate before breaking
- Measure adoption and bundle impact
