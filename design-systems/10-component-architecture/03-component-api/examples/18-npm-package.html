Design System

// Component API Design — Example 18
// Focus: npm package | Scenario: admin console

export type ButtonProps = {
  variant?: "primary" | "secondary" | "ghost" | "danger";
  size?: "sm" | "md" | "lg";
  disabled?: boolean;
  type?: "button" | "submit" | "reset";
  children: React.ReactNode;
  onClick?: () => void;
  "aria-label"?: string;
};

// Prefer boolean flags sparingly; prefer variant enums
// Never expose raw CSS color props — use tokens/variants

Checklist for Component API Design

- Version with semver; deprecate before breaking
- Measure adoption and bundle impact
