Design System

// Component Architecture — Example 08
// Focus: form field | Scenario: admin console

export type ComponentContract = {
  name: string;
  anatomy: string[];
  variants: string[];
  slots?: string[];
  a11y: string[];
  tokens: string[];
};

export const ButtonContract: ComponentContract = {
  name: "Button",
  anatomy: ["root", "icon", "label"],
  variants: ["primary", "secondary", "ghost", "danger"],
  a11y: ["native button", "focus-visible", "disabled semantics"],
  tokens: ["button.bg", "button.fg", "button.radius"],
};

- Measure adoption and bundle impact
