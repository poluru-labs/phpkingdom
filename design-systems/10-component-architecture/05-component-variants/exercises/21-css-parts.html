// Component Variants — Exercise 21

.ds-button[data-variant="primary"] { background: var(--color-action); color: var(--color-action-fg); }
.ds-button[data-variant="secondary"] {
  background: transparent; color: var(--color-action);
  box-shadow: inset 0 0 0 1px var(--color-action);
}
.ds-button[data-variant="ghost"] { background: transparent; color: var(--color-fg); }
.ds-button[data-size="sm"] { padding: var(--space-1) var(--space-2); font-size: 0.875rem; }
.ds-button[data-size="lg"] { padding: var(--space-3) var(--space-6); font-size: 1.125rem; }
