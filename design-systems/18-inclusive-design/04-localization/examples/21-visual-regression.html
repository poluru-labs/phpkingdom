Design System

// Localization — Example 21
// Focus: visual regression | Scenario: product SaaS

export const copy = {
  "button.save": { en: "Save", es: "Guardar", de: "Speichern" },
  "error.required": { en: "Required", es: "Obligatorio", de: "Erforderlich" },
};

export function t(key: keyof typeof copy, locale: "en" | "es" | "de") {
  return copy[key][locale];
}
/* Design for text growth (~30–40%) and date/number formats */
