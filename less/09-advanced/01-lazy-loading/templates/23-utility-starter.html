// Lazy Loading Variables — Template 23

.box {
  color: @brand; // uses the *last* @brand in scope (lazy evaluation)
}

@brand: #0d6efd;
@brand: #6610f2; // wins at use time

// Unlike CSS custom properties, LESS vars resolve at compile time
// but still "lazy" within the file/scope
