// CSS Custom Properties — Template 01
// Focus: starter file | Scenario: SaaS dashboard

@brand: #0d6efd;
@ink: #212529;
@radius: 12px;

:root {
  --brand: @brand;
  --ink: @ink;
  --radius: @radius;
}

.btn {
  background: var(--brand);
  color: #fff;
  border-radius: var(--radius);
}
