// CSS Custom Properties — Exercise 01
// Focus: variables challenge | Scenario: SaaS dashboard

@brand: #0d6efd;
@ink: #212529;
@radius: 12px;

:root {
  --brand: @brand;
  --ink: @ink;
  --radius: @radius;
}

.btn {
  background: var(--brand);
  color: #fff;
  border-radius: var(--radius);
}
