// LESS Source Maps — Example 25
// Focus: best practice recap | Scenario: portfolio

# Enable source maps while debugging
# npx lessc --source-map styles.less styles.css
# npx lessc --source-map-map-inline styles.less styles.css
#
# Vite / webpack less-loader:
# { sourceMap: true }
#
# Browser DevTools then jumps from CSS → original .less

@debug-tip: "Ship maps in dev only; omit from production builds";
