LESS Snippet

// LESS with Webpack/Vite — Example 03
// Focus: component styling | Scenario: blog redesign

# Vite
# npm i -D less
# import "./styles/main.less"
#
# webpack
# npm i -D less less-loader
# {
#   test: /\.less$/,
#   use: ["style-loader", "css-loader", "less-loader"]
# }
#
# less-loader options:
# { lessOptions: { javascriptEnabled: true, paths: ["src/styles"] } }

@brand: #0d6efd;
.app-shell { border-top: 3px solid @brand; }
