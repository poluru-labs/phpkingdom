// Recursive Mixins — Template 14
// Focus: theme switcher | Scenario: storefront

.cols(@i) when (@i <= 4) {
  .col-@{i} {
    width: percentage(@i / 12);
  }
  .cols(@i + 1);
}
.cols(1);

// Generates .col-1 … .col-4 at compile time
