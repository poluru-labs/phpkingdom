LESS Snippet

// Recursive Mixins — Exercise 08
// Focus: color system | Scenario: analytics page

.cols(@i) when (@i <= 4) {
  .col-@{i} {
    width: percentage(@i / 12);
  }
  .cols(@i + 1);
}
.cols(1);

// Generates .col-1 … .col-4 at compile time
