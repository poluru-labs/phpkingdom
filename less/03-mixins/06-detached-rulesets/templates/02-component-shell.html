.respond-to(@bp; @rules) {
  @media (min-width: @bp) {
    @rules();
  }
}

.hero {
  padding: 2rem;

  .respond-to(768px; {
    padding: 4rem;
    display: grid;
    grid-template-columns: 1fr 1fr;
    gap: 2rem;
  });
}

// @rules is a detached ruleset passed into the mixin
