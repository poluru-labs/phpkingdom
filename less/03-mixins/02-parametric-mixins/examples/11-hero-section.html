LESS Snippet

// Parametric Mixins — Example 11
// Focus: hero section | Scenario: SaaS dashboard

.button(@bg; @color: #fff; @pad: 0.5rem 1rem) {
  background: @bg;
  color: @color;
  padding: @pad;
  border: 0;
  border-radius: 6px;
  cursor: pointer;
}

.btn-primary { .button(#0d6efd); }
.btn-dark { .button(#212529); }
.btn-ghost { .button(transparent; #0d6efd; 0.45rem 0.9rem); }
