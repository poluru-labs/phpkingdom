LESS Snippet

// LESS Namespaces — Example 05
// Focus: navigation | Scenario: portfolio

#ui {
  .button(@bg: #0d6efd) {
    background: @bg;
    color: #fff;
    padding: 0.5rem 1rem;
    border-radius: 6px;
  }

  .card() {
    padding: 1rem;
    border: 1px solid #dee2e6;
    border-radius: 12px;
  }
}

.cta { #ui > .button(); }
.feature { #ui > .card(); }
