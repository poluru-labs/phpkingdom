// Mixin Guards — Exercise 06
// Focus: mixin practice | Scenario: admin panel

.contrast(@bg) when (lightness(@bg) >= 50%) {
  color: #111;
}
.contrast(@bg) when (lightness(@bg) < 50%) {
  color: #fff;
}

.panel-light {
  background: #e7f1ff;
  .contrast(#e7f1ff);
  padding: 1rem;
}

.panel-dark {
  background: #0d6efd;
  .contrast(#0d6efd);
  padding: 1rem;
}
