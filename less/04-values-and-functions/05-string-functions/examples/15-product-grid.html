LESS Snippet

// String Functions — Example 15
// Focus: product grid | Scenario: portfolio

@name: "phpkingdom";
@path: "/assets/logo.svg";

.quote::before {
  content: %("Hello %a", @name);
}

.icon {
  background-image: url("@{path}");
}

.label {
  // replace() swaps substrings at compile time
  content: replace("btn-primary", "btn-", "");
}
