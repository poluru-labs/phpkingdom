// List Functions — Exercise 10
// Focus: layout scaling | Scenario: mobile web app

@spaces: 4px, 8px, 16px, 24px;

.first { padding: extract(@spaces, 1); }
.last { padding: extract(@spaces, length(@spaces)); }

@merged: join(@spaces, 32px, comma);
.wide { margin: extract(@merged, 5); }

// length(), extract(), index(), join() work on comma/space lists
