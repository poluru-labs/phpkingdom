LESS Snippet

// LESS Functions — Example 05
// Focus: navigation | Scenario: portfolio

@w: 320px;
@brand: hsl(210, 100%, 50%);

.box {
  width: percentage(2 / 3);
  min-height: unit(@w, rem); // careful with unit()
  color: hsl(hue(@brand), saturation(@brand), 40%);
  z-index: ceil(2.2);
}
