LESS Snippet

// LESS Maps — Example 25
// Focus: best practice recap | Scenario: portfolio

// LESS uses keyed lists / property bags for map-like tokens
@colors: {
  brand: #0d6efd;
  ink: #212529;
  muted: #6c757d;
};

.each(@colors, {
  .text-@{key} {
    color: @value;
  }
});

// Or simple parallel lists:
@names: brand, success, danger;
@vals: #0d6efd, #198754, #dc3545;
