@brand: #0d6efd;

.btn {
  background: @brand;
  color: #fff;
  border: 1px solid fade(@brand, 40%);

  &:hover { background: lighten(@brand, 8%); }
  &:active { background: darken(@brand, 8%); }
  &:focus { outline: 2px solid spin(@brand, 20); }
}

.tint { background: tint(@brand, 80%); }
.shade { background: shade(@brand, 20%); }
