LESS Snippet

// LESS vs CSS — Example 22
// Focus: marketing block | Scenario: marketing site

/* Plain CSS */
/* .btn { background: #0d6efd; } */
/* .btn:hover { background: #0b5ed7; } */

/* Same idea in LESS — tokens + nesting */
@brand: #0d6efd;

.btn {
  background: @brand;
  &:hover { background: darken(@brand, 8%); }
}

/* LESS also adds mixins, guards, extend, and compile-time math */
