LESS Snippet

// Guard Comparison — Example 20
// Focus: theme variants | Scenario: mobile web app

.pad(@size) when (@size > 16px) {
  padding: @size;
}
.pad(@size) when (@size <= 16px) {
  padding: 8px;
}

.hero { .pad(24px); }
.chip { .pad(12px); }

.tone(@l) when (@l >= 60%) { color: #111; }
.tone(@l) when (@l < 60%) { color: #fff; }
