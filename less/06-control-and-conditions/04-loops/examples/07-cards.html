LESS Snippet

// LESS Loops — Example 07
// Focus: cards | Scenario: booking app

.spacer(@i) when (@i <= 5) {
  .p-@{i} {
    padding: (@i * 0.25rem);
  }
  .spacer(@i + 1);
}
.spacer(1);

@colors: blue, green, red;
.tint-loop(@i: 1) when (@i <= length(@colors)) {
  @c: extract(@colors, @i);
  .tint-@{c} { background: @c; }
  .tint-loop(@i + 1);
}
.tint-loop();
