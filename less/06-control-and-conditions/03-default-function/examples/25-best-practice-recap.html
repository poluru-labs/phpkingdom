.theme(@name) when (@name = dark) {
  background: #212529;
  color: #f8f9fa;
}
.theme(@name) when (@name = brand) {
  background: #0d6efd;
  color: #fff;
}
.theme(@name) when (default()) {
  background: #fff;
  color: #212529;
}

.card { .theme(light); padding: 1rem; }
.banner { .theme(dark); padding: 2rem; }
