LESS Snippet

// LESS Guards — Example 19
// Focus: animation prep | Scenario: documentation site

.button(@bg) when (iscolor(@bg)) {
  background: @bg;
  color: #fff;
}
.button(@bg) when (isstring(@bg)) {
  background: transparent;
  color: #0d6efd;
  border: 1px solid #0d6efd;
}

.primary { .button(#0d6efd); }
.outline { .button("ghost"); }
