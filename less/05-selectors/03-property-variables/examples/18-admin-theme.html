LESS Snippet

// Property Variables — Example 18
// Focus: admin theme | Scenario: analytics page

@prop: border-radius;
@side: left;

.box {
  @{prop}: 12px;
  border-@{side}: 4px solid #0d6efd;
  background: #e7f1ff;
  padding: 1rem;
}

@shadow-prop: box-shadow;
.elevated {
  @{shadow-prop}: 0 8px 24px rgba(0, 0, 0, 0.08);
}
