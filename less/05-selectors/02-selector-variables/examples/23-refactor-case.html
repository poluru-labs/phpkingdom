LESS Snippet

// Selector Variables — Example 23
// Focus: refactor case | Scenario: blog redesign

@name: primary;
@state: hover;

.btn-@{name} {
  background: #0d6efd;
  color: #fff;
  padding: 0.5rem 1rem;

  &:@{state} {
    filter: brightness(1.05);
  }
}

@theme: dark;
body.theme-@{theme} {
  background: #111;
  color: #f8f9fa;
}
