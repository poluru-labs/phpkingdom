.box {

  box-shadow+: inset 0 0 1px #000;
  box-shadow+: 0 8px 24px rgba(0, 0, 0, 0.12);

  // +_ merges into a comma-separated list
  transform+_ : scale(1);
  transform+_ : rotate(0.01turn);
}

// Useful when mixins each append shadow / transition pieces
