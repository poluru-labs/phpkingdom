LESS Snippet

// Theme System — Exercise 10
// Focus: layout scaling | Scenario: mobile web app

@theme: light;

.theme-vars() when (@theme = light) {
  @bg: #ffffff;
  @ink: #212529;
  @brand: #0d6efd;
}
.theme-vars() when (@theme = dark) {
  @bg: #0f172a;
  @ink: #e2e8f0;
  @brand: #6ea8fe;
}
.theme-vars();

body {
  background: @bg;
  color: @ink;
}
a { color: @brand; }
