LESS Snippet

// Dark Mode Tokens — Example 24
// Focus: utility pattern | Scenario: storefront

@brand-light: #0d6efd;
@brand-dark: #6ea8fe;
@bg-light: #f8f9fa;
@bg-dark: #0f172a;
@ink-light: #212529;
@ink-dark: #e2e8f0;

:root {
  --brand: @brand-light;
  --bg: @bg-light;
  --ink: @ink-light;
}

[data-theme="dark"] {
  --brand: @brand-dark;
  --bg: @bg-dark;
  --ink: @ink-dark;
}

body {
  background: var(--bg);
  color: var(--ink);
}
