LESS Snippet

// LESS Extend — Example 21
// Focus: dark surface | Scenario: SaaS dashboard

.message {
  padding: 0.75rem 1rem;
  border-radius: 8px;
  border-left: 4px solid;
}

.alert-success {
  &:extend(.message);
  background: #d1e7dd;
  border-color: #198754;
  color: #0f5132;
}

.alert-danger {
  &:extend(.message);
  background: #f8d7da;
  border-color: #dc3545;
  color: #842029;
}
