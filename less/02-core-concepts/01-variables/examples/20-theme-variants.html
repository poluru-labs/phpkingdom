LESS Snippet

// LESS Variables — Example 20
// Focus: theme variants | Scenario: mobile web app

@brand: #0d6efd;
@space-sm: 0.5rem;
@space-md: 1rem;
@radius: 8px;
@font-stack: system-ui, "Segoe UI", sans-serif;

.card {
  font-family: @font-stack;
  padding: @space-md;
  border-radius: @radius;
  border: 1px solid @brand;
  color: darken(@brand, 30%);
}
