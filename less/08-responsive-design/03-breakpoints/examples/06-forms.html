LESS Snippet

// LESS Breakpoints — Example 06
// Focus: forms | Scenario: admin panel

@bp-sm: 576px;
@bp-md: 768px;
@bp-lg: 992px;
@bp-xl: 1200px;

.above(@bp; @rules) {
  @media (min-width: @bp) { @rules(); }
}

.nav {
  flex-direction: column;
  .above(@bp-md; { flex-direction: row; gap: 1.25rem; });
}

.sidebar {
  display: none;
  .above(@bp-lg; { display: block; width: 240px; });
}
