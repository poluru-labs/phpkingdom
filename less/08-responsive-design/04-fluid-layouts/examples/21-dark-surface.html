LESS Snippet

// Fluid Layouts — Example 21
// Focus: dark surface | Scenario: SaaS dashboard

@min-w: 320px;
@max-w: 1200px;

.fluid-space(@min; @max) {
  // simple clamp-friendly tokens
  padding-inline: clamp(@min, 4vw, @max);
}

.shell {
  .fluid-space(1rem, 2.5rem);
  max-width: @max-w;
  margin-inline: auto;
}

.hero-title {
  font-size: clamp(1.75rem, 2vw + 1rem, 3rem);
}
