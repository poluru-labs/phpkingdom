// Focus: mobile optimization | Scenario: blog redesign

.respond-to(@bp; @rules) {
  @media (min-width: @bp) {
    @rules();
  }
}

.grid {
  display: block;

  .respond-to(768px; {
    display: grid;
    grid-template-columns: 1fr 1fr;
    gap: 1rem;
  });

  .respond-to(992px; {
    grid-template-columns: repeat(3, 1fr);
  });
}
