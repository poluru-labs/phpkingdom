# Pre-Commit Hook — Example 24
# Focus: reflog recover | Scenario: hotfix release

# .git/hooks/pre-commit sketch (pre-commit-hook)
cat > .git/hooks/pre-commit <<'EOF'
#!/bin/sh
if git diff --cached | grep -E 'TODO-SHIP'; then
  echo "Remove TODO-SHIP before commit"
  exit 1
fi
EOF
chmod +x .git/hooks/pre-commit
