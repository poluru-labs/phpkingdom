# Generate SSH Keys — Example 20

# Generate and register an SSH key (ssh-keys)
ssh-keygen -t ed25519 -C "mail.polurus@gmail.com" -f ~/.ssh/id_ed25519_ssh-keys
eval "$(ssh-agent -s)"
ssh-add ~/.ssh/id_ed25519_ssh-keys
cat ~/.ssh/id_ed25519_ssh-keys.pub
