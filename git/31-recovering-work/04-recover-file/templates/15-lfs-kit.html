# Recover a Deleted File — Template 15
# Focus: lfs kit | Scenario: monorepo app

# Recover a deleted file (recover-file)
git log --diff-filter=D --summary -- path/to/file
git checkout <commit-sha>^ -- path/to/file
git commit -m "restore(recover-file): bring back file"
